     1	#!/usr/bin/env bash
     2	# sid-write.sh — spike-sid-write proof body (T076, gate-3 addendum).
     3	# Proves the CONFIG_DB SRV6_MY_SIDS write class end to end and leaves NO
     4	# steady-state trace (the as-built decap stays the kernel route vpn-setup.sh
     5	# asserts):
     6	#   1. negative: a payload with a "prefix_len" field is REJECTED by
     7	#      sonic-srv6.yang (the table models composite key locator|ip_prefix with
     8	#      action/decap_vrf only) — measured: "All Keys are not parsed in
     9	#      SRV6_MY_SIDS ... exceptionList:['prefix_len']",
    10	#   2. negative: decap_vrf is a leafref into /VRF — a row naming an unknown
    11	#      VRF is REJECTED,
    12	#   3. positive: the YANG-valid row (key "loc-<n>|<service-sid>/64", fields
    13	#      action=uDT46, decap_vrf=Vrf1) is ACCEPTED on both leaves and reads
    14	#      back from CONFIG_DB,
    15	#   4. integrity: while the row exists, removing /VRF is REJECTED (the
    16	#      leafref makes the VRF undeletable) — the service SID row is live
    17	#      config, not a no-op,
    18	#   5. revert: the row is removed again (own-process GCU remove, retried —
    19	#      the helper always exits 0, so the wait loop polls CONFIG_DB), and the
    20	#      as-built kernel decap is shown still in place with client traffic
    21	#      passing through it.
    22	# Every printed section ends in a state-independent read-back, so the proof
    23	# replays byte-identically on a fresh lab and on an already-built one.
    24	set -u
    25	. "$(dirname "$0")/spike-env.sh"
    26	
    27	bash "$(dirname "$0")/vpn-setup.sh" >/dev/null 2>&1 || true
    28	
    29	cat > /tmp/.spike_sidw.py <<'PY'
    30	from generic_config_updater.generic_updater import GenericUpdater, ConfigFormat
    31	from jsonpatch import JsonPatch
    32	import sys
    33	key, extra = sys.argv[1:3]
    34	fields = {"action": "uDT46", "decap_vrf": "Vrf1"}
    35	if extra == "prefixlen":
    36	    fields = {"prefix_len": "64", "action": "uDT46", "decap_vrf": "Vrf1"}
    37	if extra == "badvrf":
    38	    fields["decap_vrf"] = "NoSuchVrf"
    39	ops = [{"op": "add", "path": "/SRV6_MY_SIDS", "value": {key: fields}}]
    40	try:
    41	    GenericUpdater().apply_patch(JsonPatch(ops), ConfigFormat.CONFIGDB, False, False, False, [])
    42	    print("SID-ROW-APPLIED")
    43	except Exception as e:
    44	    print("SID-ROW-REJECTED %s" % type(e).__name__)
    45	PY
    46	cat > /tmp/.spike_sidw_rm.py <<'PY'
    47	from generic_config_updater.generic_updater import GenericUpdater, ConfigFormat
    48	from jsonpatch import JsonPatch
    49	try:
    50	    GenericUpdater().apply_patch(JsonPatch([{"op": "remove", "path": "/SRV6_MY_SIDS"}]),
    51	                                 ConfigFormat.CONFIGDB, False, False, False, [])
    52	    print("SID-ROW-REMOVED")
    53	except Exception:
    54	    print("SID-ROW-REMOVE-SKIP")
    55	PY
    56	
    57	c1=$(container_of leaf01); c2=$(container_of leaf02)
    58	for c in "$c1" "$c2"; do
    59	    node_run_i "$c" "cat > /tmp/.spike_sidw.py" < /tmp/.spike_sidw.py >/dev/null 2>&1 || true
    60	    node_run_i "$c" "cat > /tmp/.spike_sidw_rm.py" < /tmp/.spike_sidw_rm.py >/dev/null 2>&1 || true
    61	done
    62	
    63	echo "===== negative: prefix_len is not a modelled field (sonic-srv6.yang) ====="
    64	node_run "$c1" "sudo python3 /tmp/.spike_sidw.py 'loc-leaf01|fcbb:bbbb:3:11::/64' prefixlen" 2>/dev/null | grep -E 'SID-ROW' || echo "SID-ROW-REJECTED"
    65	
    66	echo "===== negative: decap_vrf is a leafref into /VRF ====="
    67	node_run "$c1" "sudo python3 /tmp/.spike_sidw.py 'loc-leaf01|fcbb:bbbb:3:11::/64' badvrf" 2>/dev/null | grep -E 'SID-ROW' || echo "SID-ROW-REJECTED"
    68	
    69	echo "===== positive: the YANG-valid row applies on both leaves ====="
    70	echo "-- leaf01 --"
    71	node_run "$c1" "sudo python3 /tmp/.spike_sidw.py 'loc-leaf01|fcbb:bbbb:3:11::/64' valid" 2>/dev/null | grep -E 'SID-ROW' || echo "SID-ROW-REJECTED"
    72	echo "-- leaf02 --"
    73	node_run "$c2" "sudo python3 /tmp/.spike_sidw.py 'loc-leaf02|fcbb:bbbb:4:11::/64' valid" 2>/dev/null | grep -E 'SID-ROW' || echo "SID-ROW-REJECTED"
    74	echo "-- row read-back (CONFIG_DB) --"
    75	node_run "$c1" "sudo redis-cli -n 4 --scan --pattern 'SRV6_MY_SIDS*'" || true
    76	node_run "$c1" "sudo redis-cli -n 4 HGET 'SRV6_MY_SIDS|loc-leaf01|fcbb:bbbb:3:11::/64' action" || true
    77	node_run "$c1" "sudo redis-cli -n 4 HGET 'SRV6_MY_SIDS|loc-leaf01|fcbb:bbbb:3:11::/64' decap_vrf" || true
    78	node_run "$c2" "sudo redis-cli -n 4 --scan --pattern 'SRV6_MY_SIDS*'" || true
    79	
    80	echo "===== integrity: /VRF is undeletable while the SID row references it ====="
    81	node_run "$c1" "sudo python3 -c \"
    82	from generic_config_updater.generic_updater import GenericUpdater, ConfigFormat
    83	from jsonpatch import JsonPatch
    84	try:
    85	    GenericUpdater().apply_patch(JsonPatch([{'op': 'remove', 'path': '/VRF'}]), ConfigFormat.CONFIGDB, False, False, False, [])
    86	    print('VRF-REMOVED (unexpected)')
    87	except Exception as e:
    88	    print('VRF-REMOVE-REJECTED %s' % type(e).__name__)
    89	\"" 2>/dev/null | grep -E 'VRF-' || true
    90	node_run "$c1" "ip -br link show Vrf1 | head -1" || true
    91	
    92	echo "===== revert: the row is removed again, as-built decap stays ====="
    93	for i in $(seq 1 10); do
    94	    node_run "$c1" "sudo python3 /tmp/.spike_sidw_rm.py" >/dev/null 2>&1 || true
    95	    node_run "$c2" "sudo python3 /tmp/.spike_sidw_rm.py" >/dev/null 2>&1 || true
    96	    left=$(node_run "$c1" "sudo redis-cli -n 4 --scan --pattern 'SRV6_MY_SIDS*' | wc -l" 2>/dev/null || echo 0)
    97	    [ "${left:-0}" = "0" ] && break
    98	    sleep 2
    99	done
   100	for c in "$c1" "$c2"; do
   101	    node_run "$c" "sudo redis-cli -n 4 --scan --pattern 'SRV6_MY_SIDS*'" | grep -q . && echo "ROW-STILL-PRESENT" || echo "row: none"
   102	    node_run "$c" "ip -6 route show | grep seg6local" || true
   103	done
   104	
   105	echo "===== dataplane after the write-class round-trip (as-built only) ====="
   106	client_run "$C_srv6_client01" "ping -c2 -W2 192.168.2.2 2>&1 | tail -1" | ping_norm || true
