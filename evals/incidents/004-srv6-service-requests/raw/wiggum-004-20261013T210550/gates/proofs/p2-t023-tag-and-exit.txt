    81	# T023/T598 (FR-009): the sonic-vm topology names its image by TAG because
    82	# containerlab cannot resolve a local image by digest without a registry
    83	# (proof p1-image-by-digest: clab normalizes name@sha256:<digest> to a
    84	# registry reference and pulls). So the sonic-vm bring-up asserts, BEFORE
    85	# deploy, that the tagged image's ID equals the digest pinned in

    87	# otherwise — the tag is the known exception, the digest is still the pin.
    88	if [[ "${AGENTIC_NETOPS_PROFILE}" == "sonic-vm" ]]; then

    90	    $1=="sonic_images:" {sec=1; next}
    91	    sec && /^[^[:space:]]/ {exit}
    92	    sec && $1=="sonic_vm:" {vm=1; next}
    93	    sec && vm && /^[^[:space:]]/ {vm=0}
    94	    sec && vm && $1=="image:" {print $2; exit}

    96	  if [[ ! "$pinned_svm" =~ ^local/sonic-vm@sha256:[0-9a-f]{64}$ ]]; then
    97	    echo "[provision] versions.lock.yaml sonic_vm must pin one line local/sonic-vm@sha256:<digest>; got: ${pinned_svm:-<missing>}" >&2
    98	    exit 1
    99	  fi
