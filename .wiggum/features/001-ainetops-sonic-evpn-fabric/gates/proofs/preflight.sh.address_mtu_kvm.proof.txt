66\tpreflight::address_conflicts() {

72\t  read -r p1 p2 < <(preflight::cidr_range "$pod_cidr")
73\t  read -r s1 s2 < <(preflight::cidr_range "$svc_cidr")
74\t  if preflight::ranges_overlap "$m1" "$m2" "$p1" "$p2"; then

76\t  fi

78\t    preflight::die "Management CIDR $mgmt_cidr overlaps Service CIDR $svc_cidr"
79\t  fi
