     4	import (
     5		"fmt"
     6		"strings"
     7	)
     8	
     9	// OfflineValidate performs a lightweight schema/path shape validation without contacting SDC.
    10	// It rejects specs that contain keys that are not absolute gNMI-like paths (must start with '/').
    11	// Future work: integrate with pinned SDC offline validator and SONiC/OpenConfig YANG models.
    12	func OfflineValidate(spec map[string]any) error {
    13		for k := range spec {
    14			if !strings.HasPrefix(k, "/") {
    15				return fmt.Errorf("invalid rendered path (must start with '/'): %s", k)
    16			}
    17		}
    18		return nil
    19	}
