     1	#!/usr/bin/env bash
     2	set -euo pipefail
     3	
     4	# AINETOPS SONiC EVPN/VXLAN Fabric — shutdown/cleanup script (Phase 8)
     5	# Sole implementation of environment teardown per contracts/crd-api.md
     6	
     7	SCRIPT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" &>/dev/null && pwd)
     8	LIB_DIR="${SCRIPT_DIR}/lib"
     9	
    10	# Flags
    11	AINETOPS_CLUSTER_NAME=${AINETOPS_CLUSTER_NAME:-ainetops}
    12	DELETE_KIND=${AINETOPS_DELETE_KIND:-false}
    13	CAPTURE_EVIDENCE=${AINETOPS_CAPTURE_EVIDENCE:-false}
    14	
    15	usage() {
    16	  cat <<EOF
    17	Usage: $0 [--cluster-name NAME] [--delete-kind true|false] [--capture-evidence true|false]
    18	EOF
    19	}
    20	
    21	while [[ $# -gt 0 ]]; do
    22	  case "$1" in
    23	    --cluster-name) shift; AINETOPS_CLUSTER_NAME=${1:-$AINETOPS_CLUSTER_NAME} ;;
    24	    --delete-kind) shift; DELETE_KIND=${1:-$DELETE_KIND} ;;
    25	    --capture-evidence) shift; CAPTURE_EVIDENCE=${1:-$CAPTURE_EVIDENCE} ;;
    26	    -h|--help) usage; exit 0 ;;
    27	    *) echo "[off] unknown flag: $1" >&2; usage; exit 2 ;;
    28	  esac
    29	  shift || true
    30	done
    31	export AINETOPS_CLUSTER_NAME
    32	
    33	# Optional evidence capture
    34	if [[ "$CAPTURE_EVIDENCE" == "true" ]]; then
    35	  proofs="${SCRIPT_DIR}/../.wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs"
    36	  mkdir -p "$proofs"
    37	  if command -v kubectl >/dev/null 2>&1; then
    38	    CTX="kind-${AINETOPS_CLUSTER_NAME}"
    39	    kubectl --context "$CTX" get pods -A -o wide | nl -ba > "$proofs/kubectl-get-all-before-off.txt" || true
    40	    kubectl --context "$CTX" get crds | nl -ba > "$proofs/kubectl-get-crds-before-off.txt" || true
    41	  fi
    42	fi
    43	
    44	# Remove generated Secrets from the local repo BEFORE containerlab destroy:
    45	# the destroy helper verifies the lab left no generated credentials behind, so
    46	# the lifecycle-owned removal must happen first (the test phase may have
    47	# materialized ./secrets/* from in-cluster Secrets).
    48	rm -f "${SCRIPT_DIR}/../secrets/gnmi.key" "${SCRIPT_DIR}/../secrets/gnmi.crt" "${SCRIPT_DIR}/../secrets/ca.crt" 2>/dev/null || true
    49	
    50	# Idempotent containerlab teardown and cleanup checks
    51	if [[ -x "${LIB_DIR}/containerlab.sh" ]]; then
    52	  "${LIB_DIR}/containerlab.sh" destroy || {
    53	    echo "[off] containerlab destroy reported leftovers" >&2
    54	    exit 1
    55	  }
    56	else
    57	  echo "[off] WARN: containerlab helper not found; skipping lab teardown" >&2
    58	fi
    59	
    60	# Optionally delete Kind cluster if requested
    61	if [[ "$DELETE_KIND" == "true" && -x "${LIB_DIR}/kind.sh" ]]; then
    62	  "${LIB_DIR}/kind.sh" delete
    63	fi
    64	
    65	# Cleanup owned network and generated Secrets (safe, scoped)
    66	if command -v docker >/dev/null 2>&1; then
    67	  if docker network inspect ainetops-mgmt >/dev/null 2>&1; then
    68	    # Do not remove if foreign label
    69	    if docker network inspect ainetops-mgmt -f '{{json .Labels}}' | grep -q '"ainetops.owner":"ainetops"'; then
    70	      docker network rm ainetops-mgmt >/dev/null 2>&1 || true
    71	    else
    72	      echo "[off] preserving non-owned network ainetops-mgmt" >&2
    73	    fi
    74	  fi
    75	fi
    76	
    77	# Remove generated Secrets from local repo if present
    78	rm -f "${SCRIPT_DIR}/../secrets/gnmi.key" "${SCRIPT_DIR}/../secrets/gnmi.crt" "${SCRIPT_DIR}/../secrets/ca.crt" 2>/dev/null || true
    79	
    80	echo "[off] Teardown complete (idempotent)."
