     4	import (
     5		"encoding/json"
     6		"fmt"
     7	)
     8	
     9	// KubenetNetwork is a minimal upstream Network spec shape we generate deterministically.
    10	// We only model the fields we produce; the CRD allows unknown fields and KUID fills the rest.
    11	type KubenetNetwork struct {
    12		APIVersion string            `json:"apiVersion"`
    13		Kind       string            `json:"kind"`
    14		Metadata   map[string]any    `json:"metadata"`
    15		Spec       map[string]any    `json:"spec"`
    16	}
    17	
    18	// OutputBundle captures deterministic artifacts and provenance for audit.
    19	type OutputBundle struct {
    20		Translator   string            `json:"translator"`
    21		Version      string            `json:"version"`
    22		Mapping      string            `json:"mapping"`
    23		InputHash    string            `json:"inputHash"`
    24		Annotations  map[string]string `json:"annotations"`
    25		NetworkYAML  string            `json:"networkYAML"`
    26	}
    27	
    28	// Translate produces a Kubenet Network with deterministic fields and annotations.
    29	// It assumes ValidateAllOrNothing has already passed for 'in'.
    30	func Translate(in *ServiceInput) (*OutputBundle, error) {
    31		if in == nil {
    32			return nil, fmt.Errorf("nil input")
    33		}
    34		hash, err := in.CanonicalHash()
    35		if err != nil {
    36			return nil, err
    37		}
    38	
    39		name := fmt.Sprintf("migr-%s", in.ServiceID)
    40		metadata := map[string]any{
    41			"name":      name,
    42			"namespace": "kubenet-system",
    43			"annotations": map[string]string{
    44				"ainetops.io/translator":          TranslatorName,
    45				"ainetops.io/translator-version":  TranslatorVersion,
    46				"ainetops.io/mapping-version":     MappingVersion,
    47				"ainetops.io/migration-input-hash": hash,
    48				"ainetops.io/tenant":              in.Tenant,
    49				"ainetops.io/service-type":        string(in.Type),
    50			},
    51		}
    52	
    53		// Spec map assembly is type-dependent but produces a stable ordering when encoded.
    54		spec := map[string]any{
    55			"description": fmt.Sprintf("Migrated service %s (%s)", in.ServiceID, in.Type),
    56		}
    57		switch in.Type {
    58		case ServiceVPLS, ServiceVPWS:
    59			// bridgeDomains + attachments (VLANs)
    60			bd := map[string]any{
    61				"name":  fmt.Sprintf("bd-%s", in.ServiceID),
    62				"vlan":  endpointVLAN(in.Endpoints),
    63				"l2vni": in.L2VNI,
    64				"evpn": map[string]any{
    65					"routeTargets": map[string]any{
    66						"import": in.RDRT.ImportRT,
    67						"export": in.RDRT.ExportRT,
    68					},
    69				},
    70			}
    71			spec["bridgeDomains"] = []any{bd}
    72			spec["attachments"] = attachmentsForL2(in.Endpoints)
    73			if in.Type == ServiceVPWS {
    74				ann := metadata["annotations"].(map[string]string)
    75				ann["ainetops.io/limited-equivalence"] = "vpws-to-l2vni"
    76			}
    77		case ServiceL3VPN:
    78			// routers + attachments (VRF)
    79			r := map[string]any{
    80				"name": fmt.Sprintf("vrf-%s", in.ServiceID),
    81				"rd":   in.RDRT.RD,
    82				"routeTargets": map[string]any{
    83					"import": in.RDRT.ImportRT,
    84					"export": in.RDRT.ExportRT,
    85				},
    86				"l3vni":    in.L3VNI,
    87				"prefixes": concatPrefixes(in.AF),
    88			}
    89			spec["routers"] = []any{r}
    90			spec["attachments"] = attachmentsForL3(in.Endpoints, r["name"].(string))
    91		case ServiceIRB:
    92			// routers + bridgeDomains with irb
    93			r := map[string]any{
    94				"name": fmt.Sprintf("vrf-%s", in.ServiceID),
    95				"rd":   in.RDRT.RD,
    96				"routeTargets": map[string]any{
    97					"import": in.RDRT.ImportRT,
    98					"export": in.RDRT.ExportRT,
    99				},
   100				"l3vni": in.L3VNI,
   101			}
   102			bd := map[string]any{
   103				"name":  fmt.Sprintf("bd-%s", in.ServiceID),
   104				"vlan":  endpointVLAN(in.Endpoints),
   105				"l2vni": in.L2VNI,
   106				"irb": map[string]any{
   107					"vrf":         in.IRBGateway.VRF,
   108					"gatewayIPv4": in.IRBGateway.GatewayV4,
   109					"gatewayIPv6": in.IRBGateway.GatewayV6,
   110				},
   111			}
   112			spec["routers"] = []any{r}
   113			spec["bridgeDomains"] = []any{bd}
   114			spec["attachments"] = attachmentsForL2(in.Endpoints)
   115		default:
   116			return nil, fmt.Errorf("unsupported type: %s", in.Type)
   117		}
   118	
   119		net := KubenetNetwork{
   120			APIVersion: "network.kubenet.dev/v1alpha1",
   121			Kind:       "Network",
   122			Metadata:   metadata,
   123			Spec:       spec,
   124		}
   125		// Deterministic YAML: we marshal to JSON then convert to YAML using Kubernetes YAML lib at fixed version.
   126		j, err := json.Marshal(net)
   127		if err != nil { return nil, err }
   128		yml, err := jsonToYAML(j)
   129		if err != nil { return nil, err }
   130	
   131		out := &OutputBundle{
   132			Translator:  TranslatorName,
   133			Version:     TranslatorVersion,
   134			Mapping:     MappingVersion,
   135			InputHash:   hash,
   136			Annotations: metadata["annotations"].(map[string]string),
   137			NetworkYAML: string(yml),
   138		}
   139		return out, nil
   140	}
   141	
   142	func endpointVLAN(eps []Endpoint) int {
   143		if len(eps) == 0 { return 0 }
   144		return eps[0].VLAN
   145	}
   146	
   147	func attachmentsForL2(eps []Endpoint) []any {
   148		var out []any
   149		for _, ep := range eps {
   150			out = append(out, map[string]any{
   151				"node":       ep.Node,
   152				"vlan":       ep.VLAN,
   153				"attachment": ep.Attachment,
   154			})
   155		}
   156		return out
   157	}
   158	
   159	func attachmentsForL3(eps []Endpoint, vrf string) []any {
   160		var out []any
   161		for _, ep := range eps {
   162			out = append(out, map[string]any{
   163				"node":       ep.Node,
   164				"vrf":        vrf,
   165				"attachment": ep.Attachment,
   166			})
   167		}
   168		return out
   169	}
   170	
   171	func concatPrefixes(af *AddressFamilies) []string {
   172		var out []string
   173		if af == nil { return out }
   174		out = append(out, af.IPv4Prefixes...)
   175		out = append(out, af.IPv6Prefixes...)
   176		return out
   177	}
