    84	
    85	# The live fabric verification is only meaningful against an SRv6-qualified lab
    86	# (the capability gate in scripts/lib/qualify.sh is the single source of truth for
    87	# that). When the selected profile has not passed the gate — e.g. the fast sonic-vs
    88	# profile on a host without a qualified image — the SONiC gNMI endpoints are not
    89	# expected to be reachable, and this suite must SKIP (not fail) so it is never
    90	# mistaken for a passing fabric result. We probe a TCP connect to each leaf gNMI
    91	# port; if none accepts, the lab is not qualified and we skip with a clear marker.
    92	lab_gnmi_reachable() {
    93	  local t h p
    94	  IFS=',' read -ra tgts <<<"$LEAVES"
    95	  for t in "${tgts[@]}"; do
    96	    h=${t%:*}; p=${t#*:}
    97	    # bash /dev/tcp probe with a bounded timeout; fd closes with the subshell
    98	    if timeout 3 bash -c "exec 3<>/dev/tcp/${h}/${p}" 2>/dev/null; then
    99	      return 0
   100	    fi
   101	  done
   102	  return 1
   103	}

   105	run_each() {
   106	  local targets_csv=$1; shift
   107	  IFS=',' read -ra tgts <<<"$targets_csv"
   108	  local rc=0
   109	  for t in "${tgts[@]}"; do
   110	    if ! "$GNMIC_BIN" --address "$t" "${_args_common[@]}" "$@"; then rc=1; fi
   111	  done
   112	  return $rc
