     1	#!/usr/bin/env bash
     2	# Assert installed AINETOPS-owned CRDs (T079a): exactly SRv6Service.ainetops.io
     3	set -euo pipefail
     4	
     5	CTX="kind-${AINETOPS_CLUSTER_NAME:-ainetops}"
     6	
     7	# Get all CRD names in the cluster
     8	all_crds=$(kubectl --context "$CTX" get crds -o jsonpath='{.items[*].metadata.name}' 2>/dev/null || true)
     9	
    10	# AINETOPS-owned must contain exactly srv6services.ainetops.io
    11	owned_want=(srv6services.ainetops.io)
    12	
    13	# Build found list filtered by group ainetops.io
    14	owned_found=()
    15	for n in $all_crds; do
    16	  if [[ "$n" == *.ainetops.io ]]; then owned_found+=("$n"); fi
    17	done
    18	
    19	# If MigrationPlan was explicitly enabled by T060 in future, allow it; otherwise enforce single CRD
    20	allow_migration=${AINETOPS_ALLOW_MIGRATIONPLAN:-false}
    21	
    22	ok=true
    23	if [[ "$allow_migration" == "true" ]]; then
    24	  # Allow two CRDs at most
    25	  for n in "${owned_found[@]}"; do
    26	    if [[ "$n" != "srv6services.ainetops.io" && "$n" != "migrationplans.ainetops.io" ]]; then
    27	      ok=false
    28	    fi
    29	  done
    30	else
    31	  # Must be exactly one and match desired
    32	  if [[ ${#owned_found[@]} -ne 1 || "${owned_found[0]}" != "srv6services.ainetops.io" ]]; then
    33	    ok=false
    34	  fi
    35	fi
    36	
    37	if ! $ok; then
    38	  echo "[assert-crds] ERROR: AINETOPS-owned CRD set invalid: ${owned_found[*]:-<none>}" >&2
    39	  exit 1
    40	fi
    41	
    42	echo "[assert-crds] OK: AINETOPS-owned CRDs = ${owned_found[*]}"
