    13	func TestGolden_VPLS(t *testing.T) {
    14		b, err := os.ReadFile(filepath.Join("testdata", "migration", "supported_vpls.json"))
    15		if err != nil { t.Fatal(err) }
    16		inputs, err := migration.ParseStrictBatch(b)
    17		if err != nil { t.Fatalf("parse: %v", err) }
    18		outs, err := migration.RenderBatch(inputs)
    19		if err != nil { t.Fatalf("render: %s", migration.MarshalError(err)) }
    20		if len(outs) != 1 { t.Fatalf("expected 1 doc, got %d", len(outs)) }
    21		// Compare only the spec subtree for stability.
    22		spec := extractYAMLSnippet(outs[0], "spec:")
    23		gold, _ := os.ReadFile(filepath.Join("testdata", "migration", "supported_vpls.spec.golden.yaml"))
    24		if strings.TrimSpace(spec) != strings.TrimSpace(string(gold)) {
    25			// optional: write a diff for debugging
    26			_ = os.WriteFile(filepath.Join(os.TempDir(), "vpls.spec.actual.yaml"), []byte(spec), 0o644)
    27			t.Fatalf("golden mismatch for VPLS spec")
    28		}
    29	}
    30	
    31	// extractYAMLSnippet returns the YAML starting at the given key name to EOF.
    32	func extractYAMLSnippet(doc, start string) string {
    33		idx := strings.Index(doc, start)
    34		if idx < 0 { return "" }
    35		return doc[idx:]
    36	}
