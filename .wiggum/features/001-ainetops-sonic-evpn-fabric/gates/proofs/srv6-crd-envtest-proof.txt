    21	// TestSRv6ServiceCRD_Envtest installs the CRD into a test API server and validates sample CRs
    22	// using server-side dry-run (positive and negative cases).
    23	func TestSRv6ServiceCRD_Envtest(t *testing.T) {
    24		t.Parallel()
    25		var (
    26			ctx = context.TODO()
    27		)
    28	
    29		// Start the envtest API server
    30		testEnv := &envtest.Environment{
    31			CRDDirectoryPaths: []string{
    32				filepath.Join("..", "..", "config", "crd", "bases"),
    33			},
    34		}
    35		cfg, err := testEnv.Start()
    36		if err != nil {
    37			t.Fatalf("failed to start envtest: %v", err)
    38		}
    39		t.Cleanup(func() {
    40			_ = testEnv.Stop()
    41		})
    42	
    43		// Ensure apiextensions scheme is registered for dry-run operations
    44		_ = apiextensionsv1.AddToScheme(runtime.NewScheme())
    45	
    46		dc, err := dynamic.NewForConfig(cfg)
    47		if err != nil {
    48			t.Fatalf("dynamic client: %v", err)
    49		}
    50	
    51		gvr := schemaFor("ainetops.io", "v1alpha1", "srv6services")
    52	
    53		// Positive sample
    54		posYAML, err := os.ReadFile(filepath.Join("..", "..", "config", "samples", "ainetops_v1alpha1_srv6service.yaml"))
    55		if err != nil {
    56			t.Fatalf("read sample: %v", err)
    57		}
    58		posObj := mustYAMLToUnstructured(t, posYAML)
    59		posObj.SetNamespace("default")
    60		// Dry-run create should succeed
    61		if _, err := dc.Resource(gvr).Namespace("default").Create(ctx, posObj, metav1.CreateOptions{DryRun: []string{"All"}}); err != nil {
    62			t.Fatalf("dry-run create positive sample failed: %v", err)
    63		}
    64	
    65		// Negative sample: duplicate attachments (violates CEL uniqueness rule)
    66		neg := posObj.DeepCopy()
    67		att, _, _ := unstructured.NestedSlice(neg.Object, "spec", "attachments")
    68		if len(att) != 2 {
    69			t.Fatalf("unexpected attachments len in sample: %d", len(att))
    70		}
    71		// make both nodes equal to violate uniqueness
    72		_ = unstructured.SetNestedField(att[1].(map[string]any), att[0].(map[string]any)["node"], "node")
    73		_ = unstructured.SetNestedSlice(neg.Object, att, "spec", "attachments")
    74		if _, err := dc.Resource(gvr).Namespace("default").Create(ctx, neg, metav1.CreateOptions{DryRun: []string{"All"}}); err == nil {
    75			t.Fatalf("expected dry-run create to fail for duplicate attachments, got nil error")
    76		}
    77	}

    79	func mustYAMLToUnstructured(t *testing.T, y []byte) *unstructured.Unstructured {
    80		t.Helper()
    81		dec := yaml.NewDecodingSerializer(unstructured.UnstructuredJSONScheme)
    82		obj := &unstructured.Unstructured{}
    83		_, _, err := dec.Decode(y, nil, obj)
    84		if err != nil {
    85			t.Fatalf("decode yaml: %v", err)
    86		}
    87		return obj
    88	}
