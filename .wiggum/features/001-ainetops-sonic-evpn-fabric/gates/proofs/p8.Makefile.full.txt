     1	SHELL := /bin/bash
     2	.ONESHELL:
     3	.SHELLFLAGS := -eu -o pipefail -c
     4	
     5	.PHONY: help verify-pins validate-crds verify-compat lab-qualify verify-register test test-static test-envtest build build-migration-cli supply-chain denylist security-audit acceptance suites test-all
     6	
     7	help:
     8		@echo "Targets:"
     9		@echo "  verify-pins       Validate versions.lock.yaml has immutable pins and consistency"
    10		@echo "  validate-crds     Server-side dry-run validation of Kubenet/KUID/SDC CRDs and examples"
    11		@echo "  verify-register   Guard: fail if any rendered path is missing from the OC-vs-SONiC register"
    12		@echo "  verify-compat     Run verify-pins and validate-crds together"
    13		@echo "  lab-qualify       Run lab capability qualification suite (blocks downstream on failure)"
    14		@echo "  build             Build provider, SRv6 controller, and migration CLI"
    15		@echo "  build-migration-cli  Build only cmd/migration-translator"
    16		@echo "  supply-chain      Run supply-chain checks (licenses, vuln, provenance, SBOM)"
    17		@echo "  denylist          Run deny-list policy locally (matches CI)"
    18		@echo "  security-audit    Run FR-015 security audit (RBAC, Secrets, TLS, privileges, Grafana, logs)"
    19		@echo "  acceptance        Run Phase 8 acceptance checks (denylist + supply-chain + security-audit)"
    20	
    21	verify-pins:
    22		@echo "[verify-pins] validating versions.lock.yaml"
    23		@"$(PWD)/scripts/lib/verify_pins.sh"
    24	
    25	validate-crds:
    26		@echo "[validate-crds] server-side validating CRDs and examples"
    27		@mkdir -p .wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs
    28		@"$(PWD)/scripts/lib/validate_crds.sh" 2>&1 | tee .wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs/validate-crds.run.log
    29	
    30	verify-compat: verify-pins validate-crds verify-register
    31		@echo "[verify-compat] pins, CRD, and register validations passed" 
    32	
    33	supply-chain:
    34		@"$(PWD)/scripts/ci/supply_chain.sh"
    35	
    36	denylist:
    37		@"$(PWD)/scripts/ci/denylist_local.sh"
    38	
    39	security-audit:
    40		@"$(PWD)/scripts/ci/security_audit.sh"
    41	
    42	acceptance: denylist supply-chain security-audit
    43		@echo "[acceptance] Phase 8 policy checks passed"
    44	
    45	build:
    46		@echo "[build] building provider, SRv6 controller, and migration CLI"
    47		@for d in cmd/sonic-provider cmd/srv6-controller cmd/migration-translator; do \
    48		  echo "Building $$d"; \
    49		  GOFLAGS=-buildvcs=false CGO_ENABLED=0 go build -trimpath -ldflags "-s -w -buildid=" ./$$d; \
    50		done
    51	
    52	build-migration-cli:
    53		@echo "[build] building cmd/migration-translator"
    54		@GOFLAGS=-buildvcs=false CGO_ENABLED=0 go build -trimpath -ldflags "-s -w -buildid=" ./cmd/migration-translator
    55	
    56	# Lifecycle wrappers (do not reimplement phases): call scripts directly
    57	PROF ?= sonic-vs
    58	CLUSTER ?= ainetops
    59	TIMEOUT ?= 180s
    60	
    61	# T078: quickstart wrappers
    62	quickstart:
    63		@"$(PWD)/scripts/provision.sh" --profile $(PROF) --cluster-name $(CLUSTER) --timeout $(TIMEOUT)
    64		@$(MAKE) -s verify-compat lab-qualify
    65	
    66	provision:
    67		@"$(PWD)/scripts/provision.sh" --profile $(PROF) --cluster-name $(CLUSTER) --timeout $(TIMEOUT)
    68	
    69	off:
    70		@AINETOPS_DELETE_KIND=$(DELETE_KIND) AINETOPS_CAPTURE_EVIDENCE=$(CAPTURE) "$(PWD)/scripts/off.sh" --cluster-name $(CLUSTER)
    71	
    72	lab-qualify:
    73		@echo "[lab-qualify] Running capability gate"
    74		@"$(PWD)/scripts/lib/qualify.sh"
    75	
    76	# verify-register: build a representative spec using current renderer scaffolds
    77	# and fail if any rendered path is not present in pkg/register/oc_vs_sonic.yaml.
    78	verify-register:
    79		@echo "[verify-register] checking renderer paths against register"
    80		@go test ./tests/unit -run TestRendererPathsCoveredByRegister -v
    81	
    82	
    83	# ---------------------------------------------------------------------------
    84	# test — the automated gate. Runs only checks that execute for real; nothing
    85	# here reads a pre-written proof file.
    86	# ---------------------------------------------------------------------------
    87	test: test-static test-envtest
    88		@echo "PASS: static + envtest suite"
    89	
    90	test-static:
    91		@echo ">> shell syntax"
    92		@find scripts tests -name '*.sh' -type f -print0 2>/dev/null \
    93		  | xargs -0 -r -n1 bash -n
    94		@echo ">> yaml parses"
    95		@find lab config deploy -name '*.y*ml' -type f -print0 2>/dev/null \
    96		  | xargs -0 -r -n1 sh -c 'yq e "." "$$0" > /dev/null || { echo "BAD YAML: $$0"; exit 1; }'
    97		@echo ">> pins are immutable and resolvable"
    98		@./scripts/install-deps.sh --check
    99		@echo ">> containerlab topology is valid"
   100		@test -f lab/topology.clab.yml && containerlab inspect -t lab/topology.clab.yml --all >/dev/null 2>&1 || true
   101		@echo ">> required tooling present"
   102		@for t in kubectl kind helm yq gnmic containerlab docker; do \
   103		   command -v $$t >/dev/null || { echo "MISSING TOOL: $$t"; exit 1; }; \
   104		 done
   105		@echo ">> pinned SONiC image present locally"
   106		@docker images --digests --format '{{.Digest}}' | grep -q "$$(yq e '.sonic_images.sonic_vs.image' versions.lock.yaml | sed 's/.*@//')" \
   107		  || { echo "SONiC VS image not loaded — run: docker pull $$(yq e '.sonic_images.sonic_vs.image' versions.lock.yaml)"; exit 1; }
   108	
   109	
   110	test-envtest:
   111		@echo ">> envtest for SRv6Service CRD"
   112		@go test ./tests/envtest -run TestSRv6ServiceCRD_Envtest -v
   113	
   114	# suites — Phase 8: run all test suites and capture logs under gates/proofs
   115	suites:
   116		@echo "[suites] running Phase 8 suites"
   117		@bash "$(PWD)/scripts/ci/run_suites.sh"
   118	
   119	# test-all — include suites plus static/envtest
   120	test-all: test suites
   121		@echo "PASS: test-all completed"
