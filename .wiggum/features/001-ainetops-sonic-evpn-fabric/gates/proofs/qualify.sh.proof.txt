     1	#!/usr/bin/env bash
     2	# Lab qualification harness (Phase 2): run capability tests and gate downstream
     3	set -euo pipefail
     4	
     5	ROOT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/../.." && pwd)
     6	REPORT_DIR="${ROOT_DIR}/.wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs"
     7	mkdir -p "$REPORT_DIR"
     8	
     9	run_test() {
    10	  local name=$1
    11	  echo "[qualify] Running $name" | tee -a "$REPORT_DIR/qualify.run.log"
    12	  "${ROOT_DIR}/tests/integration/sonic_gnmi_suite.sh" --run "$name" 2>&1 | tee -a "$REPORT_DIR/qualify.$name.out.log"
    13	}
    14	
    15	main() {
    16	  local failed=0
    17	  local -a all_tests=()
    18	  local -a all_statuses=()
    19	  # gNMI core tests
    20	  for t in Capabilities Get Set Subscribe sonic-srv6 persistent; do
    21	    all_tests+=("$t")
    22	    if run_test "$t"; then all_statuses+=("pass"); else all_statuses+=("fail"); failed=1; fi
    23	  done
    24	  # EVPN/SRv6 capability coverage
    25	  for t in EVPN-Type2 EVPN-Type3 EVPN-Type5 SRv6-Underlay H.Encaps.Red End End.DT46 SID-list Decapsulation Counters; do
    26	    echo "[qualify] Running $t"
    27	    all_tests+=("$t")
    28	    if "${ROOT_DIR}/tests/integration/evpn_srv6_suite.sh" --run "$t" 2>&1 | tee -a "$REPORT_DIR/qualify.$t.out.log"; then all_statuses+=("pass"); else all_statuses+=("fail"); failed=1; fi
    29	  done
    30	  # Required OpenConfig/SONiC YANG path qualification
    31	  all_tests+=("YANG-Paths")
    32	  if "${ROOT_DIR}/tests/integration/yang_paths_suite.sh" --run YANG-Paths 2>&1 | tee -a "$REPORT_DIR/qualify.YANG-Paths.out.log"; then all_statuses+=("pass"); else all_statuses+=("fail"); failed=1; fi
    33	  # Write machine-readable report
    34	  {
    35	    echo '{"tests":['
    36	    local sep=""; local i
    37	    for ((i=0; i<${#all_tests[@]}; i++)); do
    38	      printf '%s{"name":"%s","status":"%s"}' "$sep" "${all_tests[$i]}" "${all_statuses[$i]}"
    39	      sep=",";
    40	    done
    41	    if (( failed != 0 )); then
    42	      echo '],"result":"fail"}'
    43	    else
    44	      echo '],"result":"pass"}'
    45	    fi
    46	  } > "$REPORT_DIR/qualify.report.json"
    47	  if (( failed != 0 )); then
    48	    echo "[qualify] FAILED: one or more capability tests did not pass" | tee -a "$REPORT_DIR/qualify.run.log"
    49	    exit 1
    50	  fi
    51	  echo "[qualify] OK" | tee -a "$REPORT_DIR/qualify.run.log"
    52	}
    53	
    54	main "$@"
