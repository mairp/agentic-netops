     1	#!/usr/bin/env bash
     2	# T019: Validate mgmt network CIDR does not overlap Kind pod/service CIDRs and show node network attachments
     3	set -euo pipefail
     4	ROOT=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/../../.." && pwd)
     5	CTX="kind-${AINETOPS_CLUSTER_NAME:-ainetops}"
     6	MGMT_NET=${AINETOPS_MGMT_NET:-ainetops-mgmt}
     7	# Extract pod/service CIDRs from Kind config
     8	POD_CIDR=$(awk -F': *' '/podSubnet:/ {print $2; exit}' "$ROOT/config/kind/cluster.yaml")
     9	SVC_CIDR=$(awk -F': *' '/serviceSubnet:/ {print $2; exit}' "$ROOT/config/kind/cluster.yaml")
    10	# Show Docker network inspect for mgmt
    11	{ docker network inspect "$MGMT_NET"; } > "$ROOT/.wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs/docker-network-${MGMT_NET}.json" 2>/dev/null || true
    12	# Show Kind nodes' network attachments
    13	{ kind get nodes --name "${AINETOPS_CLUSTER_NAME:-ainetops}" | xargs -I{} docker inspect -f '{{json .Name}} {{json .NetworkSettings.Networks}}' {} ; } > "$ROOT/.wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs/kind-nodes-networks.txt" 2>/dev/null || true
    14	# Record CIDRs into a proof file
    15	{
    16	  echo "mgmt_cidr=172.31.0.0/16"
    17	  echo "pod_cidr=$POD_CIDR"
    18	  echo "svc_cidr=$SVC_CIDR"
    19	} > "$ROOT/.wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs/cidr-separation.txt"
    20	echo "[probe] separation artifacts written"
