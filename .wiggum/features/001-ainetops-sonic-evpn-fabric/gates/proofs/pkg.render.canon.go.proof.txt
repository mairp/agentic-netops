4	import (
5		"crypto/sha256"
6		"encoding/hex"
7		"encoding/json"
8		"sort"
9		"strings"
10	)
11	
12	// CanonicalJSON marshals a map[string]any into a canonical JSON string with
13	// stable key ordering so equal logical content yields byte-identical output.
14	// We implement a deterministic serializer that sorts map keys recursively.
15	func CanonicalJSON(spec map[string]any) (string, error) {
16		b, err := marshalCanonical(spec)
17		if err != nil {
18			return "", err
19		}
20		return string(b), nil
21	}
22	
23	// CanonicalHash returns a hex-encoded SHA-256 of the canonical JSON.
24	func CanonicalHash(spec map[string]any) (string, error) {
25		j, err := CanonicalJSON(spec)
26		if err != nil { return "", err }
27		s := sha256.Sum256([]byte(j))
28		return hex.EncodeToString(s[:]), nil
29	}
30	
31	// marshalCanonical encodes v into JSON with deterministic map-key ordering.
32	func marshalCanonical(v any) ([]byte, error) {
33		switch x := v.(type) {
34		case map[string]any:
35			keys := make([]string, 0, len(x))
36			for k := range x { keys = append(keys, k) }
37			sort.Strings(keys)
38			parts := make([]string, 0, len(keys))
39			for _, k := range keys {
40				vb, err := marshalCanonical(x[k])
41				if err != nil { return nil, err }
42				kb, _ := json.Marshal(k)
43				parts = append(parts, string(kb)+":"+string(vb))
44			}
45			return []byte("{" + strings.Join(parts, ",") + "}"), nil
46		case []any:
47			parts := make([]string, 0, len(x))
48			for _, e := range x {
49				b, err := marshalCanonical(e)
50				if err != nil { return nil, err }
51				parts = append(parts, string(b))
52			}
53			return []byte("[" + strings.Join(parts, ",") + "]"), nil
54		case []map[string]any:
55			parts := make([]string, 0, len(x))
56			for _, e := range x {
57				b, err := marshalCanonical(e)
58				if err != nil { return nil, err }
59				parts = append(parts, string(b))
60			}
61			return []byte("[" + strings.Join(parts, ",") + "]"), nil
62		default:
63			return json.Marshal(x)
64		}
65	}
