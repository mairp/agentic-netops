    21	func TestRegisterGuard_PassesForRenderedPaths(t *testing.T) {
    22		// Build a representative spec using current renderers
    23		m := map[string]any{}
    24		for k, v := range render.RenderInterfaces([]model.Interface{{Name: "Ethernet1"}}, []model.Loopback{{Name: "Loopback0"}}) { m[k] = v }
    25		for k, v := range render.RenderBGP(model.BGPGlobal{ASN: 65000, RouterID: "1.1.1.1"}, []model.BGPNeighbor{{NeighborAddress: "10.0.0.2", PeerASN: 65001, EVPN: true}}) { m[k] = v }
    26		for k, v := range render.RenderVXLAN(model.VXLAN{SourceInterface: "Loopback0", UDPPort: 4789}, []model.VLAN{{ID: 10, Name: "blue", L2VNI: 10010}}) { m[k] = v }
    27		for k, v := range render.RenderSRv6(model.SRv6Locator{Name: "default", Prefix: "2001:db8:1::/48"}, []model.MySID{{SID: "2001:db8:1::1", Behavior: "End"}}) { m[k] = v }
    28		if err := sdc.ValidateSpecAgainstRegister(m, nil); err != nil {
    29			if !sdc.IsRegisterError(err) {
    30				t.Fatalf("unexpected error type: %v", err)
    31			}
    32			t.Fatalf("expected register to cover current rendered paths, got: %v", err)
    33		}
    34	}
