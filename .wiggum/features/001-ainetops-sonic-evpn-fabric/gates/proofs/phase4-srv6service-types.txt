    32	// +kubebuilder:validation:XValidation:message="locator length must be /48-/64",rule="int(self.spec.locatorPrefix.split('/')[1]) >= 48 && int(self.spec.locatorPrefix.split('/')[1]) <= 64"
    33	// +kubebuilder:validation:XValidation:message="service prefix must be /64-/96",rule="int(self.spec.servicePrefix.split('/')[1]) >= 64 && int(self.spec.servicePrefix.split('/')[1]) <= 96"
    34	// +kubebuilder:validation:Optional
    35	// +kubebuilder:validation:Type=object

    38	// +kubebuilder:validation:MinProperties=1

    40	type SRv6Service struct {
    41		metav1.TypeMeta   `json:",inline"`
    42		metav1.ObjectMeta `json:"metadata,omitempty"`
    43	
    44		Spec   SRv6ServiceSpec   `json:"spec,omitempty"`
    45		Status SRv6ServiceStatus `json:"status,omitempty"`
    46	}
    47	
    48	// SRv6ServiceSpec defines desired SRv6 service intent.
    49	type SRv6ServiceSpec struct {
    50		// attachments must contain exactly two distinct nodes
    51		Attachments []Attachment `json:"attachments"`
    52		// topologyRef names a Ready topology in Kubenet
    53		TopologyRef string `json:"topologyRef"`
    54		// locatorPrefix is the IPv6 locator pool prefix allocated by KUID
    55		LocatorPrefix string `json:"locatorPrefix"`
    56		// vrf is immutable service VRF name
    57		VRF string `json:"vrf"`
    58		// path selection with primary and alternate waypoint sequence names
    59		Path PathSelection `json:"path"`
    60		// client end of the service
    61		Client Endpoint `json:"client"`
    62		// server end of the service
    63		Server Endpoint `json:"server"`
    64		// servicePrefix identifies the service IPv6 CIDR
    65		ServicePrefix string `json:"servicePrefix"`
    66		// optional list of explicit SONiC transit waypoint device names
    67		TransitWaypoints []string `json:"transitWaypoints,omitempty"`
    68		// underlay must be ipv6 for SRv6
    69		Underlay string `json:"underlay"`
    70	}
    71	
    72	// Attachment pairs a node with an interface.
    73	type Attachment struct {
    74		Node string `json:"node"`
    75		Intf string `json:"interface"`
    76	}
    77	
    78	// PathSelection identifies named waypoint routes.
    79	type PathSelection struct {
    80		Primary string `json:"primary"`
    81		Alternate string `json:"alternate,omitempty"`
    82	}
    83	
    84	// Endpoint identifies an endpoint node and IP/prefix.
    85	type Endpoint struct {
    86		Name string `json:"name"`
    87		IP   string `json:"ip"`
    88	}
    89	
    90	// SRv6ServiceStatus exposes observedGeneration, conditions, allocations, and per-path data.
    91	type SRv6ServiceStatus struct {
    92		ObservedGeneration int64              `json:"observedGeneration,omitempty"`
    93		Conditions         []metav1.Condition `json:"conditions,omitempty"`
    94		// Hashes of generated SDC Configs per target
    95		ConfigHashes map[string]string `json:"configHashes,omitempty"`
    96		// Active path name
    97		ActivePath string `json:"activePath,omitempty"`
    98	}
