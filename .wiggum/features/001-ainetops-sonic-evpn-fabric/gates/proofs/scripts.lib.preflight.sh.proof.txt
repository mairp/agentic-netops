     1	#!/usr/bin/env bash
     2	# Strict shell reusable preflight checks (Phase 1)
     3	set -euo pipefail
     4	
     5	preflight::die() { echo "[preflight] ERROR: $*" >&2; exit 1; }
     6	preflight::warn() { echo "[preflight] WARN: $*" >&2; }
     7	
     8	preflight::require_cmd() {
     9	  local cmd=$1; command -v "$cmd" >/dev/null 2>&1 || preflight::die "missing required command: $cmd"
    10	}
    11	
    12	preflight::check_versions_lock() {
    13	  local root; root=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/../.." && pwd)
    14	  local f="$root/versions.lock.yaml"
    15	  [[ -f "$f" ]] || preflight::die "versions.lock.yaml not found at $f"
    16	  # Reject 'latest' and floating refs quickly; make verify-pins will perform full validation
    17	  if grep -nEi '\blatest\b|\bmain\b|\bmaster\b|\bHEAD\b' "$f" >/dev/null; then
    18	    preflight::die "versions.lock.yaml contains floating refs (latest/main/master/HEAD)."
    19	  fi
    20	}
    21	
    22	preflight::host_resources() {
    23	  # Minimal CPU/RAM/disk checks; allow override via env for CI
    24	  local min_cpu=${AINETOPS_MIN_CPU:-4}
    25	  local min_mem_mb=${AINETOPS_MIN_MEM_MB:-8192}
    26	  local min_disk_mb=${AINETOPS_MIN_DISK_MB:-20480}
    27	  # Best-effort checks; platform-specific commands may vary
    28	  local cores mem_kb avail_kb
    29	  cores=$(getconf _NPROCESSORS_ONLN || echo 1)
    30	  mem_kb=$(awk '/MemTotal/ {print $2}' /proc/meminfo 2>/dev/null || echo 0)
    31	  avail_kb=$(df -Pm . 2>/dev/null | awk 'NR==2{print $4}' || echo 0)
    32	  (( cores >= min_cpu )) || preflight::die "CPU cores $cores < required $min_cpu"
    33	  (( mem_kb/1024 >= min_mem_mb )) || preflight::die "Memory $((${mem_kb}/1024))MB < required ${min_mem_mb}MB"
    34	  (( avail_kb >= min_disk_mb )) || preflight::die "Disk ${avail_kb}MB free < required ${min_disk_mb}MB"
    35	}
    36	
    37	preflight::runtime_privileges() {
    38	  # Require Docker-compatible runtime and privileges for Kind/containerlab
    39	  preflight::require_cmd docker
    40	  if ! docker info >/dev/null 2>&1; then
    41	    preflight::die "docker daemon not reachable"
    42	  fi
    43	}
    44	
    45	# Utility: IPv4 dotted-quad to integer
    46	preflight::ip2int() {
    47	  local IFS=.; read -r a b c d <<<"$1"; echo $(( (a<<24) + (b<<16) + (c<<8) + d ))
    48	}
    49	
    50	# Utility: derive [start,end] integer range from CIDR
    51	preflight::cidr_range() {
    52	  local cidr=$1 ip=${1%/*} mask=${1#*/}
    53	  local ipn; ipn=$(preflight::ip2int "$ip")
    54	  local maskn=$(( 0xFFFFFFFF << (32-mask) & 0xFFFFFFFF ))
    55	  local start=$(( ipn & maskn ))
    56	  local end=$(( start | (~maskn & 0xFFFFFFFF) ))
    57	  echo "$start $end"
    58	}
    59	
    60	# Return 0 (overlap) or 1 (disjoint)
    61	preflight::ranges_overlap() {
    62	  local a1=$1 a2=$2 b1=$3 b2=$4
    63	  if (( a1 <= b2 && b1 <= a2 )); then return 0; else return 1; fi
    64	}
    65	
    66	preflight::address_conflicts() {
    67	  # Ensure management network CIDR does not overlap pod/service CIDRs
    68	  local mgmt_cidr=${AINETOPS_MGMT_CIDR:-172.31.0.0/16}
    69	  local pod_cidr=${AINETOPS_POD_CIDR:-10.244.0.0/16}
    70	  local svc_cidr=${AINETOPS_SERVICE_CIDR:-10.96.0.0/12}
    71	  read -r m1 m2 < <(preflight::cidr_range "$mgmt_cidr")
    72	  read -r p1 p2 < <(preflight::cidr_range "$pod_cidr")
    73	  read -r s1 s2 < <(preflight::cidr_range "$svc_cidr")
    74	  if preflight::ranges_overlap "$m1" "$m2" "$p1" "$p2"; then
    75	    preflight::die "Management CIDR $mgmt_cidr overlaps Pod CIDR $pod_cidr"
    76	  fi
    77	  if preflight::ranges_overlap "$m1" "$m2" "$s1" "$s2"; then
    78	    preflight::die "Management CIDR $mgmt_cidr overlaps Service CIDR $svc_cidr"
    79	  fi
    80	}
    81	
    82	preflight::mtu() {
    83	  # Verify host MTU accommodates VXLAN overhead (> 1500 suggested); warn only in Phase 1
    84	  local maxmtu=0
    85	  while read -r mtu; do (( mtu > maxmtu )) && maxmtu=$mtu; done < <(ip -o link 2>/dev/null | awk -F'mtu ' '{print $2}' | awk '{print $1}')
    86	  if (( maxmtu < 1500 )); then preflight::warn "maximum host MTU $maxmtu < 1500; VXLAN overhead may break traffic"; fi
    87	}
    88	
    89	preflight::kvm_check() {
    90	  local profile=${AINETOPS_PROFILE:-sonic-vs}
    91	  if [[ "$profile" == "sonic-vm" ]]; then
    92	    # Require KVM when sonic-vm profile selected
    93	    [[ -e /dev/kvm ]] || preflight::die "/dev/kvm not present for sonic-vm profile"
    94	  fi
    95	}
    96	
    97	# Extract value from versions.lock.yaml given a top-level section and key
    98	preflight::yaml_value() {
    99	  local section=$1 key=$2 file; file=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/../.." && pwd)/versions.lock.yaml
   100	  awk -v s="^"$section":$" -v k="^\\s*"$key":" 'f; $0~s{f=1} f && /^[^[:space:]]/{if(!p){p=1;print}else exit} f && p{if ($0~k){sub(/.*:[[:space:]]*/, ""); gsub(/\"/, ""); print; exit}}' "$file"
   101	}
   102	
   103	preflight::tool_versions() {
   104	  # Verify host tool versions match pins in versions.lock.yaml
   105	  for c in kind kubectl helm containerlab jq curl; do preflight::require_cmd "$c"; done
   106	
   107	  local kind_pin kubectl_pin helm_pin clab_pin
   108	  kind_pin=$(preflight::yaml_value kind binary)
   109	  # Prefer host_tools.kubectl if present, else fall back to kubernetes.kubernetes
   110	  kubectl_pin=$(preflight::yaml_value host_tools kubectl)
   111	  if [[ -z "${kubectl_pin}" ]]; then kubectl_pin=$(preflight::yaml_value kubernetes kubernetes); fi
   112	  helm_pin=$(preflight::yaml_value host_tools helm)
   113	  clab_pin=$(preflight::yaml_value containerlab version)
   114	
   115	  [[ -n "$kind_pin" ]] || preflight::die "missing kind.binary pin in versions.lock.yaml"
   116	  [[ -n "$kubectl_pin" ]] || preflight::die "missing host_tools.kubectl or kubernetes.kubernetes pin in versions.lock.yaml"
   117	  [[ -n "$helm_pin" ]] || preflight::die "missing host_tools.helm pin in versions.lock.yaml"
   118	  [[ -n "$clab_pin" ]] || preflight::die "missing containerlab.version pin in versions.lock.yaml"
   119	
   120	  # kind
   121	  local kind_ver; kind_ver=$(kind version 2>/dev/null | awk '{print $2}') || true
   122	  [[ "$kind_ver" == "$kind_pin" ]] || preflight::die "kind version $kind_ver != pinned $kind_pin"
   123	
   124	  # kubectl
   125	  local kubectl_ver; kubectl_ver=$(kubectl version --client -o json 2>/dev/null | jq -r '.clientVersion.gitVersion' || true)
   126	  [[ "$kubectl_ver" == "$kubectl_pin" ]] || preflight::die "kubectl version $kubectl_ver != pinned $kubectl_pin"
   127	
   128	  # helm
   129	  local helm_ver; helm_ver=$(helm version --short --client 2>/dev/null | sed 's/+.*//' || true)
   130	  [[ "$helm_ver" == "$helm_pin" ]] || preflight::die "helm version $helm_ver != pinned $helm_pin"
   131	
   132	  # containerlab (parse the 'version:' line)
   133	  local clab_ver; clab_ver=$(containerlab version 2>/dev/null | awk -F': *' '/^version:/ {print $2; exit}')
   134	  [[ "$clab_ver" == "$clab_pin" ]] || preflight::die "containerlab version $clab_ver != pinned $clab_pin"
   135	}
   136	
   137	preflight::run() {
   138	  preflight::check_versions_lock
   139	  preflight::host_resources
   140	  preflight::runtime_privileges
   141	  preflight::address_conflicts
   142	  preflight::mtu
   143	  preflight::kvm_check
   144	  preflight::tool_versions
   145	  echo "[preflight] basic host checks passed"
   146	}
   147	
   148	export -f preflight::run
