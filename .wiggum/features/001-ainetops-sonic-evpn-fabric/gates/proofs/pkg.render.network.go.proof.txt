10	// RenderNetworkInstances renders VRFs and route distinguishers/targets.
11	func RenderNetworkInstances(instances []model.NetworkInstance) map[string]any {
12		res := map[string]any{}
13		base := "/network-instances/network-instance"
14		// stable order
15		names := make([]string, 0, len(instances))
16		byName := map[string]model.NetworkInstance{}
17		for _, ni := range instances { names = append(names, ni.Name); byName[ni.Name] = ni }
18		sort.Strings(names)
19		entries := []map[string]any{}
20		for _, n := range names {
21			ni := byName[n]
22			entry := map[string]any{"name": ni.Name, "type": ni.Type}
23			if ni.RD != "" { entry["rd"] = ni.RD }
24			if len(ni.ImportRT) > 0 { entry["import-rt"] = append([]string{}, ni.ImportRT...) }
25			if len(ni.ExportRT) > 0 { entry["export-rt"] = append([]string{}, ni.ExportRT...) }
26			entries = append(entries, entry)
27		}
28		res[base] = entries
29		return res
30	}

32	// RenderVXLAN renders NVO/VTEP and L2VNI/VLAN bridges placeholders to OpenConfig-preferring paths.
33	func RenderVXLAN(vx model.VXLAN, vlans []model.VLAN) map[string]any {
34		res := map[string]any{}
35		// For MVP scaffold, record VTEP source-interface and UDP port under a canonical key
36		res["/interfaces/interface[vtep]"] = map[string]any{"source-interface": vx.SourceInterface, "udp-port": vx.UDPPort}
37		// Bridges/VLANs simplified
38		ids := make([]int, 0, len(vlans))
39		byID := map[int]model.VLAN{}
40		for _, v := range vlans { ids = append(ids, v.ID); byID[v.ID] = v }
41		sort.Ints(ids)
42		bridges := []map[string]any{}
43		for _, id := range ids {
44			v := byID[id]
45			bd := map[string]any{"vlan": v.ID, "name": v.Name}
46			if v.L2VNI != 0 { bd["l2vni"] = v.L2VNI }
47			bridges = append(bridges, bd)
48		}
49		res["/network-instances/network-instance/bridges"] = bridges
50		return res
51	}
