    10	func TestReject_MissingEndpoints_VPLS(t *testing.T) {
    11		in := migration.ServiceInput{
    12			ServiceID: "noep-vpls",
    13			Type:      migration.ServiceVPLS,
    14			Tenant:    "T",
    15			RDRT:      &migration.RdRt{RD: "65000:101", ImportRT: []string{"65000:101"}, ExportRT: []string{"65000:101"}},
    16			L2VNI:     10101,
    17			Endpoints: []migration.Endpoint{},
    18		}
    19		outs, err := migration.RenderBatch([]migration.ServiceInput{in})
    20		if err == nil {
    21			t.Fatalf("expected error for missing endpoints (VPLS)")
    22		}
    23		if len(outs) != 0 {
    24			t.Fatalf("expected no outputs on failure")
    25		}
    26		msg := migration.MarshalError(err)
    27		if !strings.Contains(msg, "endpoints:") {
    28			t.Fatalf("expected endpoints cause, got: %s", msg)
    29		}
    30	}
