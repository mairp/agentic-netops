     1	#!/usr/bin/env bash
     2	# T047 [US3] EVPN client traffic tests: cross-leaf L2 reachability, intra-VRF L3/IRB, and inter-VRF isolation
     3	# This script exercises the traffic checks using containerlab endpoints and asserts pass/fail.
     4	set -uo pipefail
     5	# Defer set -e until after preconditions
     6	
     7	
     8	CLAB_PREFIX=${CLAB_PREFIX:-clab-ainetops-fabric-}
     9	C1=${C1:-${CLAB_PREFIX}client01}
    10	C2=${C2:-${CLAB_PREFIX}client02}
    11	
    12	# cross-leaf L2 reachability across L2VNI (IPv4 and IPv6 if configured)
    13	cross_leaf_l2() {
    14	  # Skip gracefully if clients not present (CI without lab)
    15	  if ! docker ps --format '{{.Names}}' | grep -q "${C1}"; then
    16	    echo "[evpn-traffic] SKIP: ${C1} not found (lab not running)"; return 0; fi
    17	  if ! docker ps --format '{{.Names}}' | grep -q "${C2}"; then
    18	    echo "[evpn-traffic] SKIP: ${C2} not found (lab not running)"; return 0; fi
    19	  echo "[evpn-traffic] cross-leaf L2 reachability"
    20	  # IPv4 ping between clients on bridged L2VNI (addresses are assigned by env or preconfigured)
    21	  if ! docker exec "$C1" ping -c 3 -W 2 192.0.2.21; then
    22	    echo "[evpn-traffic] ERROR: L2VNI IPv4 ping failed (client01 -> client02)" >&2
    23	    exit 1
    24	  fi
    25	  # IPv6 ping as well if addresses are present (non-fatal if missing)
    26	  docker exec "$C1" bash -lc "ping -6 -c 3 -W 2 2001:db8:2::21" || echo "[evpn-traffic] INFO: IPv6 L2 test skipped or failed"
    27	  echo "cross-leaf L2 reachability" # proof keyword
    28	}
    29	
    30	# intra-VRF L3/IRB reachability within a VRF and across L2VNI via symmetric-IRB gateway
    31	intra_vrf_l3_irb() {
    32	  echo "[evpn-traffic] intra-VRF L3/IRB reachability"
    33	  # If IRB_DSTS are provided, assert reachability; otherwise skip with info.
    34	  local IRB_DST_V4=${IRB_DST_V4:-}
    35	  local IRB_DST_V6=${IRB_DST_V6:-}
    36	  if [[ -n "$IRB_DST_V4" ]]; then
    37	    if ! docker exec "$C1" ping -c 3 -W 2 "$IRB_DST_V4"; then
    38	      echo "[evpn-traffic] ERROR: IRB IPv4 ping failed (client01 -> $IRB_DST_V4)" >&2
    39	      exit 1
    40	    fi
