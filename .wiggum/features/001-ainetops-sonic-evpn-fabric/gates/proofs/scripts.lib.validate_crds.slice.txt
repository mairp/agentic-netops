     1	#!/usr/bin/env bash
     2	# Validate Kubenet/KUID and SDC CRDs/examples with Kubernetes server-side dry-run
     3	# Uses pinned commits/releases recorded in versions.lock.yaml
     4	set -euo pipefail
     5	
     6	ROOT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/../.." && pwd)
     7	LOCK_FILE="${ROOT_DIR}/versions.lock.yaml"
     8	SOFT_MODE=${AINETOPS_SOFT_TOOLCHECK:-false}
     9	
    10	require() { command -v "$1" >/dev/null 2>&1 || { echo "missing command: $1" >&2; return 1; }; }
    11	soft_warn() { echo "[validate-crds] WARN: $*" >&2; }
    12	
    13	# Check tool availability
    14	if ! require kubectl; then
    15	  if [[ "$SOFT_MODE" == "true" ]]; then
    16	    soft_warn "kubectl not available; skipping server-side CRD validation in soft mode"
    17	    exit 0
    18	  else
    19	    exit 1
    20	  fi
    21	fi
    22	require awk || { soft_warn "awk not available"; [[ "$SOFT_MODE" == "true" ]] || exit 1; }
    23	
    24	# Helper to extract a key's value block between a section and next top-level key
    25	section() { awk -v s="^$1:$" 'f; $0~s{f=1} f && /^[^[:space:]]/{if(!p){p=1;print}else exit} f && p{print}' "$LOCK_FILE"; }
    26	
    27	kubenet_commit=$(section kubenet | awk '/commit:/ {print $2; exit}')
    28	kubenet_api_shape=$(section kubenet | awk '/api_shape:/ {print $2; exit}')
    29	kuid_commit=$(section kuid | awk '/commit:/ {print $2; exit}')
    30	# Extract the sdc core (sdcio/sdc) release explicitly, not the nested config-server/schema-server
    31	sdc_core_release=$(awk '
    32	  $1=="sdc:" {sec=1; next}
    33	  sec && /^[^[:space:]]/ {exit}
    34	  sec && $1=="core:" {in_core=1; next}
    35	  in_core && /^[^[:space:]]/ {exit}
    36	  in_core && $1=="release:" {print $2; exit}
    37	' "$LOCK_FILE")
    38	
    39	[[ $kubenet_commit =~ ^[0-9a-f]{40}$ ]] || { echo "invalid kubenet commit in versions.lock.yaml" >&2; [[ "$SOFT_MODE" == "true" ]] || exit 1; }
    40	[[ $kuid_commit =~ ^[0-9a-f]{40}$ ]] || { echo "invalid kuid commit in versions.lock.yaml" >&2; [[ "$SOFT_MODE" == "true" ]] || exit 1; }
    41	[[ $sdc_core_release =~ ^v[0-9]+\.[0-9]+\.[0-9]+$ ]] || { echo "invalid sdc core release in versions.lock.yaml" >&2; [[ "$SOFT_MODE" == "true" ]] || exit 1; }
    42	
    43	# Validate against pinned local copies committed in deploy/*, derived from the pinned upstream commits/releases.
    44	# This avoids network flakiness while preserving immutability and API shape.
    45	KUBENET_CRDS=()
    46	case "$kubenet_api_shape" in
    47	  NetworkConfig)
    48	    KUBENET_CRDS=(
    49	      "${ROOT_DIR}/deploy/kubenet/crds/kubenet-crds.yaml"
    50	    )
    51	    ;;
    52	  NetworkDesign)
    53	    KUBENET_CRDS=(
    54	      "${ROOT_DIR}/deploy/kubenet/crds/kubenet-crds.yaml"
    55	    )
    56	    ;;
    57	  *)
    58	    KUBENET_CRDS=(
    59	      "${ROOT_DIR}/deploy/kubenet/crds/kubenet-crds.yaml"
    60	    )
    61	    ;;
    62	 esac
    63	KUBENET_EXAMPLES=(
    64	  "${ROOT_DIR}/deploy/kubenet/topology.yaml"
    65	)
    66	KUID_CRDS=(
    67	  "${ROOT_DIR}/deploy/kuid/crds/kuid-crds.yaml"
    68	)
    69	SDC_CRDS=(
    70	  "${ROOT_DIR}/deploy/sdc/crds/sdc-crds.yaml"
    71	)
    72	
    73	run_dry_run() {
    74	  local what=$1; shift
    75	  local -a files=("$@")
    76	  echo "[validate-crds] server-side dry-run: ${what} (${#files[@]} files)" >&2
    77	  # Build multiple -f flags so kubectl handles each manifest separately
    78	  local -a args=(apply --dry-run=server)
    79	  for f in "${files[@]}"; do args+=( -f "$f" ); done
    80	  set +e
    81	  kubectl "${args[@]}"
    82	  rc=$?
    83	  set -e
    84	  if [[ $rc -ne 0 ]]; then
    85	    if [[ "$SOFT_MODE" == "true" ]]; then
    86	      soft_warn "server-side dry-run failed for ${what}; continuing in soft mode"
    87	      return 0
    88	    else
    89	      return $rc
    90	    fi
    91	  fi
    92	}
    93	
    94	# Note: The actual cluster must be reachable for server-side validation.
    95	# This script is idempotent and performs validation only.
    96	
    97	# Validate CRDs (best-effort in soft mode)
    98	run_dry_run "Kubenet CRDs" "${KUBENET_CRDS[@]}"
    99	run_dry_run "KUID CRDs" "${KUID_CRDS[@]}"
   100	run_dry_run "SDC CRDs" "${SDC_CRDS[@]}"
   101	
   102	# Validate example manifests against CRDs
   103	run_dry_run "Kubenet examples" "${KUBENET_EXAMPLES[@]}"
   104	
   105	echo "[validate-crds] OK"
