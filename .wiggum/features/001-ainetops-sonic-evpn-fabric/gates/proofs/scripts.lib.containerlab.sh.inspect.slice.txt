    23	clab::inspect() {
    24	  clab::require
    25	  # containerlab >=0.7x uses -f/--format (not -o) for output format selection
    26	  containerlab inspect -t "${TOPO_FILE}" --format json
    27	}
    28	
    29	clab::destroy() {
    30	  clab::require
    31	  echo "[clab] destroying ${TOPO_FILE}"
    32	  containerlab destroy -t "${TOPO_FILE}" --cleanup || true
    33	  # Remove the per-node named volumes that persist SONiC /etc/sonic. Containerlab
    34	  # does not always reclaim topology-declared named volumes, so we drop them
    35	  # explicitly by their deterministic names (only the ones this lab owns).
    36	  local v
    37	  for v in ainetops-spine01-etc-sonic ainetops-spine02-etc-sonic ainetops-leaf01-etc-sonic ainetops-leaf02-etc-sonic; do
    38	    docker volume rm "$v" >/dev/null 2>&1 || true
    39	  done
    40	  # Verify teardown leaves no owned lab containers or volumes
    41	  local leftovers
    42	  leftovers=$(docker ps -a --format '{{.Names}} {{.Labels}}' | awk '/ainetops.owner=ainetops/ {print $1}') || true
    43	  if [[ -n "$leftovers" ]]; then
    44	    echo "[clab] WARN: leftover AINETOPS containers not removed:\n$leftovers" >&2
    45	    exit 1
    46	  fi
    47	  # Check for owned volumes (persistent /etc/sonic)
    48	  local vol_left
    49	  vol_left=$(docker volume ls -q | grep -E '^ainetops-.*-etc-sonic$' || true)
    50	  if [[ -n "$vol_left" ]]; then
    51	    echo "[clab] WARN: leftover AINETOPS volumes not removed:\n$vol_left" >&2
    52	    exit 1
    53	  fi
    54	  # Check for generated lab credentials under repo secrets/
    55	  if [[ -e "${ROOT_DIR}/secrets/gnmi.key" || -e "${ROOT_DIR}/secrets/gnmi.crt" || -e "${ROOT_DIR}/secrets/ca.crt" ]]; then
    56	    echo "[clab] WARN: leftover lab-generated gNMI credentials under ${ROOT_DIR}/secrets" >&2
    57	    exit 1
    58	  fi
    59	  echo "[clab] destroy complete"
    60	}
    61	
    62	case "${1:-}" in
    63	  deploy) shift; clab::deploy "$@" ;;
    64	  inspect) shift; clab::inspect "$@" ;;
    65	  destroy) shift; clab::destroy "$@" ;;
    66	  *) echo "usage: $0 {deploy|inspect|destroy}" >&2; exit 2 ;;
