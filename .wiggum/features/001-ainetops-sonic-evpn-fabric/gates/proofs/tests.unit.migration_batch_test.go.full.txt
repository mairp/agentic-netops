     9	func goodVPLS() migration.ServiceInput {
    10		return migration.ServiceInput{
    11			ServiceID: "svcVPLS",
    12			Type:      migration.ServiceVPLS,
    13			Tenant:    "T",
    14			RDRT:      &migration.RdRt{RD: "65000:1", ImportRT: []string{"65000:1"}, ExportRT: []string{"65000:1"}},
    15			L2VNI:     10001,
    16			Endpoints: []migration.Endpoint{{Node: "leaf01", Attachment: "c1", VLAN: 10}, {Node: "leaf02", Attachment: "c2", VLAN: 10}},
    17		}
    18	}
    19	
    20	func TestBatchAllOrNothing(t *testing.T) {
    21		ok := goodVPLS()
    22		bad := ok
    23		bad.ServiceID = "dup"
    24		ok.ServiceID = "dup"
    25		outs, err := migration.RenderBatch([]migration.ServiceInput{ok, bad})
    26		if err == nil { t.Fatalf("expected error for duplicate serviceId") }
    27		if len(outs) != 0 { t.Fatalf("expected no outputs on failure") }
    28		msg := migration.MarshalError(err)
    29		if !strings.Contains(msg, "duplicate serviceId") {
    30			t.Fatalf("expected duplicate cause, got: %s", msg)
    31		}
    32	}
