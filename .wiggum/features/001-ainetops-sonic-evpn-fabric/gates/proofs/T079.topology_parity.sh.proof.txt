     1	#!/usr/bin/env bash
     2	# T079: Topology parity test — verify that the generated/pinned topology ConfigMap
     3	# (deploy/observability/topology-configmap.yaml) matches the containerlab topology
     4	# (lab/topology.clab.yml) by node ids and undirected link endpoints.
     5	#
     6	# The test prefers live containerlab inspect JSON on stdin or via --inspect <path.json>.
     7	# When inspect JSON is not available, it derives expected nodes/links from the
     8	# containerlab topology YAML directly (best-effort static check suitable for CI).
     9	#
    10	# Outputs a human-readable report to stdout and exits non-zero on mismatch.
    11	# Emits the symbol TOPOLOGY_PARITY_OK on success for gate proof grepping.
    12	set -euo pipefail
    13	
    14	ROOT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/../.." && pwd)
    15	TOPO_YAML=${TOPO_YAML:-"${ROOT_DIR}/lab/topology.clab.yml"}
    16	CONFIGMAP_YAML=${CONFIGMAP_YAML:-"${ROOT_DIR}/deploy/observability/topology-configmap.yaml"}
    17	INSPECT_JSON=${INSPECT_JSON:-}
    18	
    19	usage(){
    20	  cat <<EOF
    21	Usage: $0 [--inspect path.json]
    22	  --inspect path.json  Optional containerlab inspect JSON file to prefer
    23	Environment overrides:
    24	  TOPO_YAML, CONFIGMAP_YAML, INSPECT_JSON
    25	EOF
    26	}
    27	
    28	while [[ $# -gt 0 ]]; do
    29	  case "$1" in
    30	    --inspect) shift; INSPECT_JSON=${1:-$INSPECT_JSON} ;;
    31	    -h|--help) usage; exit 0 ;;
    32	    *) echo "[topology-parity] unknown arg: $1" >&2; usage; exit 2 ;;
    33	  esac
    34	  shift || true
    35	done
    36	
    37	# read_nodes_links_from_topology: outputs two lines: "NODES: a b c" and "LINKS: a-b c-d ..."
    38	read_nodes_links_from_topology(){
    39	  local yaml=$1
    40	  # Nodes: capture lines inside "nodes:" mapping that look like "  name:" (spine01:, leaf01:, etc.)
    41	  # Accept alnum/underscore/hyphen. Trim trailing colon.
    42	  local nodes
    43	  nodes=$(awk '
    44	    $0 ~ /^\s*nodes:\s*$/ { in_nodes=1; next }
    45	    in_nodes && $0 ~ /^\s*links:\s*$/ { in_nodes=0 }
    46	    in_nodes && $0 ~ /^\s*[A-Za-z0-9_-]+:\s*$/ { gsub(":",""); gsub(/^\s+/,""); print $0 }
    47	  ' "$yaml" | tr '\n' ' ')
    48	  # Links: find "- endpoints: [a:x, b:y]" -> produce sorted endpoint pair "a-b"
    49	  local links
    50	  links=$(grep -E "^-\s*endpoints:\s*\[[^]]+\]" -n "$yaml" 2>/dev/null \
    51	    | sed -E 's/.*\[([^]]+)\].*/\1/' \
    52	    | awk -F, '{
    53	        gsub(/^[ ]+|[ ]+$/, "", $1); gsub(/^[ ]+|[ ]+$/, "", $2);
    54	        split($1, a, ":"); split($2, b, ":");
    55	        n1=a[1]; n2=b[1];
    56	        # Only keep fabric links between spine/leaf nodes
    57	        if (n1 ~ /^(spine|leaf)[0-9]+$/ && n2 ~ /^(spine|leaf)[0-9]+$/) {
    58	          if (n1 < n2) { printf "%s-%s\n", n1, n2 } else { printf "%s-%s\n", n2, n1 }
    59	        }
    60	      }' \
    61	    | sort -u | tr '\n' ' ')
    62	  # For nodes, restrict to those participating in fabric links (spine/leaf)
    63	  local nodes_from_links
    64	  nodes_from_links=$(tr ' ' '\n' <<<"$links" | awk -F- '{print $1"\n"$2}' | sort -u | tr '\n' ' ')
    65	  echo "NODES: $nodes_from_links"
    66	  echo "LINKS: $links"
    67	}
    68	
    69	# read_nodes_links_from_configmap: parse topology.json from the pinned ConfigMap YAML
    70	read_nodes_links_from_configmap(){
    71	  local cm=$1
    72	  # Extract the embedded JSON block and parse with jq if available; else use grep/sed.
    73	  local json
    74	  json=$(awk '/^\s*topology.json: \|/{flag=1; next} /^\S/{flag=0} flag{print substr($0,5)}' "$cm" | tr -d '\r')
    75	  if command -v jq >/dev/null 2>&1; then
    76	    local nodes links
    77	    nodes=$(jq -r '.nodes[].id' <<<"$json" | sort -u | tr '\n' ' ')
    78	    links=$(jq -r '.links[] | [.source, .target] | sort | join("-")' <<<"$json" | sort -u | tr '\n' ' ')
    79	    echo "NODES: $nodes"
    80	    echo "LINKS: $links"
    81	  else
    82	    # Fallback regex parsing for ids and links
    83	    local nodes links
    84	    nodes=$(grep -o '"id"\s*:\s*"[^"]\+"' <<<"$json" | sed -E 's/.*:\s*"([^"]+)"/\1/' | sort -u | tr '\n' ' ')
    85	    links=$(grep -E '"source"|"target"' <<<"$json" | sed -E 's/.*:\s*"([^"]+)".*/\1/' | paste - - \
    86	      | awk '{ a=$1; b=$2; if (a<b) {print a"-"b} else {print b"-"a} }' | sort -u | tr '\n' ' ')
    87	    echo "NODES: $nodes"
    88	    echo "LINKS: $links"
    89	  fi
    90	}
    91	
    92	# If inspect JSON provided, prefer it to derive expected nodes/links
    93	if [[ -n "${INSPECT_JSON:-}" && -f "$INSPECT_JSON" ]]; then
    94	  echo "[topology-parity] INFO: using inspect JSON: $INSPECT_JSON" >&2
    95	  # Expect JSON structure: .topology.nodes[].name and .topology.links[].a.node/.b.node
    96	  if command -v jq >/dev/null 2>&1; then
    97	    exp_nodes=$(jq -r '.topology.nodes[].name' "$INSPECT_JSON" | sort -u | tr '\n' ' ')
    98	    exp_links=$(jq -r '.topology.links[] | [.a.node, .b.node] | sort | join("-")' "$INSPECT_JSON" | sort -u | tr '\n' ' ')
    99	    echo "EXPECTED NODES: $exp_nodes"
   100	    echo "EXPECTED LINKS: $exp_links"
   101	  else
   102	    echo "[topology-parity] WARN: jq not available; falling back to YAML parsing" >&2
   103	    readarray -t arr < <(read_nodes_links_from_topology "$TOPO_YAML")
   104	    exp_nodes=${arr[0]#NODES: } ; exp_links=${arr[1]#LINKS: }
   105	  fi
   106	else
   107	  readarray -t arr < <(read_nodes_links_from_topology "$TOPO_YAML")
   108	  exp_nodes=${arr[0]#NODES: } ; exp_links=${arr[1]#LINKS: }
   109	  echo "EXPECTED NODES: $exp_nodes"
   110	  echo "EXPECTED LINKS: $exp_links"
   111	fi
   112	
   113	readarray -t cmarr < <(read_nodes_links_from_configmap "$CONFIGMAP_YAML")
   114	cm_nodes=${cmarr[0]#NODES: } ; cm_links=${cmarr[1]#LINKS: }
   115	
   116	declare -A expn cmn
   117	mismatch=0
   118	for n in $exp_nodes; do expn[$n]=1; done
   119	for n in $cm_nodes; do cmn[$n]=1; done
   120	
   121	# Compare node sets
   122	for n in "${!expn[@]}"; do if [[ -z "${cmn[$n]:-}" ]]; then echo "MISSING NODE in ConfigMap: $n"; mismatch=1; fi; done
   123	for n in "${!cmn[@]}"; do if [[ -z "${expn[$n]:-}" ]]; then echo "EXTRA NODE in ConfigMap: $n"; mismatch=1; fi; done
   124	
   125	# Compare link sets
   126	declare -A expl cml
   127	for e in $exp_links; do expl[$e]=1; done
   128	for e in $cm_links; do cml[$e]=1; done
   129	for e in "${!expl[@]}"; do if [[ -z "${cml[$e]:-}" ]]; then echo "MISSING LINK in ConfigMap: $e"; mismatch=1; fi; done
   130	for e in "${!cml[@]}"; do if [[ -z "${expl[$e]:-}" ]]; then echo "EXTRA LINK in ConfigMap: $e"; mismatch=1; fi; done
   131	
   132	exp_node_count=$(wc -w <<<"$exp_nodes" | tr -d ' ')
   133	cm_node_count=$(wc -w <<<"$cm_nodes" | tr -d ' ')
   134	exp_link_count=$(wc -w <<<"$exp_links" | tr -d ' ')
   135	cm_link_count=$(wc -w <<<"$cm_links" | tr -d ' ')
   136	
   137	echo "COUNTS: expected nodes=$exp_node_count, configmap nodes=$cm_node_count; expected links=$exp_link_count, configmap links=$cm_link_count"
   138	
   139	if [[ $mismatch -eq 0 ]]; then
   140	  echo "TOPOLOGY_PARITY_OK: nodes and links match between lab/topology.clab.yml and deploy/observability/topology-configmap.yaml"
   141	else
   142	  echo "TOPOLOGY_PARITY_FAIL: mismatch detected" >&2
   143	  exit 1
   144	fi
