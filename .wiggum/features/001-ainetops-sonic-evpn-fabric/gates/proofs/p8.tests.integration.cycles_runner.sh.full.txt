     1	#!/usr/bin/env bash
     2	# T080: Run three clean provision/test/off cycles, plus a second-provision
     3	# idempotence check, an off-from-partial-state test, and a conformance-profile
     4	# cycle. Stores logs under gates/proofs/cycles/.
     5	#
     6	# A "clean" cycle starts from Absent (off.sh --delete-kind true removes the
     7	# Kind cluster, lab, and owned network at cycle end), runs the ordered
     8	# provision workflow, the test phase against the live lab, and a full teardown.
     9	#
    10	# The capability gate (scripts/lib/qualify.sh) is executed by provision.sh and
    11	# is the single source of truth for SRv6 qualification (FR-022). When the
    12	# selected profile is not SRv6-qualified, provision exits non-zero with the
    13	# documented message; the cycle runner records that designed outcome and
    14	# continues with teardown so each cycle ends in a verified clean state.
    15	set -u
    16	
    17	ROOT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/../.." && pwd)
    18	PROOFS="$ROOT_DIR/.wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs/cycles"
    19	mkdir -p "$PROOFS"
    20	
    21	# --- Skip-if-fresh: reuse a just-completed, still-valid prior pass's evidence
    22	# wiggum's ensure_long_job() launches this command fresh on EVERY orchestrator
    23	# restart — it scopes done-markers by run-id with no concept of "recent and
    24	# still valid" (see INCIDENT-2026-08-31-ainetops-phase8.md, issue #9). Without
    25	# this check, a restart mid-phase-8 unconditionally overwrites a fully
    26	# verified ~90-minute pass sitting untouched on disk, at the very next write
    27	# below (the `tee` that truncates cycles.run.log). This check must run before
    28	# that line. It only affects future invocations of this script — it cannot
    29	# retroactively help a copy that is already mid-run.
    30	CYCLES_FRESH_MAX_AGE_SEC="${CYCLES_FRESH_MAX_AGE_SEC:-21600}"  # 6h default
    31	if [[ -z "${CYCLES_FORCE_RERUN:-}" && -f "$PROOFS/cycles.run.log" ]]; then
    32	  # The script's actual last stdout line, "CYCLES_DONE", is a bare echo (not
    33	  # tee) so it never lands in cycles.run.log itself — only "[cycles] end ..."
    34	  # (the line right before it) is tee'd into the file. That line only prints
    35	  # if the script ran to its final statement uninterrupted (no `set -e`, but
    36	  # a kill/crash mid-run stops output here too), so it's the correct on-disk
    37	  # completion marker to check. Confirmed live 2026-08-31: a bare CYCLES_DONE
    38	  # check here never matches, so the very first attempted skip silently fails
    39	  # to skip.
    40	  if [[ "$(tail -n1 "$PROOFS/cycles.run.log" 2>/dev/null)" == "[cycles] end "* ]]; then
    41	    age=$(( $(date +%s) - $(stat -c %Y "$PROOFS/cycles.run.log") ))
    42	    if (( age <= CYCLES_FRESH_MAX_AGE_SEC )); then
    43	      echo "[cycles] skip-if-fresh: cycles.run.log already complete, ${age}s old (<= ${CYCLES_FRESH_MAX_AGE_SEC}s) — reusing existing evidence, not rerunning" | tee -a "$PROOFS/cycles.run.log"
    44	      echo "CYCLES_DONE"
    45	      exit 0
    46	    fi
    47	    echo "[cycles] skip-if-fresh: prior completed run is ${age}s old (> ${CYCLES_FRESH_MAX_AGE_SEC}s) — treating as stale, rerunning" >&2
    48	  fi
    49	fi
    50	
    51	TS=$(date -u +%Y-%m-%dT%H:%M:%SZ)
    52	echo "[cycles] start $TS host=$(hostname) cpu=$(nproc) mem=$(awk '/MemTotal/{print int($2/1024)"MB"}' /proc/meminfo)" | tee "$PROOFS/cycles.run.log"
    53	
    54	run_provision() {
    55	  # $1=log-file  $2=profile
    56	  local log=$1 profile=$2
    57	  echo "[cycles] provision profile=$profile log=$log" >>"$PROOFS/cycles.run.log"
    58	  ( cd "$ROOT_DIR" && ./scripts/provision.sh --profile "$profile" --cluster-name ainetops --timeout 120s ) >"$log" 2>&1
    59	  local rc=$?
    60	  echo "[cycles] provision profile=$profile exit=$rc" | tee -a "$PROOFS/cycles.run.log"
    61	  return 0
    62	}
    63	
    64	run_off() {
    65	  local log=$1
    66	  echo "[cycles] off log=$log" >>"$PROOFS/cycles.run.log"
    67	  ( cd "$ROOT_DIR" && ./scripts/off.sh --cluster-name ainetops --delete-kind true --capture-evidence true ) >"$log" 2>&1
    68	  local rc=$?
    69	  echo "[cycles] off exit=$rc" | tee -a "$PROOFS/cycles.run.log"
    70	  return 0
    71	}
    72	
    73	run_off_noop() {
    74	  local log=$1
    75	  ( cd "$ROOT_DIR" && ./scripts/off.sh --cluster-name ainetops --delete-kind true ) >"$log" 2>&1
    76	  echo "[cycles] off-noop log=$(basename "$log") exit=$?" | tee -a "$PROOFS/cycles.run.log"
    77	  return 0
    78	}
    79	
    80	run_tests() {
    81	  local idx=$1
    82	  ( cd "$ROOT_DIR" && AINETOPS_CLUSTER_NAME=ainetops ./tests/integration/fabric_verify.sh run ) >"$PROOFS/test-fabric-$idx.log" 2>&1
    83	  echo "[cycles] test-fabric-$idx exit=$?" | tee -a "$PROOFS/cycles.run.log"
    84	  ( cd "$ROOT_DIR" && ./tests/integration/topology_parity.sh ) >"$PROOFS/test-parity-$idx.log" 2>&1
    85	  echo "[cycles] test-parity-$idx exit=$?" | tee -a "$PROOFS/cycles.run.log"
    86	  ( cd "$ROOT_DIR" && AINETOPS_CLUSTER_NAME=ainetops ./tests/integration/observability_suite.sh ) >"$PROOFS/test-observability-$idx.log" 2>&1
    87	  echo "[cycles] test-observability-$idx exit=$?" | tee -a "$PROOFS/cycles.run.log"
    88	}
    89	
    90	runtime_inventory() {
    91	  local idx=$1
    92	  if command -v kubectl >/dev/null 2>&1 && kubectl --context kind-ainetops cluster-info >/dev/null 2>&1; then
    93	    kubectl --context kind-ainetops get pods -A -o wide 2>/dev/null | tee "$PROOFS/runtime-inventory-kubectl-$idx.log" || true
    94	  else
    95	    echo "no live kind-ainetops context at inventory time" >"$PROOFS/runtime-inventory-kubectl-$idx.log"
    96	  fi
    97	  helm list -A 2>/dev/null | tee "$PROOFS/runtime-inventory-helm-$idx.log" || true
    98	  if command -v docker >/dev/null 2>&1; then
    99	    docker ps -a --format "{{.ID}} {{.Image}} {{.Names}} {{.Labels}}" 2>/dev/null | tee "$PROOFS/runtime-inventory-docker-$idx.log" || true
   100	  fi
   101	  ( cd "$ROOT_DIR" && ./scripts/ci/denylist_runtime_scan.sh ) >"$PROOFS/runtime-scan-runtime-$idx.log" 2>&1
   102	  echo "[cycles] runtime-scan-$idx exit=$?" | tee -a "$PROOFS/cycles.run.log"
   103	}
   104	
   105	# --- Three clean provision/test/off cycles (sonic-vs fast profile) -----------
   106	for idx in 1 2 3; do
   107	  echo "[cycles] ===== clean cycle $idx =====" | tee -a "$PROOFS/cycles.run.log"
   108	  run_provision "$PROOFS/provision-$idx.log" sonic-vs
   109	  run_tests "$idx"
   110	  runtime_inventory "$idx"
   111	  run_off "$PROOFS/off-$idx.log"
   112	  # Repeatable no-op: off again from the just-cleaned state must succeed
   113	  run_off_noop "$PROOFS/off-$idx-noop.log"
   114	done
   115	
   116	# --- Second-provision idempotence (provision twice without teardown) ---------
   117	echo "[cycles] ===== second-provision idempotence =====" | tee -a "$PROOFS/cycles.run.log"
   118	run_provision "$PROOFS/idempotence-provision-1.log" sonic-vs
   119	run_provision "$PROOFS/idempotence-provision-2.log" sonic-vs
   120	( cd "$ROOT_DIR" && ./scripts/off.sh --cluster-name ainetops --delete-kind true ) >"$PROOFS/idempotence-off.log" 2>&1
   121	echo "[cycles] idempotence-off exit=$?" | tee -a "$PROOFS/cycles.run.log"
   122	
   123	# --- Off from partial state (provision aborted at the capability gate) -------
   124	echo "[cycles] ===== off-from-partial =====" | tee -a "$PROOFS/cycles.run.log"
   125	run_provision "$PROOFS/partial-provision.log" sonic-vs
   126	run_off "$PROOFS/off-from-partial.log"
   127	run_off_noop "$PROOFS/off-from-partial-noop.log"
   128	
   129	# --- Conformance-profile cycle (sonic-vm, where applicable) ------------------
   130	echo "[cycles] ===== conformance profile (sonic-vm) =====" | tee -a "$PROOFS/cycles.run.log"
   131	run_provision "$PROOFS/provision-conformance.log" sonic-vm
   132	run_off "$PROOFS/off-conformance.log"
   133	
   134	# --- Final runtime scan -------------------------------------------------------
   135	( cd "$ROOT_DIR" && ./scripts/ci/denylist_runtime_scan.sh ) >"$PROOFS/runtime-scan-runtime.log" 2>&1
   136	echo "[cycles] final runtime-scan exit=$?" | tee -a "$PROOFS/cycles.run.log"
   137	echo "[cycles] end $(date -u +%Y-%m-%dT%H:%M:%SZ)" | tee -a "$PROOFS/cycles.run.log"
   138	echo "CYCLES_DONE"
