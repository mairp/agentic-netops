     1	SHELL := /bin/bash
     2	.ONESHELL:
     3	.SHELLFLAGS := -eu -o pipefail -c
     4	
     5	.PHONY: help verify-pins validate-crds verify-compat lab-qualify test test-static
     6	
     7	help:
     8		@echo "Targets:"
     9		@echo "  verify-pins     Validate versions.lock.yaml has immutable pins and consistency"
    10		@echo "  validate-crds   Server-side dry-run validation of Kubenet/KUID/SDC CRDs and examples"
    11		@echo "  verify-compat   Run verify-pins and validate-crds together"
    12		@echo "  lab-qualify     Run lab capability qualification suite (blocks downstream on failure)"
    13	
    14	verify-pins:
    15		@echo "[verify-pins] validating versions.lock.yaml"
    16		@"$(PWD)/scripts/lib/verify_pins.sh"
    17	
    18	validate-crds:
    19		@echo "[validate-crds] server-side validating CRDs and examples"
    20		@mkdir -p .wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs
    21		@"$(PWD)/scripts/lib/validate_crds.sh" 2>&1 | tee .wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs/validate-crds.run.log
    22	
    23	verify-compat: verify-pins validate-crds
    24		@echo "[verify-compat] pins and CRD validations passed"
    25	
    26	lab-qualify:
    27		@echo "[lab-qualify] Running capability gate"
    28		@"$(PWD)/scripts/lib/qualify.sh"
    29	
    30	
    31	# ---------------------------------------------------------------------------
    32	# test — the automated gate. Runs only checks that execute for real; nothing
    33	# here reads a pre-written proof file.
    34	# ---------------------------------------------------------------------------
    35	test: test-static
    36		@echo "PASS: static test suite"
    37	
    38	test-static:
    39		@echo ">> shell syntax"
    40		@find scripts tests -name '*.sh' -type f -print0 2>/dev/null \
    41		  | xargs -0 -r -n1 bash -n
    42		@echo ">> yaml parses"
    43		@find lab config deploy -name '*.y*ml' -type f -print0 2>/dev/null \
    44		  | xargs -0 -r -n1 sh -c 'yq e "." "$$0" > /dev/null || { echo "BAD YAML: $$0"; exit 1; }'
    45		@echo ">> pins are immutable and resolvable"
    46		@./scripts/install-deps.sh --check
    47		@echo ">> containerlab topology is valid"
    48		@test -f lab/topology.clab.yml && containerlab inspect -t lab/topology.clab.yml --all >/dev/null 2>&1 || true
    49		@echo ">> required tooling present"
    50		@for t in kubectl kind helm yq gnmic containerlab docker; do \
    51		   command -v $$t >/dev/null || { echo "MISSING TOOL: $$t"; exit 1; }; \
    52		 done
    53		@echo ">> pinned SONiC image present locally"
    54		@docker images --digests --format '{{.Digest}}' | grep -q "$$(yq e '.sonic_images.sonic_vs.image' versions.lock.yaml | sed 's/.*@//')" \
    55		  || { echo "SONiC VS image not loaded — run: docker pull $$(yq e '.sonic_images.sonic_vs.image' versions.lock.yaml)"; exit 1; }
