1: #!/usr/bin/env bash
2: # Persistence qualification helper: restart SONiC containers and verify gNMI-set value persists (T014)
3: set -euo pipefail
4: 
5: ROOT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/../.." && pwd)
6: TARGETS=${TARGETS:-"172.31.0.21:8080,172.31.0.22:8080"}
7: GNMIC_BIN=${GNMIC_BIN:-gnmic}
8: GNMI_USER=${GNMI_USER:-admin}
9: GNMI_PASS=${GNMI_PASS:-admin}
10: GNMI_CACERT=${GNMI_CACERT:-./secrets/ca.crt}
11: GNMI_CERT=${GNMI_CERT:-./secrets/gnmi.crt}
12: GNMI_KEY=${GNMI_KEY:-./secrets/gnmi.key}
13: GNMI_ENCODING=${GNMI_ENCODING:-JSON_IETF}
14: 
15: # read current value for telemetry port via gNMI Get
16: get_port() {
17:   local addr=$1
18:   "$GNMIC_BIN" --address "$addr" --timeout 5s --username "$GNMI_USER" --password "$GNMI_PASS" \
19:     --tls --skip-verify --encoding "$GNMI_ENCODING" --cacert "$GNMI_CACERT" --cert "$GNMI_CERT" --key "$GNMI_KEY" \
20:     get --path "/sonic-telemetry:sonic-telemetry/TELEMETRY/SERVER[name=gnmi]/port" -o json | jq -r '..|.val? // empty' | tail -n1
21: }
22: 
23: # set port to a specific value
24: set_port() {
25:   local addr=$1 val=$2
26:   "$GNMIC_BIN" --address "$addr" --timeout 5s --username "$GNMI_USER" --password "$GNMI_PASS" \
27:     --tls --skip-verify --encoding "$GNMI_ENCODING" --cacert "$GNMI_CACERT" --cert "$GNMI_CERT" --key "$GNMI_KEY" \
28:     set --update-path "/sonic-telemetry:sonic-telemetry/TELEMETRY/SERVER[name=gnmi]/port" --update-value "$val"
29: }
30: 
31: # restart SONiC containers by containerlab label
32: restart_sonic_containers() {
33:   # Find containers with ainetops.owner=ainetops and sonic in image or name
34:   local ids; ids=$(docker ps -q --filter "label=ainetops.owner=ainetops") || true
35:   if [[ -z "$ids" ]]; then echo "[persist] no AINETOPS containers found" >&2; return 1; fi
36:   docker restart $ids >/dev/null
37: }
38: 
39: main() {
40:   IFS=',' read -ra tgts <<<"$TARGETS"
41:   # Choose a test value unlikely to be default
42:   local new_port=${GNMI_PERSIST_TEST_PORT:-8099}
43:   echo "[persist] setting telemetry port to $new_port on all targets"
44:   for t in "${tgts[@]}"; do set_port "$t" "$new_port"; done
45:   echo "[persist] restarting SONiC containers"
46:   restart_sonic_containers
47:   echo "[persist] verifying telemetry port persists after restart"
48:   for t in "${tgts[@]}"; do
49:     val=$(get_port "$t")
50:     echo "[persist] $t port=$val"
51:     [[ "$val" == "$new_port" ]] || { echo "[persist] persistence failed on $t: expected $new_port got $val" >&2; exit 1; }
52:   done
53:   echo "[persist] persistence verified"
54: }
55: 
56: if [[ ${1:-} == "--run" ]]; then
57:   shift
58:   main "$@"
59: fi
