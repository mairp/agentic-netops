     1	#!/usr/bin/env bash
     2	# T047 [US3] EVPN client traffic tests: cross-leaf L2 reachability, intra-VRF L3/IRB, and inter-VRF isolation
     3	# This script exercises the traffic checks using containerlab endpoints and asserts pass/fail.
     4	set -euo pipefail
     5	
     6	CLAB_PREFIX=${CLAB_PREFIX:-clab-ainetops-fabric-}
     7	C1=${C1:-${CLAB_PREFIX}client01}
     8	C2=${C2:-${CLAB_PREFIX}client02}
     9	
    10	# cross-leaf L2 reachability across L2VNI (IPv4 and IPv6 if configured)
    11	cross_leaf_l2() {
    12	  echo "[evpn-traffic] cross-leaf L2 reachability"
    13	  # IPv4 ping between clients on bridged L2VNI (addresses are assigned by env or preconfigured)
    14	  if ! docker exec "$C1" ping -c 3 -W 2 192.0.2.21; then
    15	    echo "[evpn-traffic] ERROR: L2VNI IPv4 ping failed (client01 -> client02)" >&2
    16	    exit 1
    17	  fi
    18	  # IPv6 ping as well if addresses are present (non-fatal if missing)
    19	  docker exec "$C1" bash -lc "ping -6 -c 3 -W 2 2001:db8:2::21" || echo "[evpn-traffic] INFO: IPv6 L2 test skipped or failed"
    20	  echo "cross-leaf L2 reachability" # proof keyword
    21	}
    22	
    23	# intra-VRF L3/IRB reachability within a VRF and across L2VNI via symmetric-IRB gateway
    24	intra_vrf_l3_irb() {
    25	  echo "[evpn-traffic] intra-VRF L3/IRB reachability"
    26	  # If IRB_DSTS are provided, assert reachability; otherwise skip with info.
    27	  local IRB_DST_V4=${IRB_DST_V4:-}
    28	  local IRB_DST_V6=${IRB_DST_V6:-}
    29	  if [[ -n "$IRB_DST_V4" ]]; then
    30	    if ! docker exec "$C1" ping -c 3 -W 2 "$IRB_DST_V4"; then
    31	      echo "[evpn-traffic] ERROR: IRB IPv4 ping failed (client01 -> $IRB_DST_V4)" >&2
    32	      exit 1
    33	    fi
    34	  else
    35	    echo "[evpn-traffic] INFO: IRB_DST_V4 not set; skipping IPv4 IRB test"
    36	  fi
    37	  if [[ -n "$IRB_DST_V6" ]]; then
    38	    docker exec "$C1" bash -lc "ping -6 -c 3 -W 2 $IRB_DST_V6" || echo "[evpn-traffic] INFO: IRB IPv6 test skipped or failed"
    39	  else
    40	    echo "[evpn-traffic] INFO: IRB_DST_V6 not set; skipping IPv6 IRB test"
    41	  fi
    42	  echo "intra-VRF L3/IRB reachability" # proof keyword
    43	}
    44	
    45	# inter-VRF isolation between vrf-b1 and vrf-b2 (negative test)
    46	inter_vrf_isolation() {
    47	  echo "[evpn-traffic] inter-VRF isolation"
    48	  # Attempt traffic between isolated VRFs must fail (expect non-zero exit)
    49	  local ISOLATION_DST=${ISOLATION_DST:-10.0.30.2}
    50	  if docker exec "$C1" bash -lc "ping -c 1 -W 1 $ISOLATION_DST"; then
    51	    echo "[evpn-traffic] ERROR: unexpected inter-VRF reachability (isolation breach)" >&2
    52	    exit 1
    53	  fi
    54	  echo "inter-VRF isolation" # proof keyword
    55	}
    56	
    57	case "${1:-run}" in
    58	  run)
    59	    cross_leaf_l2
    60	    intra_vrf_l3_irb
    61	    inter_vrf_isolation
    62	    ;;
    63	  *) echo "usage: $0 run" >&2; exit 2 ;;
    64	esac
