40	search_all() {
41	  local pattern=$1
42	  # Scan only tracked repository files to avoid session artifacts under .wiggum/**/attempts/**
43	  # and other untracked logs; this enforces the policy over the source tree.
44	  if command -v git >/dev/null 2>&1 && git rev-parse --is-inside-work-tree >/dev/null 2>&1; then
45	    local files
46	    # Use nul-delimited to handle spaces; prefer ripgrep when present
47	    if [[ $have_rg -eq 1 ]]; then
48	      git ls-files -z | xargs -0 rg -i -n -e "$pattern" || true
49	    else
50	      git ls-files -z | xargs -0 grep -nE -I "$pattern" || true
51	    fi
52	  else
53	    # Fallback: whole tree except .git
54	    if [[ $have_rg -eq 1 ]]; then
55	      rg -i -n --hidden --no-ignore -g '!*/.git/*' -e "$pattern" . || true
56	    else
57	      grep -RinE --exclude-dir=.git -I "$pattern" . || true
58	    fi
59	  fi
60	}
