     1	// Package compat validates the compatibility set: image, schema, mapping, and upstream API versions.
     2	package compat
     3	
     4	import (
     5		"fmt"
     6	)
     7	
     8	// Set captures the five-part compatibility set published in deployment metadata and status.
     9	type Set struct {
    10		SonicImage          string
    11		OpenConfigCommit    string
    12		SonicNativeCommit   string
    13		MappingVersion      string
    14		UpstreamAPIVersions map[string]string // e.g., {"kubenet":"bae1c487...", "kuid":"7528e815...", "sdc":"v0.31.0"}
    15	}
    16	
    17	// ValidationError classifies terminal vs transient mismatches.
    18	type ValidationError struct {
    19		Reason  string // e.g., SchemaMismatch, CapabilityMissing
    20		Message string
    21	}
    22	
    23	func (e *ValidationError) Error() string { return fmt.Sprintf("%s: %s", e.Reason, e.Message) }
    24	
    25	// Validate performs offline validation of the compatibility set against the pinned contract
    26	// in versions.lock.yaml and optionally a discovered target capability set.
    27	func Validate(set Set, discovered map[string]bool) error {
    28		// Minimal scaffold: ensure non-empty pins and required SRv6 capability flag when mapping requires it.
    29		if set.SonicImage == "" || set.OpenConfigCommit == "" || set.SonicNativeCommit == "" || set.MappingVersion == "" {
    30			return &ValidationError{Reason: "SchemaMismatch", Message: "incomplete compatibility set"}
    31		}
    32		if discovered != nil {
    33			// SAI SRv6 capability gate (placeholder key)
    34			if requiresSRv6(set.MappingVersion) && !discovered["sai.srv6"] {
    35				return &ValidationError{Reason: "CapabilityMissing", Message: "SRv6 not supported by target"}
    36			}
    37		}
    38		return nil
    39	}
    40	
    41	func requiresSRv6(mappingVersion string) bool { return true }
