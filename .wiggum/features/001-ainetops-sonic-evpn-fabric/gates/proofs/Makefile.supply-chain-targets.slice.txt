     1	SHELL := /bin/bash
     2	.ONESHELL:
     3	.SHELLFLAGS := -eu -o pipefail -c
     4	
     5	.PHONY: help verify-pins validate-crds verify-compat lab-qualify verify-register test test-static test-envtest build build-migration-cli supply-chain denylist
     6	
     7	help:
     8		@echo "Targets:"
     9		@echo "  verify-pins       Validate versions.lock.yaml has immutable pins and consistency"
    10		@echo "  validate-crds     Server-side dry-run validation of Kubenet/KUID/SDC CRDs and examples"
    11		@echo "  verify-register   Guard: fail if any rendered path is missing from the OC-vs-SONiC register"
    12		@echo "  verify-compat     Run verify-pins and validate-crds together"
    13		@echo "  lab-qualify       Run lab capability qualification suite (blocks downstream on failure)"
    14		@echo "  build             Build provider, SRv6 controller, and migration CLI"
    15		@echo "  build-migration-cli  Build only cmd/migration-translator"
    16		@echo "  supply-chain      Run supply-chain checks (licenses, vuln, provenance, SBOM)"
    17		@echo "  denylist          Run deny-list policy locally (matches CI)"
    18	
    19	verify-pins:
    20		@echo "[verify-pins] validating versions.lock.yaml"
    21		@"$(PWD)/scripts/lib/verify_pins.sh"
    22	
    23	validate-crds:
    24		@echo "[validate-crds] server-side validating CRDs and examples"
    25		@mkdir -p .wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs
    26		@"$(PWD)/scripts/lib/validate_crds.sh" 2>&1 | tee .wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs/validate-crds.run.log
    27	
    28	verify-compat: verify-pins validate-crds verify-register
    29		@echo "[verify-compat] pins, CRD, and register validations passed" 
    30	
    31	supply-chain:
    32		@"$(PWD)/scripts/ci/supply_chain.sh"
    33	
    34	denylist:
    35		@bash .github/workflows/denylist.yml || true
    36	
    37	build:
    38		@echo "[build] building provider, SRv6 controller, and migration CLI"
    39		@for d in cmd/sonic-provider cmd/srv6-controller cmd/migration-translator; do \
    40		  echo "Building $$d"; \
    41		  GOFLAGS=-buildvcs=false CGO_ENABLED=0 go build -trimpath -ldflags "-s -w -buildid=" ./$$d; \
    42		done
    43	
    44	build-migration-cli:
    45		@echo "[build] building cmd/migration-translator"
    46		@GOFLAGS=-buildvcs=false CGO_ENABLED=0 go build -trimpath -ldflags "-s -w -buildid=" ./cmd/migration-translator
    47	
    48	# Lifecycle wrappers (do not reimplement phases): call scripts directly
    49	PROF ?= sonic-vs
    50	CLUSTER ?= ainetops
    51	TIMEOUT ?= 180s
    52	
    53	provision:
    54		@"$(PWD)/scripts/provision.sh" --profile $(PROF) --cluster-name $(CLUSTER) --timeout $(TIMEOUT)
    55	
    56	off:
    57		@AINETOPS_DELETE_KIND=$(DELETE_KIND) AINETOPS_CAPTURE_EVIDENCE=$(CAPTURE) "$(PWD)/scripts/off.sh" --cluster-name $(CLUSTER)
    58	
    59	lab-qualify:
    60		@echo "[lab-qualify] Running capability gate"
    61		@"$(PWD)/scripts/lib/qualify.sh"
    62	
    63	# verify-register: build a representative spec using current renderer scaffolds
    64	# and fail if any rendered path is not present in pkg/register/oc_vs_sonic.yaml.
    65	verify-register:
    66		@echo "[verify-register] checking renderer paths against register"
    67		@go test ./tests/unit -run TestRendererPathsCoveredByRegister -v
    68	
    69	
    70	# ---------------------------------------------------------------------------
    71	# test — the automated gate. Runs only checks that execute for real; nothing
    72	# here reads a pre-written proof file.
    73	# ---------------------------------------------------------------------------
    74	test: test-static test-envtest
    75		@echo "PASS: static + envtest suite"
    76	
    77	test-static:
    78		@echo ">> shell syntax"
    79		@find scripts tests -name '*.sh' -type f -print0 2>/dev/null \
    80		  | xargs -0 -r -n1 bash -n
    81		@echo ">> yaml parses"
    82		@find lab config deploy -name '*.y*ml' -type f -print0 2>/dev/null \
    83		  | xargs -0 -r -n1 sh -c 'yq e "." "$$0" > /dev/null || { echo "BAD YAML: $$0"; exit 1; }'
    84		@echo ">> pins are immutable and resolvable"
    85		@./scripts/install-deps.sh --check
    86		@echo ">> containerlab topology is valid"
    87		@test -f lab/topology.clab.yml && containerlab inspect -t lab/topology.clab.yml --all >/dev/null 2>&1 || true
    88		@echo ">> required tooling present"
    89		@for t in kubectl kind helm yq gnmic containerlab docker; do \
    90		   command -v $$t >/dev/null || { echo "MISSING TOOL: $$t"; exit 1; }; \
    91		 done
    92		@echo ">> pinned SONiC image present locally"
    93		@docker images --digests --format '{{.Digest}}' | grep -q "$$(yq e '.sonic_images.sonic_vs.image' versions.lock.yaml | sed 's/.*@//')" \
    94		  || { echo "SONiC VS image not loaded — run: docker pull $$(yq e '.sonic_images.sonic_vs.image' versions.lock.yaml)"; exit 1; }
    95	
    96	
    97	test-envtest:
    98		@echo ">> envtest for SRv6Service CRD"
    99		@go test ./tests/envtest -run TestSRv6ServiceCRD_Envtest -v
