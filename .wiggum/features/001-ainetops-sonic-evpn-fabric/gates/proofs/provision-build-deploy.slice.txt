     1	# Ordered phases (Phase 3): verify pins, validate CRDs, create Kind, attach mgmt net, run capability gate
     2	if [[ -f "${REPO_ROOT}/Makefile" ]]; then
     3	  make -C "${REPO_ROOT}" verify-compat
     4	fi
     5	
     6	# Ensure external management network exists (idempotent)
     7	if command -v docker >/dev/null 2>&1; then
     8	  docker network inspect ainetops-mgmt >/dev/null 2>&1 || docker network create --label ainetops.owner=ainetops ainetops-mgmt
     9	fi
    10	
    11	# Ensure Kind cluster exists and is using pinned image; attach nodes to mgmt network
    12	if [[ -x "${LIB_DIR}/kind.sh" ]]; then
    13	  "${LIB_DIR}/kind.sh" ensure
    14	  "${LIB_DIR}/kind.sh" attach-mgmt
    15	  "${LIB_DIR}/kind.sh" verify-context
    16	else
    17	  echo "[provision] WARN: kind helper not found; skipping Kind cluster ensure" >&2
    18	fi
    19	
    20	# Deploy lab topology with containerlab if available
    21	if [[ -x "${LIB_DIR}/containerlab.sh" ]]; then
    22	  "${LIB_DIR}/containerlab.sh" deploy || { echo "[provision] containerlab deploy failed" >&2; exit 1; }
    23	  # Optional inspect step for diagnostics
    24	  "${LIB_DIR}/containerlab.sh" inspect || true
    25	else
    26	  echo "[provision] WARN: containerlab helper not found; skipping lab deploy" >&2
    27	fi
    28	
    29	# Install least-privilege RBAC base
    30	if [[ -x "${LIB_DIR}/rbac.sh" ]]; then
    31	  "${LIB_DIR}/rbac.sh"
    32	fi
    33	# Install pinned Kubenet/KUID and SDC into Kind and wait basic readiness
    34	if [[ -x "${REPO_ROOT}/deploy/kubenet/install.sh" ]]; then
    35	  "${REPO_ROOT}/deploy/kubenet/install.sh"
    36	fi
    37	if [[ -x "${REPO_ROOT}/deploy/sdc/install.sh" ]]; then
    38	  "${REPO_ROOT}/deploy/sdc/install.sh"
    39	fi
    40	
    41	# Build, load, and deploy provider and srv6-controller images into Kind (T041)
    42	if command -v docker >/dev/null 2>&1 && command -v kind >/dev/null 2>&1 && command -v kubectl >/dev/null 2>&1; then
    43	  echo "[provision] building controller images"
    44	  docker build -t ainetops-sonic-provider:dev -f "${REPO_ROOT}/cmd/sonic-provider/Dockerfile" "${REPO_ROOT}"
    45	  docker build -t ainetops-srv6-controller:dev -f "${REPO_ROOT}/cmd/srv6-controller/Dockerfile" "${REPO_ROOT}"
    46	  echo "[provision] loading images into Kind"
    47	  kind load docker-image ainetops-sonic-provider:dev --name "${AINETOPS_CLUSTER_NAME:-ainetops}" || true
    48	  kind load docker-image ainetops-srv6-controller:dev --name "${AINETOPS_CLUSTER_NAME:-ainetops}" || true
    49	  echo "[provision] deploying controllers"
    50	  # Use dev images by overriding the container image in manifests
    51	  kubectl --context "kind-${AINETOPS_CLUSTER_NAME:-ainetops}" -n ainetops-system apply -f "${REPO_ROOT}/deploy/ainetops/manifests/provider.yaml"
    52	  kubectl --context "kind-${AINETOPS_CLUSTER_NAME:-ainetops}" -n ainetops-system apply -f "${REPO_ROOT}/deploy/ainetops/manifests/srv6-controller.yaml"
    53	  kubectl --context "kind-${AINETOPS_CLUSTER_NAME:-ainetops}" -n ainetops-system set image deploy/ainetops-sonic-provider provider=ainetops-sonic-provider:dev || true
    54	  kubectl --context "kind-${AINETOPS_CLUSTER_NAME:-ainetops}" -n ainetops-system set image deploy/ainetops-srv6-controller srv6-controller=ainetops-srv6-controller:dev || true
    55	  echo "[provision] waiting for controller pods ready"
    56	  kubectl --context "kind-${AINETOPS_CLUSTER_NAME:-ainetops}" -n ainetops-system rollout status deploy/ainetops-sonic-provider --timeout=120s || true
    57	  kubectl --context "kind-${AINETOPS_CLUSTER_NAME:-ainetops}" -n ainetops-system rollout status deploy/ainetops-srv6-controller --timeout=120s || true
    58	  # Capture independent observation proof
    59	  mkdir -p "${REPO_ROOT}/.wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs"
    60	  kubectl --context "kind-${AINETOPS_CLUSTER_NAME:-ainetops}" -n ainetops-system get deploy,po,svc -o wide \
    61	    | nl -ba > "${REPO_ROOT}/.wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs/kubectl-get-ainetops-system.txt"
    62	fi
