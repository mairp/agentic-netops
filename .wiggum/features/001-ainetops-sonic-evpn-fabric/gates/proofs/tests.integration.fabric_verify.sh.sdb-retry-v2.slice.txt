     1	  # an empty body — yang_paths_suite.sh accepts the same signal. Every other rpc
     2	  # error (TLS, Unavailable, Unauthenticated) is a transport failure and must NOT
     3	  # be readable as proof of absence.
     4	  if grep -q 'code = NotFound' <<<"$out"; then
     5	    printf 'ABSENT'
     6	    return 0
     7	  fi
     8	  if [[ $e -ne 0 ]] || grep -qE 'rpc error|^Error:' <<<"$out"; then
     9	    # Emit the underlying rpc error so a QUERY_FAILED in the evidence log
    10	    # self-explains (cycle-1 of the 2026-09-01 reconciliation left an
    11	    # unexplainable auth failure because only the literal marker was logged).
    12	    grep -E 'rpc error|^Error:' <<<"$out" | head -1 | sed "s/^/[$t] sonic-db query error: /" >&2 || true
    13	    # Unauthenticated/Unavailable right after a provision is a transition
    14	    # artifact (telemetry was just restarted by the bootstrap/qualify suites;
    15	    # creds themselves are consistent on a settled lab — verified 2026-09-01
    16	    # 07:35). A single refetch+retry was NOT always enough: in the 2026-09-01
    17	    # cycle run, cycle-2's BGP_NEIGHBOR queries failed after one retry while the
    18	    # identical queries passed on the settled lab and in cycle-3 — so retry up
    19	    # to 3 times with increasing backoff before declaring the query unaskable.
    20	    if grep -qE 'Unauthenticated|Unavailable' <<<"$out"; then
    21	      local attempt backoff u2 p2
    22	      for attempt in 1 2 3; do
    23	        backoff=$(( 8 * attempt + (attempt - 1) * 2 ))  # 8s, 18s, 28s
    24	        u2=$(kubectl --context "$KUBE_CTX" -n ainetops-system get secret gnmi-lab-creds -o jsonpath='{.data.username}' 2>/dev/null | base64 -d 2>/dev/null || true)
    25	        p2=$(kubectl --context "$KUBE_CTX" -n ainetops-system get secret gnmi-lab-creds -o jsonpath='{.data.password}' 2>/dev/null | base64 -d 2>/dev/null || true)
    26	        if [[ -z "$u2" || -z "$p2" ]]; then break; fi
    27	        sleep "$backoff"
    28	        out=$("$GNMIC_BIN" --address "$t" --timeout 10s --username "$u2" --password "$p2" --tls-ca "$GNMI_CACERT" --tls-cert "$GNMI_CERT" --tls-key "$GNMI_KEY" --encoding "$GNMI_ENCODING" get --path "/$table" --target CONFIG_DB 2>&1)
    29	        e=$?
    30	        if [[ $e -eq 0 ]] && ! grep -qE 'rpc error|^Error:' <<<"$out"; then
    31	          printf '%s' "$out" | tr -d ' \n\t'
    32	          return 0
    33	        fi
    34	        grep -E 'rpc error|^Error:' <<<"$out" | head -1 | sed "s/^/[$t] sonic-db retry $attempt error: /" >&2 || true
    35	        # Stop early if the error class changed (no longer a transition artifact).
    36	        grep -qE 'Unauthenticated|Unavailable' <<<"$out" || break
    37	      done
    38	    fi
    39	    printf 'QUERY_FAILED'
    40	    return 0
    41	  fi
