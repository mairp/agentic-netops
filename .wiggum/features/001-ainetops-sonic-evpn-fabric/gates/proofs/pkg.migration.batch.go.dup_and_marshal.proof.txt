10: // RenderBatch validates a batch all-or-nothing and returns one YAML document per
11: // valid service in the input order. On validation failure, returns a ValidationError
12: // with aggregated causes and no output documents.
13: func RenderBatch(inputs []ServiceInput) ([]string, error) {
14: 	// Detect duplicates
15: 	ids := map[string]int{}
16: 	for i, in := range inputs {
17: 		if _, ok := ids[in.ServiceID]; ok {
18: 			ids[in.ServiceID] = -1
19: 		} else {
20: 			ids[in.ServiceID] = i
21: 		}
22: 	}
23: 	var causes []string
24: 	for i := range inputs {
25: 		in := inputs[i]
26: 		dup := false
27: 		if idx, ok := ids[in.ServiceID]; ok && idx == -1 { dup = true }
28: 		if err := in.ValidateAllOrNothing(i, dup); err != nil {
29: 			if ve, ok := err.(*ValidationError); ok {
30: 				causes = append(causes, ve.Causes...)
31: 			} else {
32: 				causes = append(causes, err.Error())
33: 			}
34: 		}
35: 	}
36: 	if len(causes) > 0 {
37: 		return nil, &ValidationError{Causes: causes}
38: 	}
39: 	out := make([]string, 0, len(inputs))
40: 	for i := range inputs {
41: 		in := inputs[i]
42: 		bundle, err := Translate(&in)
43: 		if err != nil { return nil, err }
44: 		out = append(out, bundle.NetworkYAML)
45: 	}
46: 	return out, nil
47: }

49: // MarshalError produces a deterministic JSON string for structured validation errors.
50: // It disables HTML escaping so comparison substrings like ">=1" remain readable
51: // and stable in tests and CLI output.
52: func MarshalError(err error) string {
53: 	if err == nil { return "" }
54: 	m := map[string]any{"error": "validation"}
55: 	if ve, ok := err.(*ValidationError); ok {
56: 		m["causes"] = ve.Causes
57: 	} else {
58: 		m["causes"] = []string{err.Error()}
59: 	}
60: 	// Use an Encoder with HTML escaping disabled to avoid sequences like \u003e
61: 	// for '>' which make human/fixture comparison brittle.
62: 	var buf bytes.Buffer
63: 	enc := json.NewEncoder(&buf)
64: 	enc.SetEscapeHTML(false)
65: 	enc.SetIndent("", "  ")
66: 	_ = enc.Encode(m)
67: 	// json.Encoder adds a trailing newline; trim it for stable output.
68: 	out := buf.String()
69: 	if len(out) > 0 && out[len(out)-1] == '\n' { out = out[:len(out)-1] }
70: 	return out
71: }
