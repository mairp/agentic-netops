2	package compat
3	
4	import (
5		"regexp"
6	)
7	
8	var sha1re = regexp.MustCompile(`^[0-9a-f]{7,40}$`)
9	var semverRe = regexp.MustCompile(`^v\d+\.\d+\.\d+(-[0-9A-Za-z.-]+)?$`)
10	
11	// FromAnnotations extracts a compatibility Set from labels/annotations maps.
12	func FromAnnotations(ann map[string]string) Set {
13		set := Set{
14			SonicImage:        ann["ainetops.dev/sonic-image"],
15			OpenConfigCommit:  ann["ainetops.dev/openconfig-commit"],
16			SonicNativeCommit: ann["ainetops.dev/sonic-native-commit"],
17			MappingVersion:    ann["ainetops.dev/mapping-version"],
18			UpstreamAPIVersions: map[string]string{
19				"kubenet": ann["ainetops.dev/kubenet-commit"],
20				"kuid":    ann["ainetops.dev/kuid-commit"],
21				"sdc":     ann["ainetops.dev/sdc-release"],
22			},
23		}
24		return set
25	}
26	
27	// ValidatePins performs stricter checks on the Set pins for shape and presence.
28	func ValidatePins(set Set) error {
29		if set.SonicImage == "" {
30			return &ValidationError{Reason: "SchemaMismatch", Message: "missing sonic image pin"}
31		}
32		if !sha1re.MatchString(set.OpenConfigCommit) || !sha1re.MatchString(set.SonicNativeCommit) {
33			return &ValidationError{Reason: "SchemaMismatch", Message: "schema pins must be commit SHAs"}
34		}
35		if !semverRe.MatchString(set.UpstreamAPIVersions["sdc"]) {
36			return &ValidationError{Reason: "SchemaMismatch", Message: "sdc release must be semver"}
37		}
38		return nil
39	}
40	
41	// ValidateContracts ensures that pinned telemetry/topology label contracts are present.
42	func ValidateContracts(labels map[string]string) error {
43		if labels["ainetops.dev/topology-label-contract"] == "" {
44			return &ValidationError{Reason: "SchemaMismatch", Message: "missing topology label contract pin"}
45		}
46		if labels["ainetops.dev/telemetry-label-contract"] == "" {
47			return &ValidationError{Reason: "SchemaMismatch", Message: "missing telemetry label contract pin"}
48		}
49		return nil
50	}
51	
52	// FullValidate runs pin checks plus optional capability gates.
53	func FullValidate(set Set, labels map[string]string, discovered map[string]bool) error {
54		if err := ValidatePins(set); err != nil {
55			return err
56		}
57		if err := ValidateContracts(labels); err != nil {
58			return err
59		}
60		if err := Validate(set, discovered); err != nil {
61			return err
62		}
63		return nil
64	}
65	
66	func ReasonFor(err error) string {
67		if v, ok := err.(*ValidationError); ok {
68			return v.Reason
69		}
70		return "ValidationFailed"
71	}
