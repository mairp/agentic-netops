    21	// TestProviderFinalization simulates deletion flow and ensures ordered finalization evidence.
    22	func TestProviderFinalization(t *testing.T) {
    23		scheme := runtime.NewScheme()
    24		_ = clientgoscheme.AddToScheme(scheme)
    25		_ = kubenet.AddToScheme(scheme)
    26		_ = sdc.AddToScheme(scheme)
    27		c := fake.NewClientBuilder().WithScheme(scheme).Build()
    28		rec := &sonicprovider.Reconciler{Client: c, Scheme: scheme}
    29	
    30		ctx := context.TODO()
    31		nd := &kubenet.NetworkDevice{ObjectMeta: metav1.ObjectMeta{Name: "leaf01", Namespace: "default"}}
    32		if err := c.Create(ctx, nd); err != nil { t.Fatalf("create nd: %v", err) }
    33		// Add finalizer and a Config owned by ND
    34		if err := c.Get(ctx, types.NamespacedName{Namespace: nd.Namespace, Name: nd.Name}, nd); err != nil { t.Fatalf("get nd: %v", err) }
    35		nd.Finalizers = append(nd.Finalizers, "ainetops.dev/finalizer")
    36		if err := c.Update(ctx, nd); err != nil { t.Fatalf("update nd: %v", err) }
    37		cfg := &sdc.Config{ObjectMeta: metav1.ObjectMeta{Name: "nd-leaf01", Namespace: "default"}}
    38		if err := c.Create(ctx, cfg); err != nil { t.Fatalf("create cfg: %v", err) }
    39		// Mark ND as deleting by issuing a Delete (fake client sets deletionTimestamp when finalizers exist)
    40		if err := c.Delete(ctx, nd); err != nil { t.Fatalf("delete nd: %v", err) }
    41	
    42		// First reconcile should issue delete; in fake client the Config may be removed immediately
    43		if _, err := rec.Reconcile(ctx, ctrl.Request{NamespacedName: types.NamespacedName{Namespace: "default", Name: "leaf01"}}); err != nil { t.Fatalf("reconcile: %v", err) }
    44		// If the Config still exists, delete it to simulate confirmation
    45		if err := c.Get(ctx, types.NamespacedName{Namespace: "default", Name: "nd-leaf01"}, cfg); err == nil {
    46			if err := c.Delete(ctx, cfg); err != nil { t.Fatalf("delete cfg: %v", err) }
    47		}
    48		// Second reconcile should record finalized-at and remove finalizer
    49		if _, err := rec.Reconcile(ctx, ctrl.Request{NamespacedName: types.NamespacedName{Namespace: "default", Name: "leaf01"}}); err != nil { t.Fatalf("reconcile2: %v", err) }
    50		if err := c.Get(ctx, types.NamespacedName{Namespace: nd.Namespace, Name: nd.Name}, nd); err == nil {
    51			if nd.Annotations["ainetops.dev/finalized-at"] == "" { t.Fatalf("expected finalized-at annotation") }
    52			if contains(nd.Finalizers, "ainetops.dev/finalizer") { t.Fatalf("expected finalizer removed") }
    53		}
    54	}
