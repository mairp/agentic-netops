    54	kind::verify_node_image() {
    55	  # Verify that nodes run the pinned image from versions.lock.yaml
    56	  local pin base image_ok=1
    57	  pin="${NODE_IMAGE_PIN}"
    58	  if [[ -z "${pin}" ]]; then
    59	    echo "[kind] WARN: no node image pin found in versions.lock.yaml" >&2
    60	    return 0
    61	  fi
    62	  while read -r n; do
    63	    [[ -z "${n}" ]] && continue
    64	    base=$(docker inspect -f '{{.Config.Image}}' "${n}" 2>/dev/null || true)
    65	    if [[ "${base%@*}" != "${pin%@*}" || "${base#*@}" != "${pin#*@}" ]]; then
    66	      echo "[kind] ERROR: node ${n} image ${base} != pinned ${pin}" >&2
    67	      image_ok=0
    68	    fi
    69	  done < <(kind::nodes)
    70	  (( image_ok == 1 )) || { echo "[kind] node image mismatch" >&2; exit 1; }
    71	}
