     9	# Utility: IPv4 dotted-quad to integer
    10	preflight::ip2int() {
    11	  local IFS=.; read -r a b c d <<<"$1"; echo $(( (a<<24) + (b<<16) + (c<<8) + d ))
    12	}

    14	# Utility: derive [start,end] integer range from CIDR
    15	preflight::cidr_range() {
    16	  local cidr=$1 ip=${1%/*} mask=${1#*/}
    17	  local ipn; ipn=$(preflight::ip2int "$ip")
    18	  local maskn=$(( 0xFFFFFFFF << (32-mask) & 0xFFFFFFFF ))
    19	  local start=$(( ipn & maskn ))
    20	  local end=$(( start | (~maskn & 0xFFFFFFFF) ))
    21	  echo "$start $end"
    22	}

    24	# Return 0 (overlap) or 1 (disjoint)
    25	preflight::ranges_overlap() {
    26	  local a1=$1 a2=$2 b1=$3 b2=$4
    27	  if (( a1 <= b2 && b1 <= a2 )); then return 0; else return 1; fi
    28	}

    30	preflight::address_conflicts() {
    31	  # Ensure management network CIDR does not overlap pod/service CIDRs
    32	  local mgmt_cidr=${AINETOPS_MGMT_CIDR:-172.31.0.0/16}
    33	  local pod_cidr=${AINETOPS_POD_CIDR:-10.244.0.0/16}
    34	  local svc_cidr=${AINETOPS_SERVICE_CIDR:-10.96.0.0/12}
    35	  read -r m1 m2 < <(preflight::cidr_range "$mgmt_cidr")
    36	  read -r p1 p2 < <(preflight::cidr_range "$pod_cidr")
    37	  read -r s1 s2 < <(preflight::cidr_range "$svc_cidr")
    38	  if preflight::ranges_overlap "$m1" "$m2" "$p1" "$p2"; then
    39	    preflight::die "Management CIDR $mgmt_cidr overlaps Pod CIDR $pod_cidr"
    40	  fi
    41	  if preflight::ranges_overlap "$m1" "$m2" "$s1" "$s2"; then
    42	    preflight::die "Management CIDR $mgmt_cidr overlaps Service CIDR $svc_cidr"
    43	  fi
    44	}

    46	preflight::mtu() {
    47	  # Verify host MTU accommodates VXLAN overhead (> 1500 suggested); warn only in Phase 1
    48	  local maxmtu=0
    49	  while read -r mtu; do (( mtu > maxmtu )) && maxmtu=$mtu; done < <(ip -o link 2>/dev/null | awk -F'mtu ' '{print $2}' | awk '{print $1}')
    50	  if (( maxmtu < 1500 )); then preflight::warn "maximum host MTU $maxmtu < 1500; VXLAN overhead may break traffic"; fi
    51	}

    53	preflight::kvm_check() {
    54	  local profile=${AINETOPS_PROFILE:-sonic-vs}
    55	  if [[ "$profile" == "sonic-vm" ]]; then
    56	    # Require KVM when sonic-vm profile selected
    57	    [[ -e /dev/kvm ]] || preflight::die "/dev/kvm not present for sonic-vm profile"
    58	  fi
    59	}
