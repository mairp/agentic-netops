      1	#!/usr/bin/env bash
      2	# Install pinned SDC CRDs/controllers into Kind and wait for readiness (Phase 3)
      3	# This script applies the ACTUAL upstream SDC CRDs pinned by versions.lock.yaml release.
      4	set -euo pipefail
      5	DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd)
      6	ROOT=$(cd -- "${DIR}/../.." && pwd)
      7	KIND_CONTEXT="kind-${AINETOPS_CLUSTER_NAME:-ainetops}"
      8	
      9	command -v kubectl >/dev/null 2>&1 || { echo "missing kubectl" >&2; exit 1; }
     10	command -v awk >/dev/null 2>&1 || { echo "missing awk" >&2; exit 1; }
     11	
     12	LOCK_FILE="${ROOT}/versions.lock.yaml"
     13	section() { awk -v s="^$1:$" 'f; $0~s{f=1} f && /^[^[:space:]]/{if(!p){p=1;print}else exit} f && p{print}' "${LOCK_FILE}"; }
     14	
     15	# Read core SDC release (sdcio/sdc)
     16	sdc_release=$(awk '
     17	  $1=="sdc:" {sec=1; next}
     18	  sec && /^[^[:space:]]/ {exit}
     19	  sec && $1=="core:" {in_core=1; next}
     20	  in_core && /^[^[:space:]]/ {exit}
     21	  in_core && $1=="release:" {print $2; exit}
     22	' "${LOCK_FILE}")
     23	
     24	SDC_CRDS=(
     25	  "https://raw.githubusercontent.com/sdcio/sdc/${sdc_release}/deploy/crds/sdc.sdcio.dev_schemas.yaml"
     26	  "https://raw.githubusercontent.com/sdcio/sdc/${sdc_release}/deploy/crds/sdc.sdcio.dev_configs.yaml"
     27	  "https://raw.githubusercontent.com/sdcio/sdc/${sdc_release}/deploy/crds/sdc.sdcio.dev_targets.yaml"
     28	)
     29	
     30	# Apply CRDs from pinned upstream
     31	for u in "${SDC_CRDS[@]}"; do
     32	  kubectl --context "${KIND_CONTEXT}" apply -f "$u"
     33	done
     34	
     35	# Apply SDC components (Deployments/Services) and PVCs
     36	kubectl --context "${KIND_CONTEXT}" apply -f "${ROOT}/deploy/sdc/components.yaml"
     37	
     38	# PVCs for SDC state (ensure created)
     39	cat <<EOF | kubectl --context "${KIND_CONTEXT}" apply -f -
     40	apiVersion: v1
     41	kind: PersistentVolumeClaim
     42	metadata:
     43	  name: sdc-data
     44	  namespace: sdc-system
     45	spec:
     46	  accessModes: [ "ReadWriteOnce" ]
     47	  resources:
     48	    requests:
     49	      storage: 1Gi
     50	---
     51	apiVersion: v1
     52	kind: PersistentVolumeClaim
     53	metadata:
     54	  name: sdc-cache
     55	  namespace: sdc-system
     56	spec:
     57	  accessModes: [ "ReadWriteOnce" ]
     58	  resources:
     59	    requests:
     60	      storage: 1Gi
     61	EOF
     62	
     63	# Wait for CRDs (basic)
     64	for crd in schemas.sdc.sdcio.dev configs.sdc.sdcio.dev targets.sdc.sdcio.dev; do
     65	  kubectl --context "${KIND_CONTEXT}" wait --for=condition=Established --timeout=180s crd/${crd} || true
     66	done
     67	
     68	# Wait for SDC component pods ready
     69	kubectl --context "${KIND_CONTEXT}" -n sdc-system wait --for=condition=Ready --timeout=300s pods -l app.kubernetes.io/name=sdc-schema || true
     70	kubectl --context "${KIND_CONTEXT}" -n sdc-system wait --for=condition=Ready --timeout=300s pods -l app.kubernetes.io/name=sdc-config || true
     71	kubectl --context "${KIND_CONTEXT}" -n sdc-system wait --for=condition=Ready --timeout=300s pods -l app.kubernetes.io/name=sdc-data || true
     72	kubectl --context "${KIND_CONTEXT}" -n sdc-system wait --for=condition=Ready --timeout=300s pods -l app.kubernetes.io/name=sdc-cache || true
     73	
     74	echo "[sdc-install] Pinned upstream CRDs and components applied; basic readiness achieved"
