     1	#!/usr/bin/env bash
     2	# Install pinned SDC CRDs/controllers into Kind and wait for readiness (Phase 3)
     3	# This script applies the ACTUAL upstream SDC CRDs pinned by versions.lock.yaml release.
     4	set -euo pipefail
     5	DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd)
     6	ROOT=$(cd -- "${DIR}/../.." && pwd)
     7	KIND_CONTEXT="kind-${AINETOPS_CLUSTER_NAME:-ainetops}"
     8	
     9	command -v kubectl >/dev/null 2>&1 || { echo "missing kubectl" >&2; exit 1; }
    10	command -v awk >/dev/null 2>&1 || { echo "missing awk" >&2; exit 1; }
    11	
    12	LOCK_FILE="${ROOT}/versions.lock.yaml"
    13	section() { awk -v s="^$1:$" 'f; $0~s{f=1} f && /^[^[:space:]]/{if(!p){p=1;print}else exit} f && p{print}' "${LOCK_FILE}"; }
    14	
    15	sdc_release=$(section sdc | awk '/release:/ {print $2; exit}')
    16	
    17	SDC_CRDS=(
    18	  "https://raw.githubusercontent.com/sdcio/sdc/${sdc_release}/deploy/crds/sdc.sdcio.dev_schemas.yaml"
    19	  "https://raw.githubusercontent.com/sdcio/sdc/${sdc_release}/deploy/crds/sdc.sdcio.dev_configs.yaml"
    20	  "https://raw.githubusercontent.com/sdcio/sdc/${sdc_release}/deploy/crds/sdc.sdcio.dev_targets.yaml"
    21	)
    22	
    23	# Apply CRDs from pinned upstream
    24	for u in "${SDC_CRDS[@]}"; do
    25	  kubectl --context "${KIND_CONTEXT}" apply -f "$u"
    26	done
    27	
    28	# Apply SDC components (Deployments/Services) and PVCs
    29	kubectl --context "${KIND_CONTEXT}" apply -f "${ROOT}/deploy/sdc/components.yaml"
    30	
    31	# PVCs for SDC state (ensure created)
    32	cat <<EOF | kubectl --context "${KIND_CONTEXT}" apply -f -
    33	apiVersion: v1
    34	kind: PersistentVolumeClaim
    35	metadata:
    36	  name: sdc-data
    37	  namespace: sdc-system
    38	spec:
    39	  accessModes: [ "ReadWriteOnce" ]
    40	  resources:
    41	    requests:
    42	      storage: 1Gi
    43	---
    44	apiVersion: v1
    45	kind: PersistentVolumeClaim
    46	metadata:
    47	  name: sdc-cache
    48	  namespace: sdc-system
    49	spec:
    50	  accessModes: [ "ReadWriteOnce" ]
    51	  resources:
    52	    requests:
    53	      storage: 1Gi
    54	EOF
    55	
    56	# Wait for CRDs (basic)
    57	for crd in schemas.sdc.sdcio.dev configs.sdc.sdcio.dev targets.sdc.sdcio.dev; do
    58	  kubectl --context "${KIND_CONTEXT}" wait --for=condition=Established --timeout=180s crd/${crd} || true
    59	done
    60	
    61	# Wait for SDC component pods ready
    62	kubectl --context "${KIND_CONTEXT}" -n sdc-system wait --for=condition=Ready --timeout=300s pods -l app.kubernetes.io/name=sdc-schema || true
    63	kubectl --context "${KIND_CONTEXT}" -n sdc-system wait --for=condition=Ready --timeout=300s pods -l app.kubernetes.io/name=sdc-config || true
    64	kubectl --context "${KIND_CONTEXT}" -n sdc-system wait --for=condition=Ready --timeout=300s pods -l app.kubernetes.io/name=sdc-data || true
    65	kubectl --context "${KIND_CONTEXT}" -n sdc-system wait --for=condition=Ready --timeout=300s pods -l app.kubernetes.io/name=sdc-cache || true
    66	
    67	echo "[sdc-install] Pinned upstream CRDs and components applied; basic readiness achieved"
