     1	#!/usr/bin/env bash
     2	# Supply-chain advisory checks for the fully open-source distribution (T074)
     3	# - Dependency licenses: best-effort collection (advisory)
     4	# - Vulnerabilities: run govulncheck when available (advisory)
     5	# - Image provenance: require pinned digests for platform images (enforced)
     6	# - SBOM: run syft when available to generate repository SBOM (advisory)
     7	# - Enforce SR Linux absence from dependency graph/runtime manifests per FR-020 (enforced)
     8	set -euo pipefail
     9	
    10	ROOT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/../.." && pwd)
    11	REPORT_DIR="${ROOT_DIR}/.wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs"
    12	mkdir -p "$REPORT_DIR"
    13	
    14	fail=0
    15	
    16	log() { echo "[supply-chain] $*" >&2; }
    17	
    18	# 1) Enforce SR Linux absence in go dependency graph and manifests (FR-020)
    19	log "Checking SR Linux absence in go.mod/go.sum, Dockerfiles, and Kubernetes manifests"
    20	SR_PAT='\bsr[ -]?linux\b|ghcr\.io/nokia/srlinux|nokia_srlinux'
    21	if command -v rg >/dev/null 2>&1; then
    22	  SR_MATCHES=$(rg -i -n --hidden \
    23	    -g '!**/.git/**' \
    24	    -g '!**/vendor/**' \
    25	    -e "$SR_PAT" \
    26	    go.mod go.sum cmd config deploy || true)
    27	else
    28	  # Fallback to grep when ripgrep is unavailable (CI/minimal envs)
    29	  SR_MATCHES=$(grep -RinE -I \
    30	    --exclude-dir=.git \
    31	    --exclude-dir=vendor \
    32	    -e "$SR_PAT" \
    33	    go.mod go.sum cmd config deploy 2>/dev/null || true)
    34	fi
    35	if [[ -n "$SR_MATCHES" ]]; then
    36	  echo "::error title=SR Linux mention(s) in dependency graph/manifests::${SR_MATCHES}" | tee "$REPORT_DIR/supply-chain.srlinux.matches.txt"
    37	  fail=1
    38	else
    39	  echo "No SR Linux artifacts detected in dependency graph/manifests" | tee "$REPORT_DIR/supply-chain.srlinux.ok.txt"
    40	fi
    41	
    42	# 2) Image provenance: require pinned digests for platform images in deploy/**.yaml
    43	log "Verifying platform images are pinned by immutable digests"
    44	if command -v rg >/dev/null 2>&1; then
    45	  IMG_LINES=$(rg -n '^\s*image:\s*[^#]+' deploy || true)
    46	else
    47	  IMG_LINES=$(grep -RinE '^\s*image:\s*[^#]+' deploy 2>/dev/null || true)
    48	fi
    49	MISSING_DIGEST=$(printf "%s" "$IMG_LINES" | awk -F: 'BEGIN{ok=1} {line=$0; if ($0 !~ /@sha256:[0-9a-f]{64}/) {print line; ok=0}} END{exit ok==1 ? 0 : 1}') || {
    50	  echo "::error title=Unpinned images found::The following image lines are not pinned by digest:" | tee "$REPORT_DIR/supply-chain.unpinned-images.txt"
    51	  printf "%s\n" "$IMG_LINES" | awk '$0 !~ /@sha256:[0-9a-f]{64}/' | tee -a "$REPORT_DIR/supply-chain.unpinned-images.txt"
    52	  fail=1
    53	}
    54	# Positive artifact when all images are pinned by digest
    55	if [[ -z "${MISSING_DIGEST:-}" ]]; then
    56	  printf "%s\n" "$IMG_LINES" | tee "$REPORT_DIR/supply-chain.images-pinned.ok.txt" >/dev/null
    57	fi
    58	
    59	# 3) Vulnerabilities: govulncheck (advisory)
    60	if command -v govulncheck >/dev/null 2>&1; then
    61	  log "Running govulncheck (advisory)"
    62	  (cd "$ROOT_DIR" && govulncheck ./... || true) | tee "$REPORT_DIR/supply-chain.govulncheck.txt"
    63	else
    64	  log "govulncheck not installed; skipping (advisory)"
    65	fi
    66	
    67	# 4) SBOM: syft (advisory)
    68	if command -v syft >/dev/null 2>&1; then
    69	  log "Generating SBOM with syft (advisory)"
    70	  syft packages dir:"$ROOT_DIR" -o json > "$REPORT_DIR/syft.sbom.json" || true
    71	else
    72	  log "syft not installed; skipping SBOM generation (advisory)"
    73	fi
    74	
    75	# 5) Licenses: go-licenses (advisory)
    76	if command -v go-licenses >/dev/null 2>&1; then
    77	  log "Collecting dependency licenses with go-licenses (advisory)"
    78	  (cd "$ROOT_DIR" && go-licenses report ./... || true) | tee "$REPORT_DIR/supply-chain.licenses.txt"
    79	else
    80	  log "go-licenses not installed; skipping license report (advisory)"
    81	fi
    82	
    83	if [[ "$fail" -ne 0 ]]; then
    84	  log "Supply-chain checks failed"
    85	  exit 1
    86	fi
    87	log "Supply-chain checks passed (enforced: SR Linux absence, image digests; advisory: others)"
