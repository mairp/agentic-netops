     2	package main
     3	
     4	import (
     5		"encoding/json"
     6		"flag"
     7		"fmt"
     8		"io"
     9		"os"
    10	
    11		"github.com/mairp/ainetops/pkg/migration"
    12	)
    13	
    14	// Deterministic CLI: reads a JSON array or object from stdin or --file, validates all-or-nothing,
    15	// then emits a single concatenated YAML stream of Kubenet Network resources in stable order.
    16	// No cluster interaction occurs in this phase.
    17	func main() {
    18		var file string
    19		flag.StringVar(&file, "file", "", "Path to normalized service input JSON (object or array); default stdin")
    20		flag.Parse()
    21	
    22		var data []byte
    23		var err error
    24		if file == "" {
    25			data, err = io.ReadAll(os.Stdin)
    26		} else {
    27			data, err = os.ReadFile(file)
    28		}
    29		if err != nil {
    30			fatal(fmt.Errorf("read input: %w", err))
    31		}
    32		// Strict parsing rejects unknown fields and raw CLI leakage.
    33		inputs, err := migration.ParseStrictBatch(data)
    34		if err != nil {
    35			// Emit structured JSON error (consistent shape) before any output.
    36			msg := migration.MarshalError(err)
    37			fmt.Fprintln(os.Stderr, msg)
    38			os.Exit(2)
    39		}
    40		if err := processBatch(inputs); err != nil { fatal(err) }
    41	}
    42	
    43	
    44	func processBatch(inputs []migration.ServiceInput) error {
    45		// Detect duplicate IDs within the batch for collision reporting.
    46		ids := map[string]int{}
    47		for i, in := range inputs {
    48			if _, ok := ids[in.ServiceID]; ok { ids[in.ServiceID] = -1 } else { ids[in.ServiceID] = i }
    49		}
    50		// Validate all-or-nothing; collect causes with stable ordering by index.
    51		var causes []string
    52		for i := range inputs {
    53			in := inputs[i]
    54			dup := false
    55			if idx, ok := ids[in.ServiceID]; ok && idx == -1 { dup = true }
    56			if err := in.ValidateAllOrNothing(i, dup); err != nil {
    57				if ve, ok := err.(*migration.ValidationError); ok { causes = append(causes, ve.Causes...) } else { causes = append(causes, err.Error()) }
    58			}
    59		}
    60		if len(causes) > 0 {
    61			// Emit structured JSON error to stderr and exit non-zero.
    62			enc := json.NewEncoder(os.Stderr)
    63			enc.SetIndent("", "  ")
    64			_ = enc.Encode(map[string]any{"error": "validation", "causes": causes})
    65			os.Exit(2)
    66		}
    67		// Translate in order; separator is "---\n"
    68		for i := range inputs {
    69			in := inputs[i]
    70			bundle, err := migration.Translate(&in)
    71			if err != nil { return err }
    72			if i > 0 { fmt.Println("---") }
    73			fmt.Println(bundle.NetworkYAML)
    74		}
    75		return nil
    76	}
    77	
    78	func fatal(err error) {
    79		fmt.Fprintf(os.Stderr, "migration-translator: %v\n", err)
    80		os.Exit(1)
    81	}
