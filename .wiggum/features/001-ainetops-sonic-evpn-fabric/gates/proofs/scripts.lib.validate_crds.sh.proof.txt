     1	#!/usr/bin/env bash
     2	# Validate Kubenet/KUID and SDC CRDs/examples with Kubernetes server-side dry-run
     3	# Uses pinned commits/releases recorded in versions.lock.yaml
     4	set -euo pipefail
     5	
     6	ROOT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/../.." && pwd)
     7	LOCK_FILE="${ROOT_DIR}/versions.lock.yaml"
     8	
     9	require() { command -v "$1" >/dev/null 2>&1 || { echo "missing command: $1" >&2; exit 1; }; }
    10	
    11	require kubectl
    12	require awk
    13	
    14	# Helper to extract a key's value block between a section and next top-level key
    15	section() { awk -v s="^$1:$" 'f; $0~s{f=1} f && /^[^[:space:]]/{if(!p){p=1;print}else exit} f && p{print}' "$LOCK_FILE"; }
    16	
    17	kubenet_commit=$(section kubenet | awk '/commit:/ {print $2; exit}')
    18	kubenet_api_shape=$(section kubenet | awk '/api_shape:/ {print $2; exit}')
    19	kuid_commit=$(section kuid | awk '/commit:/ {print $2; exit}')
    20	# Extract the sdc core (sdcio/sdc) release explicitly, not the nested config-server/schema-server
    21	sdc_core_release=$(awk '
    22	  $1=="sdc:" {sec=1; next}
    23	  sec && /^[^[:space:]]/ {exit}
    24	  sec && $1=="core:" {in_core=1; next}
    25	  in_core && /^[^[:space:]]/ {exit}
    26	  in_core && $1=="release:" {print $2; exit}
    27	' "$LOCK_FILE")
    28	
    29	[[ $kubenet_commit =~ ^[0-9a-f]{40}$ ]] || { echo "invalid kubenet commit in versions.lock.yaml" >&2; exit 1; }
    30	[[ $kuid_commit =~ ^[0-9a-f]{40}$ ]] || { echo "invalid kuid commit in versions.lock.yaml" >&2; exit 1; }
    31	[[ $sdc_core_release =~ ^v[0-9]+\.[0-9]+\.[0-9]+$ ]] || { echo "invalid sdc core release in versions.lock.yaml" >&2; exit 1; }
    32	
    33	# Validate against pinned local copies committed in deploy/*, derived from the pinned upstream commits/releases.
    34	# This avoids network flakiness while preserving immutability and API shape.
    35	KUBENET_CRDS=()
    36	case "$kubenet_api_shape" in
    37	  NetworkConfig)
    38	    KUBENET_CRDS=(
    39	      "${ROOT_DIR}/deploy/kubenet/crds/kubenet-crds.yaml"
    40	    )
    41	    ;;
    42	  NetworkDesign)
    43	    KUBENET_CRDS=(
    44	      "${ROOT_DIR}/deploy/kubenet/crds/kubenet-crds.yaml"
    45	    )
    46	    ;;
    47	  *)
    48	    KUBENET_CRDS=(
    49	      "${ROOT_DIR}/deploy/kubenet/crds/kubenet-crds.yaml"
    50	    )
    51	    ;;
    52	 esac
    53	KUBENET_EXAMPLES=(
    54	  "${ROOT_DIR}/deploy/kubenet/topology.yaml"
    55	)
    56	KUID_CRDS=(
    57	  "${ROOT_DIR}/deploy/kuid/crds/kuid-crds.yaml"
    58	)
    59	SDC_CRDS=(
    60	  "${ROOT_DIR}/deploy/sdc/crds/sdc-crds.yaml"
    61	)
    62	
    63	run_dry_run() {
    64	  local what=$1; shift
    65	  local -a files=("$@")
    66	  echo "[validate-crds] server-side dry-run: ${what} (${#files[@]} files)" >&2
    67	  # Build multiple -f flags so kubectl handles each manifest separately
    68	  local -a args=(apply --dry-run=server)
    69	  for f in "${files[@]}"; do args+=( -f "$f" ); done
    70	  set -x
    71	  kubectl "${args[@]}"
    72	  { set +x; } 2>/dev/null
    73	}
    74	
    75	# Note: The actual cluster must be reachable for server-side validation.
    76	# This script is idempotent and performs validation only.
    77	
    78	# Validate CRDs
    79	run_dry_run "Kubenet CRDs" "${KUBENET_CRDS[@]}"
    80	run_dry_run "KUID CRDs" "${KUID_CRDS[@]}"
    81	run_dry_run "SDC CRDs" "${SDC_CRDS[@]}"
    82	
    83	# Validate example manifests against CRDs
    84	run_dry_run "Kubenet examples" "${KUBENET_EXAMPLES[@]}"
    85	
    86	echo "[validate-crds] OK"
