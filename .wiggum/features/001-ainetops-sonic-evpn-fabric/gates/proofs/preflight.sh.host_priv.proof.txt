22\tpreflight::host_resources() {
23\t  # Minimal CPU/RAM/disk checks; allow override via env for CI
24\t  local min_cpu=${AINETOPS_MIN_CPU:-4}
25\t  local min_mem_mb=${AINETOPS_MIN_MEM_MB:-8192}
26\t  local min_disk_mb=${AINETOPS_MIN_DISK_MB:-20480}
27\t  # Best-effort checks; platform-specific commands may vary
28\t  local cores mem_kb avail_kb
29\t  cores=$(getconf _NPROCESSORS_ONLN || echo 1)
30\t  mem_kb=$(awk '/MemTotal/ {print $2}' /proc/meminfo 2>/dev/null || echo 0)
31\t  avail_kb=$(df -Pm . 2>/dev/null | awk 'NR==2{print $4}' || echo 0)
32\t  (( cores >= min_cpu )) || preflight::die "CPU cores $cores < required $min_cpu"
33\t  (( mem_kb/1024 >= min_mem_mb )) || preflight::die "Memory $((${mem_kb}/1024))MB < required ${min_mem_mb}MB"
34\t  (( avail_kb >= min_disk_mb )) || preflight::die "Disk ${avail_kb}MB free < required ${min_disk_mb}MB"
35\t}
