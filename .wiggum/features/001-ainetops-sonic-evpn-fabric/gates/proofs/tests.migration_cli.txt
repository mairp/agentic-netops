    14	// Note: Some environments do not build the CLI during unit tests; we keep a smoke test
    15	// that simply checks the binary path exists when built. Core behavior is validated via the library.
    16	func TestMigrationCLI_StructuredUnknownField(t *testing.T) {
    17		bad := `{"serviceId":"svcX","type":"VPLS","tenant":"A","rdRt":{"rd":"65000:1","importRT":["65000:1"],"exportRT":["65000:1"]},"l2vni":10001,"endpoints":[{"node":"leaf01","attachment":"c1","vlan":10},{"node":"leaf02","attachment":"c2","vlan":10}],"unknown":"field"}`
    18		_, err := migration.ParseStrictBatch([]byte(bad))
    19		if err == nil { t.Fatalf("expected parse error due to unknown field") }
    20		msg := migration.MarshalError(err)
    21		// Ensure structured JSON with causes and the unknown field message.
    22		var m map[string]any
    23		if json.Unmarshal([]byte(msg), &m) != nil { t.Fatalf("not JSON: %s", msg) }
    24		if m["error"] != "validation" { t.Fatalf("missing error key: %v", m) }
    25		causes, ok := m["causes"].([]any)
    26		if !ok || len(causes) == 0 { t.Fatalf("missing causes: %v", m) }
    27		joined := msg
    28		if !strings.Contains(joined, "unknown field") { t.Fatalf("expected unknown field in causes: %s", msg) }
    29	}

    31	func TestMigrationCLI_ExecutesAndRejectsUnknown(t *testing.T) {
    32		// Try to run the CLI with go run so the test does not depend on a prebuilt binary.
    33		bad := `{"serviceId":"svcX","type":"VPLS","tenant":"A","rdRt":{"rd":"65000:1","importRT":["65000:1"],"exportRT":["65000:1"]},"l2vni":10001,"endpoints":[{"node":"leaf01","attachment":"c1","vlan":10},{"node":"leaf02","attachment":"c2","vlan":10}],"unknown":"field"}`
    34		cmd := exec.Command("go", "run", "./cmd/migration-translator")
    35		cmd.Dir = filepath.Join("..", "..")
    36		cmd.Stdin = bytes.NewBufferString(bad)
    37		out, err := cmd.CombinedOutput()
    38		if err == nil { t.Fatalf("expected non-zero exit; output=%s", string(out)) }
    39		// Expect structured JSON on stderr; CombinedOutput merges streams; check JSON presence and absence of YAML spec.
    40		if !strings.Contains(string(out), "\"error\": \"validation\"") { t.Fatalf("missing structured error: %s", string(out)) }
    41		if strings.Contains(string(out), "spec:") { t.Fatalf("unexpected YAML output on parse failure: %s", string(out)) }
    42	}
