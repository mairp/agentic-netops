     1	#!/usr/bin/env bash
     2	# T047a [US3] MTU and ECMP tests: verify maximum effective MTU accommodates VXLAN overhead and ECMP hashing
     3	set -euo pipefail
     4	
     5	CLAB_PREFIX=${CLAB_PREFIX:-clab-ainetops-fabric-}
     6	SRC=${SRC:-${CLAB_PREFIX}client01}
     7	DST_IP=${DST_IP:-192.0.2.21}
     8	
     9	GNMIC_BIN=${GNMIC_BIN:-gnmic}
    10	GNMI_USER=${GNMI_USER:-admin}
    11	GNMI_PASS=${GNMI_PASS:-admin}
    12	GNMI_CACERT=${GNMI_CACERT:-./secrets/ca.crt}
    13	GNMI_CERT=${GNMI_CERT:-./secrets/gnmi.crt}
    14	GNMI_KEY=${GNMI_KEY:-./secrets/gnmi.key}
    15	GNMI_ENCODING=${GNMI_ENCODING:-JSON_IETF}
    16	LEAF1=${LEAF1:-172.31.0.21:8080}
    17	
    18	_args_common=(--timeout 10s --username "$GNMI_USER" --password "$GNMI_PASS" --tls --skip-verify --encoding "$GNMI_ENCODING" --cacert "$GNMI_CACERT" --cert "$GNMI_CERT" --key "$GNMI_KEY")
    19	
    20	# Verify effective MTU across the overlay accommodates VXLAN overhead
    21	vxlan_mtu_test() {
    22	  echo "[mtu-ecmp] maximum effective MTU accommodates VXLAN overhead"
    23	  # attempt a near-jumbo payload; 8900 bytes payload should pass with 9216 underlay MTU
    24	  docker exec "$SRC" ping -c 1 -W 2 -M do -s 8900 "$DST_IP"
    25	}
    26	
    27	_read_counter() {
    28	  local ifname=$1; shift
    29	  "$GNMIC_BIN" --address "$LEAF1" "${_args_common[@]}" get --path \
    30	    "/openconfig-interfaces:interfaces/interface[name=${ifname}]/state/counters/out-octets" -o json \
    31	    | jq -r '..|objects|select(has("out-octets"))|."out-octets"' | head -n1
    32	}
    33	
    34	# ECMP hashing verification (distribution across spines)
    35	ecmp_hashing_test() {
    36	  echo "[mtu-ecmp] ECMP hashing where qualified"
    37	  local IF_A=${IF_A:-Ethernet1} # uplink to spine01
    38	  local IF_B=${IF_B:-Ethernet2} # uplink to spine02
    39	  local pre_a pre_b post_a post_b
    40	  pre_a=$(_read_counter "$IF_A"); pre_b=$(_read_counter "$IF_B")
    41	  # Send a burst of UDP flows with varying src ports to exercise ECMP hashing (overlay)
    42	  for p in 10000 10001 10002 11000 11001 11002 12000 12001 12002; do
    43	    docker exec "$SRC" bash -lc "timeout 0.2 bash -c '>/dev/udp/${DST_IP}/$p' || true"
    44	  done
    45	  sleep 1
    46	  post_a=$(_read_counter "$IF_A"); post_b=$(_read_counter "$IF_B")
    47	  echo "[mtu-ecmp] ${IF_A} out-octets: $pre_a -> $post_a"
    48	  echo "[mtu-ecmp] ${IF_B} out-octets: $pre_b -> $post_b"
    49	  # Assert both interfaces saw traffic increments
    50	  if [[ -z "$pre_a" || -z "$post_a" || -z "$pre_b" || -z "$post_b" ]]; then
    51	    echo "[mtu-ecmp] ERROR: missing interface counters" >&2; exit 1
    52	  fi
    53	  if (( post_a <= pre_a )); then echo "[mtu-ecmp] ERROR: no increment on $IF_A" >&2; exit 1; fi
    54	  if (( post_b <= pre_b )); then echo "[mtu-ecmp] ERROR: no increment on $IF_B" >&2; exit 1; fi
    55	}
    56	
    57	case "${1:-run}" in
    58	  run)
    59	    vxlan_mtu_test
    60	    ecmp_hashing_test
    61	    ;;
    62	  *) echo "usage: $0 run" >&2; exit 2 ;;
    63	esac
