     1	#!/usr/bin/env bash
     2	# Run the Phase 8 verification test suites and capture logs under gates/proofs.
     3	# T079 contract: this runner is STRICT. Every suite records its real exit code;
     4	# the script prints a PASS/FAIL/SKIP summary and exits non-zero if any suite
     5	# failed. Suites that legitimately require a live lab report their skip reason
     6	# explicitly, so a green summary always means every listed suite actually ran
     7	# or was skipped with a printed, greppable reason.
     8	set -u
     9	
    10	ROOT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/../.." && pwd)
    11	PROOFS_DIR="$ROOT_DIR/.wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs"
    12	mkdir -p "$PROOFS_DIR"
    13	SUMMARY="$PROOFS_DIR/tests.summary.txt"
    14	: > "$SUMMARY"
    15	
    16	# envtest control-plane binaries (real assets required — a skipped envtest is
    17	# a failure, not a pass).
    18	if [[ -z "${KUBEBUILDER_ASSETS:-}" ]]; then
    19	  for d in /root/.local/share/kubebuilder-envtest/*-linux-amd64 /usr/local/kubebuilder/bin; do
    20	    if [[ -x "$d/kube-apiserver" ]]; then export KUBEBUILDER_ASSETS="$d"; break; fi
    21	  done
    22	fi
    23	
    24	record(){ printf '%s %s\n' "$2" "$1" >>"$SUMMARY"; }
    25	
    26	suite(){
    27	  local name=$1 out=$2; shift 2
    28	  echo "[run] $name: $*" >"$out"
    29	  set +e
    30	  ( "$@" ) >>"$out" 2>&1
    31	  local rc=$?
    32	  set -e
    33	  if [[ $rc -eq 0 ]]; then
    34	    record "$name" PASS
    35	  elif grep -q 'SKIP-LIVE' "$out"; then
    36	    record "$name" "SKIP-LIVE($(grep -m1 -o 'SKIP-LIVE[^"]*' "$out" | head -c 60))"
    37	  else
    38	    record "$name" "FAIL(rc=$rc)"
    39	  fi
    40	}
    41	
    42	go_bin=""
    43	command -v go >/dev/null 2>&1 && go_bin=go
    44	
    45	# 1) API suite — CRD/envtest sanity (requires real envtest assets)
    46	if [[ -n "$go_bin" ]]; then
    47	  suite api "$PROOFS_DIR/tests.api.log" env KUBEBUILDER_ASSETS="${KUBEBUILDER_ASSETS:-}" \
    48	    "$go_bin" test ./tests/envtest -v
    49	else
    50	  echo "FAIL: no go toolchain" >"$PROOFS_DIR/tests.api.log"; record api "FAIL(no-go)"
    51	fi
    52	
    53	# 2) Unit suite — all unit tests
    54	[[ -n "$go_bin" ]] && suite unit "$PROOFS_DIR/tests.unit.log" "$go_bin" test ./tests/unit -v
    55	
    56	# 3) Golden suite — table/golden tests only
    57	[[ -n "$go_bin" ]] && suite golden "$PROOFS_DIR/tests.golden.log" "$go_bin" test ./tests/unit -run Golden -v
    58	
    59	# 4) SDC validation suite — offline validators + path-register guards
    60	[[ -n "$go_bin" ]] && suite sdc-validation "$PROOFS_DIR/tests.sdc-validation.log" \
