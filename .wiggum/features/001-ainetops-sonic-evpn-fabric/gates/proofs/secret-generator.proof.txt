     1	# T022: In-cluster generator Job to populate lab Secrets without storing credentials in Git
     2	apiVersion: batch/v1
     3	kind: Job
     4	metadata:
     5	  name: ainetops-secret-generator
     6	  namespace: ainetops-system
     7	spec:
     8	  template:
     9	    spec:
    10	      restartPolicy: OnFailure
    11	      serviceAccountName: ainetops-controller
    12	      containers:
    13	      - name: generator
    14	        image: ghcr.io/ainetops/secret-generator@sha256:3333333333333333333333333333333333333333333333333333333333333333
    15	        command: ["/bin/sh","-c"]
    16	        args:
    17	        - |
    18	          set -euo pipefail
    19	          # Create random admin/admin pair for lab use only; replace with stronger in real lab
    20	          USER=$(cat /proc/sys/kernel/random/uuid | cut -c1-8)
    21	          PASS=$(cat /proc/sys/kernel/random/uuid | cut -c1-12)
    22	          kubectl -n ainetops-system create secret generic gnmi-lab-creds --dry-run=client \
    23	            --from-literal=username=$USER --from-literal=password=$PASS -o yaml | kubectl apply -f -
    24	          # Generate ephemeral self-signed TLS via openssl
    25	          openssl req -x509 -nodes -newkey rsa:2048 -keyout /tmp/tls.key -out /tmp/tls.crt -subj "/CN=ainetops" -days 1
    26	          cp /tmp/tls.crt /tmp/ca.crt
    27	          kubectl -n ainetops-system create secret generic gnmi-lab-tls --dry-run=client \
    28	            --from-file=ca.crt=/tmp/ca.crt --from-file=tls.crt=/tmp/tls.crt --from-file=tls.key=/tmp/tls.key -o yaml | kubectl apply -f -
    29	          echo "generated"
