1	package sonicprovider
2	
3	import (
4		"context"
5	
6		"github.com/mairp/ainetops/pkg/kubenet"
7		"sigs.k8s.io/controller-runtime/pkg/client"
8		ctrl "sigs.k8s.io/controller-runtime"

11	// SetupIndexes declares controller-runtime field indexes used by watches and dependency lookups.
12	func SetupIndexes(ctx context.Context, mgr ctrl.Manager) error {
13		idx := mgr.GetFieldIndexer()
14		// Index NetworkDevices by the 'network.kubenet.dev/derived' label presence
15		return idx.IndexField(ctx, &kubenet.NetworkDevice{}, "metadata.labels.network.kubenet.dev/derived", func(o client.Object) []string {
16			if o == nil {
17				return nil
18			}
19			labels := o.GetLabels()
20			if labels == nil {
21				return nil
22			}
23			if v, ok := labels["network.kubenet.dev/derived"]; ok && v == "true" {
24				return []string{"true"}
25			}
26			return nil
27		})
28	}
