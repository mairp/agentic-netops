    14		"sigs.k8s.io/controller-runtime/pkg/client/fake"
    15

    18		"github.com/mairp/ainetops/pkg/sdc"
    19	)
    20	
    21	// TestProvider_SDCStatusPropagation verifies that SDC Config status drives NetworkDevice conditions.
    22	func TestProvider_SDCStatusPropagation(t *testing.T) {
    23		scheme := runtime.NewScheme()
    24		_ = clientgoscheme.AddToScheme(scheme)
    25		_ = kubenet.AddToScheme(scheme)
    26		_ = sdc.AddToScheme(scheme)
    27	
    28		ctx := context.TODO()
    29		nd := &kubenet.NetworkDevice{TypeMeta: metav1.TypeMeta{APIVersion: kubenet.GroupVersion.String(), Kind: "NetworkDevice"}, ObjectMeta: metav1.ObjectMeta{Name: "leaf01", Namespace: "default"}}
    30		// Case 1: Deviation present => Degraded=True
    31		cfg := &sdc.Config{TypeMeta: metav1.TypeMeta{APIVersion: sdc.GroupVersion.String(), Kind: "Config"}, ObjectMeta: metav1.ObjectMeta{Name: "nd-leaf01", Namespace: "default"}, Status: sdc.ConfigStatus{Deviation: []sdc.DeviationRecord{{Path: "/if", Message: "drift"}}}}
    32	
    33		// Pre-seed objects to avoid any scheme/restmapper surprises with Create+Get roundtrip
    34		c := fake.NewClientBuilder().WithScheme(scheme).WithStatusSubresource(&kubenet.NetworkDevice{}, &sdc.Config{}).WithObjects(nd, cfg).Build()
    35		rec := &sonicprovider.Reconciler{Client: c, Scheme: scheme}
    36	
    37		// sanity: object is retrievable from fake client prior to reconcile
    38		if err := c.Get(ctx, types.NamespacedName{Namespace: nd.Namespace, Name: nd.Name}, &kubenet.NetworkDevice{}); err != nil { t.Fatalf("pre-get nd: %v", err) }
    39		// and listed
    40		ndl := &kubenet.NetworkDeviceList{}
    41		if err := c.List(ctx, ndl); err != nil { t.Fatalf("pre-list: %v", err) }
    42		if len(ndl.Items) == 0 { t.Fatalf("pre-list: expected at least one item") }
    43	
    44		if _, err := rec.Reconcile(ctx, ctrl.Request{NamespacedName: types.NamespacedName{Namespace: nd.Namespace, Name: nd.Name}}); err != nil { t.Logf("err type: %T", err); t.Fatalf("reconcile: %v", err) }
    45		if err := c.Get(ctx, types.NamespacedName{Namespace: nd.Namespace, Name: nd.Name}, nd); err != nil { t.Fatalf("get nd: %v", err) }
    46		condsAny := nd.Status["conditions"]
    47		if condsAny == nil { t.Fatalf("expected conditions on NetworkDevice status") }
    48		// best-effort type assertion
    49		if conds, ok := condsAny.([]metav1.Condition); ok {
    50			found := false
    51			for _, cnd := range conds { if cnd.Type == "Degraded" && cnd.Status == metav1.ConditionTrue { found = true; break } }
    52			if !found { t.Fatalf("expected Degraded=True condition from SDC deviation") }
    53		}
    54	
    55		// Case 2: Ready => Ready=True
    56		cfg.Status.Deviation = nil
    57		cfg.Status.Ready = true
    58		if err := c.Update(ctx, cfg); err != nil { t.Fatalf("update cfg: %v", err) }
    59		if _, err := rec.Reconcile(ctx, ctrl.Request{NamespacedName: types.NamespacedName{Namespace: nd.Namespace, Name: nd.Name}}); err != nil { t.Fatalf("reconcile2: %v", err) }
    60		if err := c.Get(ctx, types.NamespacedName{Namespace: nd.Namespace, Name: nd.Name}, nd); err != nil { t.Fatalf("get nd2: %v", err) }
    61		condsAny = nd.Status["conditions"]
    62		if conds, ok := condsAny.([]metav1.Condition); ok {
    63			found := false
    64			for _, cnd := range conds { if cnd.Type == "Ready" && cnd.Status == metav1.ConditionTrue { found = true; break } }
    65			if !found { t.Fatalf("expected Ready=True condition from SDC Ready") }
    66		}
    67	}
