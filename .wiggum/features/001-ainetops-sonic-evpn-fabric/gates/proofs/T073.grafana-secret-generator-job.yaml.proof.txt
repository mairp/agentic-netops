     1	# T073: In-cluster generator Job to create Grafana admin Secret with random credentials
     2	apiVersion: batch/v1
     3	kind: Job
     4	metadata:
     5	  name: grafana-admin-secret-generator
     6	  namespace: monitoring
     7	spec:
     8	  template:
     9	    spec:
    10	      restartPolicy: OnFailure
    11	      serviceAccountName: grafana-secret-generator
    12	      containers:
    13	      - name: generator
    14	        image: ghcr.io/ainetops/secret-generator@sha256:3333333333333333333333333333333333333333333333333333333333333333
    15	        command: ["/bin/sh","-c"]
    16	        args:
    17	        - |
    18	          set -euo pipefail
    19	          # Generate non-default random admin credentials and store only in Secret
    20	          USER=$(cat /proc/sys/kernel/random/uuid | cut -c1-8)
    21	          PASS=$(cat /proc/sys/kernel/random/uuid | cut -c1-16)
    22	          kubectl -n monitoring create secret generic grafana-admin --dry-run=client \
    23	            --from-literal=admin-user=$USER --from-literal=admin-password=$PASS -o yaml | kubectl apply -f -
    24	          echo "grafana admin secret generated"
