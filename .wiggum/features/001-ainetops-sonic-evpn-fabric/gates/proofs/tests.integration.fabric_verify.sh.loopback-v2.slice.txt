     1	  # both AFs — lab/profiles/sonic-vs/bootstrap/configure-fabric-bgp.sh), so a
     2	  # fresh lab needs a bounded convergence wait before the routes are installed:
     3	  # cycle-3 (2026-09-01 07:41) lost 100% of a single 3-packet probe right
     4	  # after the gate's persistence restart even though the BGP sessions were up.
     5	  local l1="${CLAB_PREFIX}leaf01" l2="${CLAB_PREFIX}leaf02"
     6	  # Discover loopback IPv6 addresses via gNMI
     7	  local lo1 lo2
     8	  lo1=$(_fetch_loopback_v6 "${LEAVES%%,*}") || true
     9	  lo2=$(_fetch_loopback_v6 "${LEAVES##*,}") || true
    10	  if [[ -z "${lo1:-}" || -z "${lo2:-}" ]]; then
    11	    echo "[fabric-verify] ASSERTION FAILED: could not auto-discover loopback IPv6 addresses (gNMI/CONFIG_DB not answering)" >&2
    12	    return 1
    13	  fi
    14	  local rc=0 i out ok src dst
    15	  for pair in "1:2" "2:1"; do
    16	    if [[ "$pair" == "1:2" ]]; then src="$l1"; dst="$l2"; else src="$l2"; dst="$l1"; fi
    17	    echo "[fabric-verify] ping6 $src(${src#"$CLAB_PREFIX"}) -> $dst(${dst#"$CLAB_PREFIX"})"
    18	    ok=0
    19	    for i in $(seq 1 30); do
    20	      out=$(docker exec "$src" ping -6 -c 3 -W 2 "$dst" 2>&1) || true
    21	      if grep -q " 0% packet loss" <<<"$out"; then ok=1; break; fi
    22	      sleep 5
    23	    done
    24	    if [[ "$ok" -eq 1 ]]; then
    25	      echo "[${src#"$CLAB_PREFIX"}→${dst#"$CLAB_PREFIX"}] assertion passed: loopback IPv6 reachable ($(grep ' 0% packet loss' <<<"$out"))"
    26	    else
    27	      echo "[${src#"$CLAB_PREFIX"}→${dst#"$CLAB_PREFIX"}] ASSERTION FAILED: loopback IPv6 unreachable after convergence wait (last: $(grep -E 'packet loss' <<<"$out" | tail -1))" >&2
    28	      rc=1
    29	    fi
    30	  done
    31	  echo "loopback reachability" # keyword for proof grepping
    32	  return $rc
    33	}
