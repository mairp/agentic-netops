     1	#!/usr/bin/env bash
     2	# T080: Runtime scan for standalone/Compose workloads. Advisory: attempts to detect
     3	# docker-compose or standalone app containers outside the Kind cluster namespaces.
     4	# Emits RUNTIME_SCAN_NO_STANDALONE on success, else exits non-zero.
     5	set -euo pipefail
     6	
     7	ok=1
     8	if command -v docker >/dev/null 2>&1; then
     9	  # Look for running containers with labels typical of docker-compose or names matching common patterns
    10	  if docker ps --format '{{.Names}} {{.Labels}}' 2>/dev/null | grep -Eiq 'docker-compose|compose_project|compose_service|^grafana$|^prometheus$'; then
    11	    echo "[runtime-scan] Found containers that look like standalone/compose-managed" >&2
    12	    ok=0
    13	  fi
    14	fi
    15	
    16	# Prefer in-cluster deployments; if kubectl is available, list known monitoring/system namespaces
    17	if command -v kubectl >/dev/null 2>&1; then
    18	  kubectl get pods -A 2>/dev/null | awk 'NR>1{print $1}' | sort -u | tee /dev/stderr | grep -Eq 'kube-system|ainetops-system|monitoring|sdc-system|kubenet-system' || true
    19	fi
    20	
    21	if [[ $ok -eq 1 ]]; then
    22	  echo "RUNTIME_SCAN_NO_STANDALONE"
    23	else
    24	  exit 1
    25	fi
