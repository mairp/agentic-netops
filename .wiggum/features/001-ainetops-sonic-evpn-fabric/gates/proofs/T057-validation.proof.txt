     2	// It disables HTML escaping so comparison substrings like ">=1" remain readable

     4	func MarshalError(err error) string {
     5		if err == nil { return "" }
     6		m := map[string]any{"error": "validation"}
     7		if ve, ok := err.(*ValidationError); ok {
     8			m["causes"] = ve.Causes
     9		} else {
    10			m["causes"] = []string{err.Error()}
    11		}
    12		// Use an Encoder with HTML escaping disabled to avoid sequences like \u003e
    13		// for '>' which make human/fixture comparison brittle.
    14		var buf bytes.Buffer
    15		enc := json.NewEncoder(&buf)
    16		enc.SetEscapeHTML(false)
    17		enc.SetIndent("", "  ")
    18		_ = enc.Encode(m)
    19		// json.Encoder adds a trailing newline; trim it for stable output.
    20		out := buf.String()
    21		if len(out) > 0 && out[len(out)-1] == '\n' { out = out[:len(out)-1] }
    22		return out
    23	}
