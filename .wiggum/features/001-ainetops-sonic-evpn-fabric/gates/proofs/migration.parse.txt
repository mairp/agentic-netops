     4	import (
     5		"bytes"
     6		"encoding/json"
     7		"fmt"
     8	)
     9	
    10	// ParseStrictBatch decodes a JSON object or array of ServiceInput with
    11	// DisallowUnknownFields to ensure unknown properties are rejected.
    12	func ParseStrictBatch(data []byte) ([]ServiceInput, error) {
    13		trim := bytes.TrimSpace(data)
    14		if len(trim) == 0 {
    15			return nil, fmt.Errorf("empty input")
    16		}
    17		switch trim[0] {
    18		case '[':
    19			var raws []json.RawMessage
    20			if err := json.Unmarshal(trim, &raws); err != nil {
    21				return nil, err
    22			}
    23			out := make([]ServiceInput, 0, len(raws))
    24			for i := range raws {
    25				var in ServiceInput
    26				if err := strictUnmarshal(raws[i], &in); err != nil {
    27					return nil, fmt.Errorf("item %d: %w", i, err)
    28				}
    29				out = append(out, in)
    30			}
    31			return out, nil
    32		case '{':
    33			var in ServiceInput
    34			if err := strictUnmarshal(trim, &in); err != nil {
    35				return nil, err
    36			}
    37			return []ServiceInput{in}, nil
    38		default:
    39			return nil, fmt.Errorf("invalid JSON: expected object or array")
    40		}
    41	}
    42	
    43	func strictUnmarshal(b []byte, v any) error {
    44		dec := json.NewDecoder(bytes.NewReader(b))
    45		dec.DisallowUnknownFields()
    46		if err := dec.Decode(v); err != nil {
    47			return err
    48		}
    49		return nil
    50	}
