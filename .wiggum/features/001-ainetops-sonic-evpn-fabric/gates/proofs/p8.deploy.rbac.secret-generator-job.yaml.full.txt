     1	# T022: In-cluster generator Job to populate lab Secrets without storing credentials in Git
     2	apiVersion: batch/v1
     3	kind: Job
     4	metadata:
     5	  name: ainetops-secret-generator
     6	  namespace: ainetops-system
     7	spec:
     8	  template:
     9	    spec:
    10	      restartPolicy: OnFailure
    11	      serviceAccountName: ainetops-controller
    12	      containers:
    13	      - name: generator
    14	        # Pinned Kind node image (kind.node_image in versions.lock.yaml). It carries
    15	        # kubectl + openssl + curl, so the generator needs no separate image and no
    16	        # external registry pull (reproducible on an air-gapped qualified host).
    17	        image: kindest/node@sha256:3abb816a5b1061fb15c6e9e60856ec40d56b7b52bcea5f5f1350bc6e2320b6f8
    18	        command: ["/bin/bash","-c"]
    19	        args:
    20	        - |
    21	          set -euo pipefail
    22	          # Create random admin/admin pair for lab use only; replace with stronger in real lab
    23	          USER=$(cat /proc/sys/kernel/random/uuid | cut -c1-8)
    24	          PASS=$(cat /proc/sys/kernel/random/uuid | cut -c1-12)
    25	          kubectl -n ainetops-system create secret generic gnmi-lab-creds --dry-run=client \
    26	            --from-literal=username=$USER --from-literal=password=$PASS -o yaml | kubectl apply -f -
    27	          # Generate ephemeral self-signed TLS via openssl.
    28	          # SANs MUST cover the deterministic lab mgmt IPs (lab/topology.clab.yml:
    29	          # spine01 .11, spine02 .12, leaf01 .21, leaf02 .22) plus localhost so
    30	          # gnmic clients can verify the server certificate by the IP they dial.
    31	          openssl req -x509 -nodes -newkey rsa:2048 -keyout /tmp/tls.key -out /tmp/tls.crt \
    32	            -subj "/CN=ainetops" -days 30 \
    33	            -addext "subjectAltName=DNS:localhost,DNS:ainetops,IP:127.0.0.1,IP:172.31.0.11,IP:172.31.0.12,IP:172.31.0.21,IP:172.31.0.22"
    34	          cp /tmp/tls.crt /tmp/ca.crt
    35	          kubectl -n ainetops-system create secret generic gnmi-lab-tls --dry-run=client \
    36	            --from-file=ca.crt=/tmp/ca.crt --from-file=tls.crt=/tmp/tls.crt --from-file=tls.key=/tmp/tls.key -o yaml | kubectl apply -f -
    37	          echo "generated"
