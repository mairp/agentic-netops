106: // CanonicalHash returns a deterministic SHA256 over the normalized input.
107: func (in *ServiceInput) CanonicalHash() (string, error) {
108: 	if in == nil {
109: 		return "", errors.New("nil input")
110: 	}
111: 	// Marshal the typed struct (no maps) to stable JSON and hash it.
112: 	b, err := json.Marshal(in)
113: 	if err != nil {
114: 		return "", err
115: 	}
116: 	sum := sha256.Sum256(b)
117: 	return hex.EncodeToString(sum[:]), nil
118: }
