     2	package render
     3	
     4	import "github.com/mairp/ainetops/pkg/model"
     5	
     6	// RenderSRv6 renders SRv6 global and MySID behaviors using SONiC-native paths for the pinned image.
     7	func RenderSRv6(locator model.SRv6Locator, mysid []model.MySID) map[string]any {
     8		res := map[string]any{}
     9		res["/sonic-srv6:sonic-srv6/SRV6_GLOBAL"] = map[string]any{"locator-prefix": locator.Prefix}
    10		// MySID entries simplified; real path would be sonic-srv6 tables
    11		list := make([]map[string]any, 0, len(mysid))
    12		for _, m := range mysid {
    13			entry := map[string]any{"sid": m.SID, "behavior": m.Behavior}
    14			if m.VRF != "" { entry["vrf"] = m.VRF }
    15			list = append(list, entry)
    16		}
    17		res["/sonic-srv6:sonic-srv6/MYSID"] = list
    18		return res
    19	}
