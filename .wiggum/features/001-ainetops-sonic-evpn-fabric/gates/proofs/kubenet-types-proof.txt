     1	// SPDX-License-Identifier: Apache-2.0
     2	// Package kubenet provides minimal pinned upstream Kubenet types required for watches/indexes.
     3	// This avoids importing the full upstream module while allowing controller-runtime to
     4	// watch NetworkDevice resources and reconcile against them.
     5	package kubenet
     6	
     7	import (
     8		metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
     9		"k8s.io/apimachinery/pkg/runtime"
    10		"k8s.io/apimachinery/pkg/runtime/schema"
    11		"sigs.k8s.io/controller-runtime/pkg/scheme"
    12	)
    13	
    14	// GroupVersion for pinned Kubenet APIs (see deploy/kubenet/crds/kubenet-crds.yaml).
    15	var GroupVersion = schema.GroupVersion{Group: "network.kubenet.dev", Version: "v1alpha1"}
    16	
    17	// NetworkDevice is a minimal representation of Kubenet's derived per-device intent.
    18	// We preserve unknown fields and do not attempt to model the full schema here.
    19	//
    20	// +k8s:deepcopy-gen=false
    21	// +kubebuilder:object:root=true
    22	// +kubebuilder:subresource:status
    23	// +kubebuilder:resource:scope=Namespaced
    24	// +kubebuilder:printcolumn:name="Ready",type=string,JSONPath=`.status.conditions[?(@.type=="Ready")].status`
    25	// +kubebuilder:validation:Optional
    26	// +kubebuilder:validation:Type=object
    27	// +kubebuilder:validation:PreserveUnknownFields=true
    28	// +kubebuilder:validation:MaxProperties=0
    29	// +kubebuilder:validation:MinProperties=0
    30	// NOTE: This type is intentionally loose — controllers read metadata and labels and do
    31	// not modify upstream status directly.
    32	//
    33	//nolint:tagliatelle // external API shape
    34	//nolint:revive // external API shape
    35	//nolint:stylecheck // external API shape
    36	//nolint:godox
    37	//
    38	// We implement only the minimal fields needed to act as a client.Object.
    39	// Spec/Status remain untyped to avoid drift.
    40	type NetworkDevice struct {
    41		metav1.TypeMeta   `json:",inline"`
    42		metav1.ObjectMeta `json:"metadata,omitempty"`
    43		Spec              map[string]any `json:"spec,omitempty"`
    44		Status            map[string]any `json:"status,omitempty"`
    45	}
    46	
    47	// DeepCopyObject implements runtime.Object (manual since we skip codegen here).
    48	func (in *NetworkDevice) DeepCopyObject() runtime.Object {
    49		if in == nil {
    50			return nil
    51		}
    52		out := new(NetworkDevice)
    53		*out = *in
    54		// shallow copy of maps is sufficient for our read-only use in watches
    55		return out
    56	}
    57	
    58	// NetworkDeviceList is required for informer cache.
    59	type NetworkDeviceList struct {
    60		metav1.TypeMeta `json:",inline"`
    61		metav1.ListMeta `json:"metadata,omitempty"`
    62		Items           []NetworkDevice `json:"items"`
    63	}
    64	
    65	// DeepCopyObject implements runtime.Object.
    66	func (in *NetworkDeviceList) DeepCopyObject() runtime.Object {
    67		if in == nil {
    68			return nil
    69		}
    70		out := new(NetworkDeviceList)
    71		*out = *in
    72		return out
    73	}
    74	
    75	var (
    76		// SchemeBuilder registers Kubenet types.
    77		SchemeBuilder = &scheme.Builder{GroupVersion: GroupVersion}
    78		// AddToScheme adds Kubenet types.
    79		AddToScheme = SchemeBuilder.AddToScheme
    80	)
    81	
    82	func init() {
    83		SchemeBuilder.Register(&NetworkDevice{}, &NetworkDeviceList{})
    84	}
