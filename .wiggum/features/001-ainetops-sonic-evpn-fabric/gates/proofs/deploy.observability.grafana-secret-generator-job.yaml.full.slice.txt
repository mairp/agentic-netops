     1	# T073: In-cluster generator Job to create Grafana admin Secret with random credentials
     2	apiVersion: batch/v1
     3	kind: Job
     4	metadata:
     5	  name: grafana-admin-secret-generator
     6	  namespace: monitoring
     7	spec:
     8	  template:
     9	    spec:
    10	      restartPolicy: OnFailure
    11	      serviceAccountName: grafana-secret-generator
    12	      containers:
    13	      - name: generator
    14	        # Pinned Kind node image (kind.node_image in versions.lock.yaml): carries
    15	        # kubectl + bash; no external registry pull required for the generator.
    16	        image: kindest/node@sha256:3abb816a5b1061fb15c6e9e60856ec40d56b7b52bcea5f5f1350bc6e2320b6f8
    17	        command: ["/bin/bash","-c"]
    18	        args:
    19	        - |
    20	          set -euo pipefail
    21	          # Generate non-default random admin credentials and store only in Secret
    22	          USER=$(cat /proc/sys/kernel/random/uuid | cut -c1-8)
    23	          PASS=$(cat /proc/sys/kernel/random/uuid | cut -c1-16)
    24	          kubectl -n monitoring create secret generic grafana-admin --dry-run=client \
    25	            --from-literal=admin-user=$USER --from-literal=admin-password=$PASS -o yaml | kubectl apply -f -
    26	          echo "grafana admin secret generated"
