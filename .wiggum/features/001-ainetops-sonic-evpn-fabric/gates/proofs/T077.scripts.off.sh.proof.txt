     1	#!/usr/bin/env bash
     2	set -euo pipefail
     3	
     4	# AINETOPS SONiC EVPN/VXLAN Fabric — shutdown/cleanup script (Phase 8)
     5	# Sole implementation of environment teardown per contracts/crd-api.md
     6	
     7	SCRIPT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" &>/dev/null && pwd)
     8	LIB_DIR="${SCRIPT_DIR}/lib"
     9	
    10	# Flags
    11	AINETOPS_CLUSTER_NAME=${AINETOPS_CLUSTER_NAME:-ainetops}
    12	DELETE_KIND=${AINETOPS_DELETE_KIND:-false}
    13	CAPTURE_EVIDENCE=${AINETOPS_CAPTURE_EVIDENCE:-false}
    14	
    15	usage() {
    16	  cat <<EOF
    17	Usage: $0 [--cluster-name NAME] [--delete-kind true|false] [--capture-evidence true|false]
    18	EOF
    19	}
    20	
    21	while [[ $# -gt 0 ]]; do
    22	  case "$1" in
    23	    --cluster-name) shift; AINETOPS_CLUSTER_NAME=${1:-$AINETOPS_CLUSTER_NAME} ;;
    24	    --delete-kind) shift; DELETE_KIND=${1:-$DELETE_KIND} ;;
    25	    --capture-evidence) shift; CAPTURE_EVIDENCE=${1:-$CAPTURE_EVIDENCE} ;;
    26	    -h|--help) usage; exit 0 ;;
    27	    *) echo "[off] unknown flag: $1" >&2; usage; exit 2 ;;
    28	  esac
    29	  shift || true
    30	done
    31	export AINETOPS_CLUSTER_NAME
    32	
    33	# Optional evidence capture
    34	if [[ "$CAPTURE_EVIDENCE" == "true" ]]; then
    35	  proofs="${SCRIPT_DIR}/../.wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs"
    36	  mkdir -p "$proofs"
    37	  if command -v kubectl >/dev/null 2>&1; then
    38	    CTX="kind-${AINETOPS_CLUSTER_NAME}"
    39	    kubectl --context "$CTX" get pods -A -o wide | nl -ba > "$proofs/kubectl-get-all-before-off.txt" || true
    40	    kubectl --context "$CTX" get crds | nl -ba > "$proofs/kubectl-get-crds-before-off.txt" || true
    41	  fi
    42	fi
    43	
    44	# Idempotent containerlab teardown and cleanup checks
    45	if [[ -x "${LIB_DIR}/containerlab.sh" ]]; then
    46	  "${LIB_DIR}/containerlab.sh" destroy || {
    47	    echo "[off] containerlab destroy reported leftovers" >&2
    48	    exit 1
    49	  }
    50	else
    51	  echo "[off] WARN: containerlab helper not found; skipping lab teardown" >&2
    52	fi
    53	
    54	# Optionally delete Kind cluster if requested
    55	if [[ "$DELETE_KIND" == "true" && -x "${LIB_DIR}/kind.sh" ]]; then
    56	  "${LIB_DIR}/kind.sh" delete
    57	fi
    58	
    59	# Cleanup owned network and generated Secrets (safe, scoped)
    60	if command -v docker >/dev/null 2>&1; then
    61	  if docker network inspect ainetops-mgmt >/dev/null 2>&1; then
    62	    # Do not remove if foreign label
    63	    if docker network inspect ainetops-mgmt -f '{{json .Labels}}' | grep -q '"ainetops.owner":"ainetops"'; then
    64	      docker network rm ainetops-mgmt >/dev/null 2>&1 || true
    65	    else
    66	      echo "[off] preserving non-owned network ainetops-mgmt" >&2
    67	    fi
    68	  fi
    69	fi
    70	
    71	# Remove generated Secrets from local repo if present
    72	rm -f "${SCRIPT_DIR}/../secrets/gnmi.key" "${SCRIPT_DIR}/../secrets/gnmi.crt" "${SCRIPT_DIR}/../secrets/ca.crt" 2>/dev/null || true
    73	
    74	echo "[off] Teardown complete (idempotent)."
