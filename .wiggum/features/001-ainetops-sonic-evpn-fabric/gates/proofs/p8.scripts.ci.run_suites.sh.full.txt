     1	#!/usr/bin/env bash
     2	# Run the Phase 8 verification test suites and capture logs under gates/proofs.
     3	# T079 contract: this runner is STRICT. Every suite records its real exit code;
     4	# the script prints a PASS/FAIL/SKIP summary and exits non-zero if any suite
     5	# failed. Suites that legitimately require a live lab report their skip reason
     6	# explicitly, so a green summary always means every listed suite actually ran
     7	# or was skipped with a printed, greppable reason.
     8	set -u
     9	
    10	ROOT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/../.." && pwd)
    11	PROOFS_DIR="$ROOT_DIR/.wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs"
    12	mkdir -p "$PROOFS_DIR"
    13	SUMMARY="$PROOFS_DIR/tests.summary.txt"
    14	: > "$SUMMARY"
    15	
    16	# envtest control-plane binaries (real assets required — a skipped envtest is
    17	# a failure, not a pass).
    18	if [[ -z "${KUBEBUILDER_ASSETS:-}" ]]; then
    19	  for d in /root/.local/share/kubebuilder-envtest/*-linux-amd64 /usr/local/kubebuilder/bin; do
    20	    if [[ -x "$d/kube-apiserver" ]]; then export KUBEBUILDER_ASSETS="$d"; break; fi
    21	  done
    22	fi
    23	
    24	record(){ printf '%s %s\n' "$2" "$1" >>"$SUMMARY"; }
    25	
    26	suite(){
    27	  local name=$1 out=$2; shift 2
    28	  echo "[run] $name: $*" >"$out"
    29	  set +e
    30	  ( "$@" ) >>"$out" 2>&1
    31	  local rc=$?
    32	  set -e
    33	  if [[ $rc -eq 0 ]]; then
    34	    record "$name" PASS
    35	  elif grep -q 'SKIP-LIVE' "$out"; then
    36	    record "$name" "SKIP-LIVE($(grep -m1 -o 'SKIP-LIVE[^"]*' "$out" | head -c 60))"
    37	  else
    38	    record "$name" "FAIL(rc=$rc)"
    39	  fi
    40	}
    41	
    42	go_bin=""
    43	command -v go >/dev/null 2>&1 && go_bin=go
    44	
    45	# 1) API suite — CRD/envtest sanity (requires real envtest assets)
    46	if [[ -n "$go_bin" ]]; then
    47	  suite api "$PROOFS_DIR/tests.api.log" env KUBEBUILDER_ASSETS="${KUBEBUILDER_ASSETS:-}" \
    48	    "$go_bin" test ./tests/envtest -v
    49	else
    50	  echo "FAIL: no go toolchain" >"$PROOFS_DIR/tests.api.log"; record api "FAIL(no-go)"
    51	fi
    52	
    53	# 2) Unit suite — all unit tests
    54	[[ -n "$go_bin" ]] && suite unit "$PROOFS_DIR/tests.unit.log" "$go_bin" test ./tests/unit -v
    55	
    56	# 3) Golden suite — table/golden tests only
    57	[[ -n "$go_bin" ]] && suite golden "$PROOFS_DIR/tests.golden.log" "$go_bin" test ./tests/unit -run Golden -v
    58	
    59	# 4) SDC validation suite — offline validators + path-register guards
    60	[[ -n "$go_bin" ]] && suite sdc-validation "$PROOFS_DIR/tests.sdc-validation.log" \
    61	  "$go_bin" test ./tests/unit -run 'OfflineValidate|RegisterGuard|RendererPathsCoveredByRegister|FullValidate' -v
    62	
    63	# 5) Integration: API-level probes over gNMI and inventory checks (live lab)
    64	suite integration "$PROOFS_DIR/tests.integration.log" "$ROOT_DIR/tests/integration/fabric_verify.sh" run
    65	
    66	# 6) Failure suite — negative cases and recovery (live lab)
    67	suite failure "$PROOFS_DIR/tests.failure.log" "$ROOT_DIR/tests/integration/failure_recovery_invalid_yang.sh" run
    68	
    69	# 7) Traffic suite — EVPN client traffic (live lab)
    70	suite traffic "$PROOFS_DIR/tests.traffic.log" "$ROOT_DIR/tests/integration/evpn_traffic.sh" run
    71	
    72	# 8) SRv6 packet-capture suite (live lab)
    73	suite srv6-capture "$PROOFS_DIR/tests.srv6-capture.log" "$ROOT_DIR/tests/integration/srv6_capture_counters.sh" run
    74	
    75	# 9) SRv6 failover/path-change suite (live lab)
    76	suite srv6-failover "$PROOFS_DIR/tests.srv6-failover.log" "$ROOT_DIR/tests/integration/srv6_failover_path_change.sh" run
    77	
    78	# 10) Topology parity suite (live lab)
    79	suite topology-parity "$PROOFS_DIR/tests.topology-parity.log" "$ROOT_DIR/tests/integration/topology_parity.sh"
    80	
    81	# 11) Observability suite
    82	suite observability "$PROOFS_DIR/tests.observability.log" "$ROOT_DIR/tests/integration/observability_suite.sh"
    83	
    84	# 12) Teardown suite (lab-state safe/idempotent semantics)
    85	suite teardown "$PROOFS_DIR/tests.teardown.log" "$ROOT_DIR/tests/integration/teardown_suite.sh"
    86	
    87	echo "================ tests.summary.txt ================"
    88	cat "$SUMMARY"
    89	fails=$(grep -c '^FAIL' "$SUMMARY" || true)
    90	skips=$(grep -c '^SKIP' "$SUMMARY" || true)
    91	passes=$(grep -c '^PASS' "$SUMMARY" || true)
    92	echo "[run_suites] PASS=$passes FAIL=$fails SKIP=$skips"
    93	if [[ "$fails" -ne 0 ]]; then
    94	  echo "ALL_SUITES_FAILED=$fails"
    95	  exit 1
    96	fi
    97	echo "ALL_SUITES_PASSED"
