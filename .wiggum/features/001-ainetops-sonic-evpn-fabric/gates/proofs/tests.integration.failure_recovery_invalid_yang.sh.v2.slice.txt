     1	#!/usr/bin/env bash
     2	# T049 [US2] Partial target failure/recovery, provider restart mid-transaction, and invalid-YANG tests
     3	set -euo pipefail
     4	
     5	ROOT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/../.." && pwd)
     6	TOPO=${TOPO:-$ROOT_DIR/lab/topology.clab.yml}
     7	CTX=${CTX:-kind-ainetops}
     8	PROOF_DIR=${PROOF_DIR:-.wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs}
     9	mkdir -p "$PROOF_DIR"
    10	
    11	assert_aggregate_not_ready() {
    12	  # When any target is not Ready, aggregate readiness must not be True
    13	  local any_not_ready
    14	  any_not_ready=$(kubectl --context "$CTX" -n sdc get target -o json \
    15	    | jq -r '[.items[].status.conditions[]|select(.type=="Ready")|.status] | any(. != "True")')
    16	  if [[ "$any_not_ready" != "true" ]]; then
    17	    echo "[failure] WARN: all targets appear Ready; the environment may be already healthy" >&2
    18	  fi
    19	  local agg
    20	  agg=$(kubectl --context "$CTX" -n sdc get target -o json \
    21	    | jq -r '[.items[].status.conditions[]|select(.type=="Ready")|.status] | all(. == "True")')
    22	  if [[ "$agg" == "true" && "$any_not_ready" == "true" ]]; then
    23	    echo "[failure] ERROR: false aggregate Ready detected (some targets not Ready but aggregate all True)" >&2
    24	    exit 1
    25	  fi
    26	}
    27	
    28	partial_target_failure_recovery() {
    29	  echo "[failure] partial target failure/recovery"
    30	  # containerlab 0.7x CLI: `stop`/`start` take node names (no `node` subcommand)
    31	  containerlab stop -t "$TOPO" leaf02 || true
    32	  # Expect aggregate not Ready (Degraded) reflected in provider/SDC/Network status
    33	  kubectl --context "$CTX" -n sdc get target -o wide | tee "$PROOF_DIR/partial-failure.targets.txt" >/dev/null
    34	  assert_aggregate_not_ready
    35	  # Restore
    36	  containerlab start -t "$TOPO" leaf02 || true
    37	}
