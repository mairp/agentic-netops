     1	#!/usr/bin/env bash
     2	# T080: Runtime scan for standalone/Compose workloads. The goal is to ensure
     3	# the AINETOPS project itself does not run standalone/Compose application
     4	# containers. Other unrelated host containers must not cause a failure.
     5	#
     6	# Policy:
     7	# - PASS when no containers owned by AINETOPS (by label ainetops.owner=ainetops
     8	#   or name prefix ainetops-/clab-ainetops-) are managed by docker-compose or
     9	#   are running outside Kubernetes.
    10	# - Advisory print of other host containers, but do not fail on them.
    11	# - Emit RUNTIME_SCAN_NO_STANDALONE on success; exit non-zero on violation.
    12	set -euo pipefail
    13	
    14	ok=1
    15	violations=()
    16	
    17	# Detect AINETOPS-owned or -named containers that are managed by compose
    18	if command -v docker >/dev/null 2>&1; then
    19	  # List: name labels
    20	  while IFS= read -r line; do
    21	    name=$(cut -d' ' -f1 <<<"$line")
    22	    labels=$(cut -d' ' -f2- <<<"$line")
    23	    # Only consider containers clearly tied to this project
    24	    if [[ "$labels" == *"ainetops.owner=ainetops"* ]] || [[ "$name" =~ ^(ainetops|clab-ainetops) ]]; then
    25	      if grep -Eq 'docker-compose|compose_project|compose.service|com.docker.compose' <<<"$labels"; then
    26	        violations+=("$name has compose labels: $labels")
    27	        ok=0
    28	      fi
    29	    fi
    30	  done < <(docker ps -a --format '{{.Names}} {{.Labels}}' 2>/dev/null || true)
    31	fi
    32	
    33	# Prefer in-cluster deployments; if kubectl is available, list known namespaces (advisory)
    34	if command -v kubectl >/dev/null 2>&1; then
    35	  kubectl get pods -A 2>/dev/null | awk 'NR>1{print $1}' | sort -u | tee /dev/stderr || true
    36	fi
    37	
    38	if [[ ${#violations[@]} -gt 0 ]]; then
    39	  echo "[runtime-scan] Found AINETOPS-owned containers managed by docker-compose:" >&2
    40	  printf ' - %s\n' "${violations[@]}" >&2
    41	  ok=0
    42	fi
    43	
    44	if [[ $ok -eq 1 ]]; then
    45	  echo "RUNTIME_SCAN_NO_STANDALONE"
    46	else
    47	  exit 1
    48	fi
