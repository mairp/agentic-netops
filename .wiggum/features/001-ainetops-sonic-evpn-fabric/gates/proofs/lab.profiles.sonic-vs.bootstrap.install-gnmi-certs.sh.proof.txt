1: #!/usr/bin/env bash
2: # Idempotently install gNMI TLS certs into /etc/sonic/telemetry
3: set -euo pipefail
4: 
5: dst=/etc/sonic/telemetry
6: src=${GNMI_CERT_SRC_DIR:-/etc/ainetops/gnmi}
7: mkdir -p "$dst"
8: install -m 0600 -T "$src/gnmi.key" "$dst/gnmi.key"
9: install -m 0644 -T "$src/gnmi.crt" "$dst/gnmi.crt"
10: install -m 0644 -T "$src/ca.crt" "$dst/ca.crt"
11: 
12: echo "[install-gnmi-certs] installed certs to $dst"
