     1	package unit
     2	
     3	import (
     4		"os/exec"
     5		"path/filepath"
     6		"strings"
     7		"testing"
     8	)
     9	
    10	// Ensure the CLI rejects a validation failure (unsupported TE) with structured JSON and no YAML outputs.
    11	func TestMigrationCLI_RejectsUnsupportedTE_NoOutput(t *testing.T) {
    12		cmd := exec.Command("go", "run", "./cmd/migration-translator", "--file", "tests/unit/testdata/migration/unsupported_te.json")
    13		cmd.Dir = filepath.Join("..", "..")
    14		out, err := cmd.CombinedOutput()
    15		if err == nil {
    16			t.Fatalf("expected non-zero exit; output=%s", string(out))
    17		}
    18		if !strings.Contains(string(out), "\"error\": \"validation\"") {
    19			t.Fatalf("missing structured validation error: %s", string(out))
    20		}
    21		if strings.Contains(string(out), "spec:") {
    22			t.Fatalf("unexpected YAML output on validation failure: %s", string(out))
    23		}
    24	}
