     1	#!/usr/bin/env bash
     2	# T043 [US3] Verify underlay/EVPN sessions, loopback reachability, IPv6 waypoint reachability,
     3	# and absence of tenant VTEP/VRF state on spines (FR-004)
     4	set -euo pipefail
     5	
     6	GNMIC_BIN=${GNMIC_BIN:-gnmic}
     7	GNMI_USER=${GNMI_USER:-admin}
     8	GNMI_PASS=${GNMI_PASS:-admin}
     9	GNMI_CACERT=${GNMI_CACERT:-./secrets/ca.crt}
    10	GNMI_CERT=${GNMI_CERT:-./secrets/gnmi.crt}
    11	GNMI_KEY=${GNMI_KEY:-./secrets/gnmi.key}
    12	GNMI_ENCODING=${GNMI_ENCODING:-JSON_IETF}
    13	# SONiC targets (management gNMI endpoints from lab/topology.clab.yml)
    14	LEAVES=${LEAVES:-"172.31.0.21:8080,172.31.0.22:8080"}
    15	SPINES=${SPINES:-"172.31.0.11:8080,172.31.0.12:8080"}
    16	CLAB_PREFIX=${CLAB_PREFIX:-clab-ainetops-fabric-}
    17	
    18	_args_common=(--timeout 10s --username "$GNMI_USER" --password "$GNMI_PASS" --insecure --encoding "$GNMI_ENCODING" --cacert "$GNMI_CACERT" --cert "$GNMI_CERT" --key "$GNMI_KEY")
    19	
    20	run_each() {
    21	  local targets_csv=$1; shift
    22	  IFS=',' read -ra tgts <<<"$targets_csv"
    23	  local rc=0
    24	  for t in "${tgts[@]}"; do
    25	    if ! "$GNMIC_BIN" --address "$t" "${_args_common[@]}" "$@"; then rc=1; fi
    26	  done
    27	  return $rc
    28	}
    29	
    30	run_each_capture() {
    31	  # Like run_each but captures output per target to stdout, prefixing target for clarity
    32	  local targets_csv=$1; shift
    33	  IFS=',' read -ra tgts <<<"$targets_csv"
    34	  local rc=0
    35	  for t in "${tgts[@]}"; do
    36	    if ! "$GNMIC_BIN" --address "$t" "${_args_common[@]}" "$@" | sed -e "s/^/[$t] /"; then rc=1; fi
    37	  done
    38	  return $rc
    39	}
    40	
    41	assert_grep_all() {
    42	  # assert_grep_all <targets_csv> <gnmi_path> <grep_pattern>
    43	  local targets_csv=$1; shift
    44	  local path=$1; shift
    45	  local pattern=$1; shift
    46	  IFS=',' read -ra tgts <<<"$targets_csv"
    47	  local rc=0
    48	  for t in "${tgts[@]}"; do
    49	    local out
    50	    set +e
    51	    out=$("$GNMIC_BIN" --address "$t" "${_args_common[@]}" get --path "$path" 2>&1)
    52	    local e=$?
    53	    set -e
    54	    echo "[$t] get $path" >&2
    55	    if [[ $e -ne 0 ]] || ! grep -qE "$pattern" <<<"$out"; then
    56	      echo "[$t] ASSERTION FAILED: expected to find pattern '$pattern' in gNMI get $path output" >&2
    57	      echo "----- BEGIN [$t] OUTPUT -----" >&2
    58	      echo "$out" >&2
    59	      echo "----- END [$t] OUTPUT -----" >&2
    60	      rc=1
    61	    else
    62	      echo "[$t] assertion passed: pattern '$pattern' present"
    63	    fi
    64	  done
    65	  return $rc
    66	}
    67	
    68	verify_underlay_bgp() {
    69	  echo "[fabric-verify] underlay BGP neighbors/ESTABLISHED (IPv4/IPv6) on spines and leaves"
    70	  local path_sess="/openconfig-network-instance:network-instances/network-instance/protocols/protocol[identifier=BGP][name=BGP]/neighbors/neighbor/state/session-state"
    71	  assert_grep_all "$LEAVES" "$path_sess" 'ESTABLISHED'
    72	  assert_grep_all "$SPINES" "$path_sess" 'ESTABLISHED'
    73	  echo "[fabric-verify] verify EVPN AF activation on BGP neighbors (L2VPN_EVPN)"
    74	  local path_evpn_af="/openconfig-network-instance:network-instances/network-instance/protocols/protocol[identifier=BGP][name=BGP]/neighbors/neighbor/afi-safis/afi-safi[afi-safi-name=L2VPN_EVPN]/state/enabled"
    75	  assert_grep_all "$LEAVES" "$path_evpn_af" 'true'
    76	}
    77	
    78	verify_evpn_overlay() {
    79	  echo "[fabric-verify] EVPN overlay route-table presence (Type2/Type3/Type5) on leaves"
    80	  run_each "$LEAVES" get --path "/openconfig-network-instance:network-instances/network-instance/evpn/route-tables/route-table[type=EVPN_TYPE2]"
    81	  run_each "$LEAVES" get --path "/openconfig-network-instance:network-instances/network-instance/evpn/route-tables/route-table[type=EVPN_TYPE3]"
    82	  run_each "$LEAVES" get --path "/openconfig-network-instance:network-instances/network-instance/evpn/route-tables/route-table[type=EVPN_TYPE5_IP_PREFIX]"
    83	}
    84	
    85	_fetch_loopback_v6() {
    86	  # fetch Loopback0 IPv6 address on a single target (first match)
    87	  local target=$1
    88	  local path="/openconfig-interfaces:interfaces/interface[name=Loopback0]/subinterfaces/subinterface[index=0]/ipv6/addresses/address/state/ip"
    89	  "$GNMIC_BIN" --address "$target" "${_args_common[@]}" get --path "$path" -o json \
    90	    | jq -r '..|objects|select(has("ip"))|.ip' | head -n1
    91	}
    92	
    93	verify_loopback_reachability() {
    94	  echo "[fabric-verify] loopback reachability across all nodes (IPv6)"
    95	  # Use containerlab exec to ping between leaf loopbacks via mgmt net namespace
    96	  local l1="${CLAB_PREFIX}leaf01" l2="${CLAB_PREFIX}leaf02"
    97	  # Discover loopback IPv6 addresses via gNMI
    98	  local lo1 lo2
    99	  lo1=$(_fetch_loopback_v6 "${LEAVES%%,*}") || true
   100	  lo2=$(_fetch_loopback_v6 "${LEAVES##*,}") || true
   101	  if [[ -n "${lo1:-}" && -n "${lo2:-}" ]]; then
   102	    echo "[fabric-verify] ping6 leaf01(${lo1}) -> leaf02(${lo2})"
   103	    docker exec "$l1" bash -lc "ping -6 -c 3 -W 2 ${lo2}"
   104	    echo "[fabric-verify] ping6 leaf02(${lo2}) -> leaf01(${lo1})"
   105	    docker exec "$l2" bash -lc "ping -6 -c 3 -W 2 ${lo1}"
   106	    echo "loopback reachability" # keyword for proof grepping
   107	  else
   108	    echo "[fabric-verify] WARN: could not auto-discover loopback IPv6 addresses; skipping loopback ping" >&2
   109	    return 1
   110	  fi
   111	}
   112	
   113	verify_ipv6_waypoint_reachability() {
   114	  echo "[fabric-verify] IPv6 waypoint reachability for SRv6 path"
   115	  # Probe connectivity to configured transit waypoints (if any) by traceroute6 from leaf01
   116	  local src="${CLAB_PREFIX}leaf01"
   117	  local waypoints=${SRV6_WAYPOINTS:-""}
   118	  if [[ -z "$waypoints" ]]; then
   119	    echo "[fabric-verify] INFO: no SRV6_WAYPOINTS provided; attempting traceroute to peer leaf loopback as waypoint surrogate"
   120	    local dst_v6
   121	    dst_v6=$(_fetch_loopback_v6 "${LEAVES##*,}") || true
   122	    if [[ -n "${dst_v6:-}" ]]; then
   123	      docker exec "$src" bash -lc "traceroute -6 -n -q1 -m 10 ${dst_v6} || true"
   124	      echo "IPv6 waypoint reachability" # keyword for proof grepping
   125	    else
   126	      echo "[fabric-verify] WARN: no waypoint or dst loopback; skipping" >&2
   127	      return 1
   128	    fi
   129	  else
   130	    for wp in $waypoints; do
   131	      docker exec "$src" bash -lc "traceroute -6 -n -q1 -m 10 ${wp} || true"
   132	    done
   133	    echo "IPv6 waypoint reachability"
   134	  fi
   135	}

   137	assert_no_tenant_state_on_spines() {
   138	  echo "[fabric-verify] Assert absence of tenant VTEP/VRF state on spines (FR-004)"
   139	  # Negative checks:
   140	  # 1) No VXLAN tunnel objects on spines (SONiC native)
   141	  local vxlan_path="/sonic-vxlan:sonic-vxlan/VXLAN_TUNNEL"
   142	  IFS=',' read -ra tgts <<<"$SPINES"
   143	  local rc=0
   144	  for t in "${tgts[@]}"; do
   145	    set +e
   146	    local out
   147	    out=$("$GNMIC_BIN" --address "$t" "${_args_common[@]}" get --path "$vxlan_path" -o json 2>&1)
   148	    set -e
   149	    if grep -qiE 'VXLAN_TUNNEL|VTEP' <<<"$out"; then
   150	      echo "[$t] ASSERTION FAILED: spine has VXLAN/VTEP state present" >&2
   151	      rc=1
   152	    else
   153	      echo "[$t] OK: no VXLAN/VTEP state on spine"
   154	    fi
   155	  done
   156	  # 2) No tenant VRFs present on spines (only default or mgmt)
   157	  local ni_path="/openconfig-network-instance:network-instances/network-instance/name"
   158	  for t in "${tgts[@]}"; do
   159	    local out
   160	    out=$("$GNMIC_BIN" --address "$t" "${_args_common[@]}" get --path "$ni_path" -o json 2>/dev/null | jq -r '..|scalars? // empty' | tr '\n' ' ')
   161	    if grep -Eqi 'vrf-|tenant-|l3vni|100|101|102' <<<"$out"; then
   162	      echo "[$t] ASSERTION FAILED: unexpected tenant VRF/network-instance names on spine: $out" >&2
   163	      rc=1
   164	    else
   165	      echo "[$t] OK: no tenant VRF names detected on spine"
   166	    fi
   167	  done
   168	  echo "absence of tenant VTEP/VRF state on spines (FR-004)" # keyword for proof grepping
   169	  return $rc
   170	}
   171	
   172	case "${1:-run}" in
   173	  run)
   174	    verify_underlay_bgp
   175	    verify_evpn_overlay
   176	    verify_loopback_reachability
   177	    verify_ipv6_waypoint_reachability
   178	    assert_no_tenant_state_on_spines
   179	    ;;
   180	  *)
   181	    echo "usage: $0 run" >&2; exit 2
   182	    ;;
   183	esac
