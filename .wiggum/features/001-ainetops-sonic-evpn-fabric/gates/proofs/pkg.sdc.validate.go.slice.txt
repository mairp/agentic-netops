     4	import (

     9		yaml "gopkg.in/yaml.v3"

    11	
    12	// RegisterError reports missing register coverage for rendered paths.
    13	type RegisterError struct {
    14		MissingPaths []string
    15	}
    16	
    17	func (e *RegisterError) Error() string { return fmt.Sprintf("unregistered rendered paths: %v", e.MissingPaths) }
    18	
    19	// IsRegisterError returns true when err is a RegisterError.
    20	func IsRegisterError(err error) bool {
    21		var re *RegisterError
    22		return errors.As(err, &re)
    23	}
    24	
    25	// ValidateSpecAgainstRegister ensures each rendered path is present in the OpenConfig-vs-SONiC register
    26	// and prefers OpenConfig when available. This is a lightweight CI-friendly guard; the full
    27	// SDC schema validation is integrated later.
    28	func ValidateSpecAgainstRegister(spec map[string]any, registerYAML []byte) error {
    29		paths := make([]string, 0, len(spec))
    30		for p := range spec { paths = append(paths, p) }
    31		sort.Strings(paths)
    32		reg := struct{
    33			Entries []struct{
    34				Path         string `yaml:"path"`
    35				Prefer       string `yaml:"prefer"`
    36				NativePath   string `yaml:"native_path"`
    37				Justification string `yaml:"justification"`
    38			} `yaml:"entries"`
    39		}{}
    40		var data []byte
    41		var err error
    42		if registerYAML == nil || len(registerYAML) == 0 {
    43			data, err = defaultRegister()
    44			if err != nil { return fmt.Errorf("load default register: %w", err) }
    45		} else { data = registerYAML }
    46		if err := yaml.Unmarshal(data, &reg); err != nil { return fmt.Errorf("parse register: %w", err) }
    47		byPath := map[string]string{}
    48		for _, e := range reg.Entries { byPath[e.Path] = e.Prefer }
    49		missing := []string{}
    50		for _, p := range paths {
    51			if _, ok := byPath[p]; !ok {
    52				missing = append(missing, p)
    53			}
    54		}
    55		if len(missing) > 0 {
    56			return &RegisterError{MissingPaths: missing}
    57		}
    58		return nil
    59	}
    60
