103\tpreflight::tool_versions() {
104\t  # Verify host tool versions match pins in versions.lock.yaml
105\t  for c in kind kubectl helm containerlab jq curl; do preflight::require_cmd "$c"; done
106\t
107\t  local kind_pin kubectl_pin helm_pin clab_pin
108\t  kind_pin=$(preflight::yaml_value kind binary)
109\t  # Prefer host_tools.kubectl if present, else fall back to kubernetes.kubernetes
110\t  kubectl_pin=$(preflight::yaml_value host_tools kubectl)
111\t  if [[ -z "${kubectl_pin}" ]]; then kubectl_pin=$(preflight::yaml_value kubernetes kubernetes); fi
112\t  helm_pin=$(preflight::yaml_value host_tools helm)
113\t  clab_pin=$(preflight::yaml_value containerlab version)
114\t
115\t  [[ -n "$kind_pin" ]] || preflight::die "missing kind.binary pin in versions.lock.yaml"
116\t  [[ -n "$kubectl_pin" ]] || preflight::die "missing host_tools.kubectl or kubernetes.kubernetes pin in versions.lock.yaml"
117\t  [[ -n "$helm_pin" ]] || preflight::die "missing host_tools.helm pin in versions.lock.yaml"
118\t  [[ -n "$clab_pin" ]] || preflight::die "missing containerlab.version pin in versions.lock.yaml"
119\t
120\t  # kind
121\t  local kind_ver; kind_ver=$(kind version 2>/dev/null | awk '{print $2}') || true
122\t  [[ "$kind_ver" == "$kind_pin" ]] || preflight::die "kind version $kind_ver != pinned $kind_pin"
123\t
124\t  # kubectl
125\t  local kubectl_ver; kubectl_ver=$(kubectl version --client -o json 2>/dev/null | jq -r '.clientVersion.gitVersion' || true)
126\t  [[ "$kubectl_ver" == "$kubectl_pin" ]] || preflight::die "kubectl version $kubectl_ver != pinned $kubectl_pin"
127\t
128\t  # helm
129\t  local helm_ver; helm_ver=$(helm version --short --client 2>/dev/null | sed 's/+.*//' || true)
130\t  [[ "$helm_ver" == "$helm_pin" ]] || preflight::die "helm version $helm_ver != pinned $helm_pin"
131\t
132\t  # containerlab (parse the 'version:' line)
133\t  local clab_ver; clab_ver=$(containerlab version 2>/dev/null | awk -F': *' '/^version:/ {print $2; exit}')
134\t  [[ "$clab_ver" == "$clab_pin" ]] || preflight::die "containerlab version $clab_ver != pinned $clab_pin"
135\t}
