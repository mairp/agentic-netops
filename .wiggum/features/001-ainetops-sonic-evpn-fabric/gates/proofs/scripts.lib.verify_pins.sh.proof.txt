     1	#!/usr/bin/env bash
     2	# Verify versions.lock.yaml pins per NFR-003
     3	set -euo pipefail
     4	
     5	fail() { echo "[verify-pins] ERROR: $*" >&2; exit 1; }
     6	info() { echo "[verify-pins] $*" >&2; }
     7	
     8	ROOT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/../.." && pwd)
     9	LOCK_FILE="${ROOT_DIR}/versions.lock.yaml"
    10	
    11	[[ -f "$LOCK_FILE" ]] || fail "missing $LOCK_FILE"
    12	
    13	# Reject floating refs
    14	if grep -En '\blatest\b|\bmain\b|\bmaster\b|\bHEAD\b' "$LOCK_FILE" >/dev/null; then
    15	  fail "floating refs found (latest/main/master/HEAD)"
    16	fi
    17	
    18	# Extract helper: get yaml value by key path (simple, indentation-based)
    19	get_block() {
    20	  # get_block SECTION_NAME returns block lines until next top-level key
    21	  awk -v key="$1:" '
    22	    $1==key {print; in=1; next}
    23	    in && /^[^[:space:]]/ {exit}
    24	    in {print}
    25	  ' "$LOCK_FILE"
    26	}
    27	
    28	# Check kind node image digest
    29	if ! awk '/^kind:/,/^[^[:space:]]/ {print}' "$LOCK_FILE" | grep -E '^\s*node_image:\s*[^@]+@sha256:[0-9a-f]{64}\s*$' >/dev/null; then
    30	  fail "kind.node_image must include @sha256 digest"
    31	fi
    32	
    33	# Check controller-runtime and Go
    34	awk '/^kubernetes:/,/^[^[:space:]]/ {print}' "$LOCK_FILE" | grep -E '^\s*controller_runtime:\s*v[0-9]+\.[0-9]+\.[0-9]+' >/dev/null || fail "missing kubernetes.controller_runtime semver"
    35	awk '/^kubernetes:/,/^[^[:space:]]/ {print}' "$LOCK_FILE" | grep -E '^\s*go:\s*\"?1\.[0-9]+(\.[0-9]+)?\"?$' >/dev/null || fail "missing kubernetes.go version"
    36	
    37	# Check Kubenet/KUID/SDC pinned release+commit
    38	for sec in kubenet kuid sdc; do
    39	  block=$(awk -v s="^${sec}:$" 'f; $0~s{f=1} f && /^[^[:space:]]/{if (!p){p=1; print} else exit} f && p{print}' "$LOCK_FILE") || true
    40	  grep -E "^\s*release:\s*v?[0-9]+\.[0-9]+\.[0-9]+" <<<"$block" >/dev/null || fail "$sec.release missing or not semver"
    41	  grep -E "^\s*commit:\s*[0-9a-f]{40}\s*$" <<<"$block" >/dev/null || fail "$sec.commit must be 40-hex"
    42	  if [[ "$sec" == "kubenet" ]]; then
    43	    grep -E '^\s*api_shape:\s*(NetworkConfig|NetworkDesign)\s*$' <<<"$block" >/dev/null || fail "kubenet.api_shape must be NetworkConfig or NetworkDesign"
    44	  fi
    45	done
    46	
    47	# Tooling images must be pinned by digest
    48	awk '/^tooling:/,/^[^[:space:]]/ {print}' "$LOCK_FILE" | grep -E '@sha256:[0-9a-f]{64}' >/dev/null || fail "tooling images must include @sha256 digests"
    49	
    50	# Containerlab version pinned
    51	awk '/^containerlab:/,/^[^[:space:]]/ {print}' "$LOCK_FILE" | grep -E '^\s*version:\s*[0-9]+\.[0-9]+\.[0-9]+' >/dev/null || fail "containerlab.version must be semver"
    52	
    53	# SONiC images pinned: allow image+separate digest, but require digest present
    54	svs_img=$(awk '/sonic_images:/,/^[^[:space:]]/ {print}' "$LOCK_FILE" | awk '/sonic_vs:/, /^\s*[^[:space:]]/ {print}' | grep -E '^\s*image:\s*' | sed 's/.*image:\s*//')
    55	svs_dig=$(awk '/sonic_images:/,/^[^[:space:]]/ {print}' "$LOCK_FILE" | awk '/sonic_vs:/, /^\s*[^[:space:]]/ {print}' | grep -E '^\s*digest:\s*' | sed 's/.*digest:\s*//')
    56	svm_img=$(awk '/sonic_images:/,/^[^[:space:]]/ {print}' "$LOCK_FILE" | awk '/sonic_vm:/, /^\s*[^[:space:]]/ {print}' | grep -E '^\s*image:\s*' | sed 's/.*image:\s*//')
    57	svm_dig=$(awk '/sonic_images:/,/^[^[:space:]]/ {print}' "$LOCK_FILE" | awk '/sonic_vm:/, /^\s*[^[:space:]]/ {print}' | grep -E '^\s*digest:\s*' | sed 's/.*digest:\s*//')
    58	
    59	[[ -n "$svs_img" && -n "$svs_dig" ]] || fail "sonic_vs image and digest required"
    60	[[ -n "$svm_img" && -n "$svm_dig" ]] || fail "sonic_vm image and digest required"
    61	[[ "$svs_dig" =~ ^sha256:[0-9a-f]{64}$ ]] || fail "sonic_vs digest must be sha256:..."
    62	[[ "$svm_dig" =~ ^sha256:[0-9a-f]{64}$ ]] || fail "sonic_vm digest must be sha256:..."
    63	
    64	svs_full="${svs_img%@*}@${svs_dig}"
    65	svm_full="${svm_img%@*}@${svm_dig}"
    66	
    67	# YANG compatibility must include entries for both images and match commit prefixes
    68	oc_commit=$(awk '/^sonic_yang:/,/^[^[:space:]]/ {print}' "$LOCK_FILE" | grep -E '^\s*openconfig_commit:' | sed 's/.*openconfig_commit:\s*//')
    69	na_commit=$(awk '/^sonic_yang:/,/^[^[:space:]]/ {print}' "$LOCK_FILE" | grep -E '^\s*sonic_native_commit:' | sed 's/.*sonic_native_commit:\s*//')
    70	[[ "$oc_commit" =~ ^[0-9a-f]{40}$ ]] || fail "openconfig_commit must be 40-hex"
    71	[[ "$na_commit" =~ ^[0-9a-f]{40}$ ]] || fail "sonic_native_commit must be 40-hex"
    72	
    73	oc_pref=${oc_commit:0:8}
    74	na_pref=${na_commit:0:8}
    75	
    76	compat_block=$(awk '/^sonic_yang:/,/^[^[:space:]]/ {print}' "$LOCK_FILE" | awk '/compatibility:/, /^[^[:space:]]/ {print}')
    77	
    78	grep -F "image: ${svs_full}" <<<"$compat_block" >/dev/null || fail "compatibility missing sonic_vs image ${svs_full}"
    79	
    80	grep -F "image: ${svm_full}" <<<"$compat_block" >/dev/null || fail "compatibility missing sonic_vm image ${svm_full}"
    81	
    82	grep -F "oc_version: openconfig@${oc_pref}" <<<"$compat_block" >/dev/null || fail "compatibility oc_version must match openconfig commit prefix ${oc_pref}"
    83	
    84	grep -F "native_version: sonic_yang@${na_pref}" <<<"$compat_block" >/dev/null || fail "compatibility native_version must match native commit prefix ${na_pref}"
    85	
    86	info "versions.lock.yaml pins and compatibility are consistent"
