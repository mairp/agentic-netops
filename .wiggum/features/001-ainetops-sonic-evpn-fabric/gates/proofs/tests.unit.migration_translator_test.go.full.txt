    11	func TestVPLSTranslation(t *testing.T) {
    12		in := migration.ServiceInput{
    13			ServiceID: "svc1",
    14			Type:      migration.ServiceVPLS,
    15			Tenant:    "A",
    16			RDRT:      &migration.RdRt{RD: "65000:100", ImportRT: []string{"65000:100"}, ExportRT: []string{"65000:100"}},
    17			L2VNI:     10010,
    18			Endpoints: []migration.Endpoint{{Node: "leaf01", Attachment: "client01", VLAN: 10}, {Node: "leaf02", Attachment: "client02", VLAN: 10}},
    19		}
    20		if err := in.ValidateAllOrNothing(0, false); err != nil { t.Fatalf("validate: %v", err) }
    21		out, err := migration.Translate(&in)
    22		if err != nil { t.Fatalf("translate: %v", err) }
    23		if !strings.Contains(out.NetworkYAML, "bridgeDomains") { t.Fatalf("expected bridgeDomains in YAML") }
    24		if !strings.Contains(out.NetworkYAML, "l2vni: 10010") { t.Fatalf("expected l2vni") }
    25	}

    96	func TestDeterministicHash(t *testing.T) {
    97		in := migration.ServiceInput{
    98			ServiceID: "svc6",
    99			Type:      migration.ServiceVPLS,
   100			Tenant:    "A",
   101			RDRT:      &migration.RdRt{RD: "65000:600", ImportRT: []string{"65000:600"}, ExportRT: []string{"65000:600"}},
   102			L2VNI:     10600,
   103			Endpoints: []migration.Endpoint{{Node: "leaf01", Attachment: "client01", VLAN: 60}, {Node: "leaf02", Attachment: "client02", VLAN: 60}},
   104		}
   105		h1, _ := in.CanonicalHash()
   106		// Re-marshal through JSON roundtrip to prove determinism
   107		b, _ := json.Marshal(in)
   108		var in2 migration.ServiceInput
   109		_ = json.Unmarshal(b, &in2)
   110		h2, _ := in2.CanonicalHash()
   111		if h1 != h2 { t.Fatalf("hashes differ: %s vs %s", h1, h2) }
   112	}
