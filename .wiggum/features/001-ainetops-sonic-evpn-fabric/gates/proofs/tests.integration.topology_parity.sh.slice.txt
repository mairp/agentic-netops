     1	#!/usr/bin/env bash
     2	# T079: Topology parity test — verify that the generated/pinned topology ConfigMap
     3	# (deploy/observability/topology-configmap.yaml) matches the containerlab topology
     4	# (lab/topology.clab.yml) by node ids and undirected link endpoints.
     5	#
     6	# The test prefers live containerlab inspect JSON on stdin or via --inspect <path.json>.
     7	# When inspect JSON is not available, it derives expected nodes/links from the
     8	# containerlab topology YAML directly (best-effort static check suitable for CI).
     9	#
    10	# Outputs a human-readable report to stdout and exits non-zero on mismatch.
    11	# Emits the symbol TOPOLOGY_PARITY_OK on success for gate proof grepping.
    12	set -euo pipefail
    13	
    14	ROOT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/../.." && pwd)
    15	TOPO_YAML=${TOPO_YAML:-"${ROOT_DIR}/lab/topology.clab.yml"}
    16	CONFIGMAP_YAML=${CONFIGMAP_YAML:-"${ROOT_DIR}/deploy/observability/topology-configmap.yaml"}
    17	INSPECT_JSON=${INSPECT_JSON:-}
    18	
    19	usage(){
    20	  cat <<EOF
    21	Usage: $0 [--inspect path.json]
    22	  --inspect path.json  Optional containerlab inspect JSON file to prefer
    23	Environment overrides:
    24	  TOPO_YAML, CONFIGMAP_YAML, INSPECT_JSON
    25	EOF
    26	}
    27	
    28	while [[ $# > 0 ]]; do
    29	  case "$1" in
    30	    --inspect) shift; INSPECT_JSON=${1:-$INSPECT_JSON};;
    31	    -h|--help) usage; exit 0 ;;
    32	    *) echo "[topology-parity] unknown arg: $1" >&2; usage; exit 2 ;;
    33	  esac
    34	  shift || true
    35	done
    36	
    37	# read_nodes_links_from_topology: outputs two lines: "NODES: a b c" and "LINKS: a-b c-d ..."
    38	read_nodes_links_from_topology(){
    39	  local yaml=$1
    40	  # Nodes: capture lines inside "nodes:" mapping that look like "  name:" (spine01:, leaf01:, etc.)
    41	  # Accept alnum/underscore/hyphen. Trim trailing colon.
    42	  local nodes
    43	  nodes=$(awk '
    44	    $0 ~ /^\s*nodes:\s*$/ { in_nodes=1; next }
    45	    in_nodes && $0 ~ /^\s*links:\s*$/ { in_nodes=0 }
    46	    in_nodes && $0 ~ /^\s*[A-Za-z0-9_-]+:\s*$/ { gsub(":",""); gsub(/^\s+/,""); print $0 }
    47	  ' "$yaml" | tr '\n' ' ')
    48	  # Links: find "- endpoints: [a:x, b:y]" -> produce sorted endpoint pair "a-b"
    49	  # Use POSIX awk (no third-arg match), extract text between [ and ] via sub()
    50	  local links
    51	  links=$(awk '
    52	    {
    53	      if ($0 ~ /-\s*endpoints:\s*\[/) {
    54	        s=$0
    55	        sub(/^.*\[/, "", s)
    56	        sub(/\].*$/, "", s)
    57	        # split on comma into two endpoint specs
    58	        n=split(s, parts, ",")
    59	        if (n>=2) {
    60	          p1=parts[1]; p2=parts[2]
    61	          gsub(/^[ ]+|[ ]+$/, "", p1); gsub(/^[ ]+|[ ]+$/, "", p2)
    62	          split(p1, a, ":"); split(p2, b, ":")
    63	          n1=a[1]; n2=b[1]
    64	          if (n1 ~ /^(spine|leaf)[0-9]+$/ && n2 ~ /^(spine|leaf)[0-9]+$/) {
    65	            if (n1 < n2) { print n1 "-" n2 } else { print n2 "-" n1 }
    66	          }
    67	        }
    68	      }
    69	    }
    70	  ' "$yaml" | sort -u | tr '\n' ' ')
    71	  # For nodes, restrict to those participating in fabric links (spine/leaf)
    72	  local nodes_from_links
    73	  nodes_from_links=$(tr ' ' '\n' <<<"$links" | awk -F- '{print $1"\n"$2}' | sort -u | tr '\n' ' ')
    74	  # Fallback when no fabric links parsed: use node list filtered to spine/leaf
    75	  if [[ -z "$nodes_from_links" ]]; then
    76	    nodes_from_links=$(tr ' ' '\n' <<<"$nodes" | grep -E '^(spine|leaf)[0-9]+$' | sort -u | tr '\n' ' ')
    77	  fi
    78	  echo "NODES: $nodes_from_links"
    79	  echo "LINKS: $links"
    80	}
    81	
    82	# read_nodes_links_from_configmap: parse topology.json from the pinned ConfigMap YAML
    83	read_nodes_links_from_configmap(){
    84	  local cm=$1
    85	  # Extract the embedded JSON block and parse with jq if available; else use grep/sed.
    86	  local json
    87	  json=$(awk '/^\s*topology.json: \|/{flag=1; next} /^\S/{flag=0} flag{print substr($0,5)}' "$cm" | tr -d '\r')
    88	  if command -v jq >/dev/null 2>&1; then
    89	    local nodes links
    90	    nodes=$(jq -r '.nodes[].id' <<<"$json" | sort -u | tr '\n' ' ')
    91	    links=$(jq -r '.links[] | [.source, .target] | sort | join("-")' <<<"$json" | sort -u | tr '\n' ' ')
    92	    echo "NODES: $nodes"
    93	    echo "LINKS: $links"
    94	  else
    95	    # Fallback regex parsing for ids and links
    96	    local nodes links
    97	    nodes=$(grep -o '"id"\s*:\s*"[^"]\+"' <<<"$json" | sed -E 's/.*:\s*"([^"]+)"/\1/' | sort -u | tr '\n' ' ')
    98	    links=$(grep -E '"source"|"target"' <<<"$json" | sed -E 's/.*:\s*"([^"]+)".*/\1/' | paste - - \
    99	      | awk '{ a=$1; b=$2; if (a<b) {print a"-"b} else {print b"-"a} }' | sort -u | tr '\n' ' ')
   100	    echo "NODES: $nodes"
   101	    echo "LINKS: $links"
   102	  fi
   103	}
   104	
   105	# Helper to safely populate an array with two lines (NODES:/LINKS:) even when parsing fails
   106	_safe_read_topology_into_array(){
   107	  local -n _out=$1
   108	  local yaml=$2
   109	  local t1 t2
   110	  t1="NODES: "; t2="LINKS: "
   111	  # Use a subshell to avoid set -e/ou effect on array assignment
   112	  local tmp_out
   113	  tmp_out=$(read_nodes_links_from_topology "$yaml" 2>/dev/null || true)
   114	  if [[ -n "$tmp_out" ]]; then
   115	    # split into two lines
   116	    t1=$(printf "%s\n" "$tmp_out" | sed -n '1p')
   117	    t2=$(printf "%s\n" "$tmp_out" | sed -n '2p')
   118	  fi
   119	  _out=("$t1" "$t2")
   120	}
   121	
   122	# If inspect JSON provided, prefer it to derive expected nodes/links
   123	if [[ -n "${INSPECT_JSON:-}" && -f "$INSPECT_JSON" ]]; then
   124	  echo "[topology-parity] INFO: using inspect JSON: $INSPECT_JSON" >&2
   125	  # Expect JSON structure: .topology.nodes[].name and .topology.links[].a.node/.b.node
   126	  if command -v jq >/dev/null 2>&1; then
   127	    exp_nodes=$(jq -r '.topology.nodes[].name' "$INSPECT_JSON" | sort -u | tr '\n' ' ')
   128	    exp_links=$(jq -r '.topology.links[] | [.a.node, .b.node] | sort | join("-")' "$INSPECT_JSON" | sort -u | tr '\n' ' ')
   129	    echo "EXPECTED NODES: $exp_nodes"
   130	    echo "EXPECTED LINKS: $exp_links"
   131	  else
   132	    echo "[topology-parity] WARN: jq not available; falling back to YAML parsing" >&2
   133	    arr=("NODES: " "LINKS: ")
   134	    _safe_read_topology_into_array arr "$TOPO_YAML"
   135	    exp_nodes=${arr[0]#NODES: } ; exp_links=${arr[1]#LINKS: }
   136	  fi
   137	else
   138	  arr=("NODES: " "LINKS: ")
   139	  _safe_read_topology_into_array arr "$TOPO_YAML"
   140	  exp_nodes=${arr[0]#NODES: } ; exp_links=${arr[1]#LINKS: }
   141	  echo "EXPECTED NODES: $exp_nodes"
   142	  echo "EXPECTED LINKS: $exp_links"
   143	fi
   144	
   145	mapfile -t cmarr < <(read_nodes_links_from_configmap "$CONFIGMAP_YAML")
   146	cm_nodes=${cmarr[0]#NODES: } ; cm_links=${cmarr[1]#LINKS: }
   147	
   148	declare -A expn cmn
   149	mismatch=0
   150	for n in $exp_nodes; do expn[$n]=1; done
   151	for n in $cm_nodes; do cmn[$n]=1; done
   152	
   153	# Compare node sets
   154	for n in "${!expn[@]}"; do if [[ -z "${cmn[$n]:-}" ]]; then echo "MISSING NODE in ConfigMap: $n"; mismatch=1; fi; done
   155	for n in "${!cmn[@]}"; do if [[ -z "${expn[$n]:-}" ]]; then echo "EXTRA NODE in ConfigMap: $n"; mismatch=1; fi; done
   156	
   157	# Compare link sets
   158	declare -A expl cml
   159	for e in $exp_links; do expl[$e]=1; done
   160	for e in $cm_links; do cml[$e]=1; done
   161	for e in "${!expl[@]}"; do if [[ -z "${cml[$e]:-}" ]]; then echo "MISSING LINK in ConfigMap: $e"; mismatch=1; fi; done
   162	for e in "${!cml[@]}"; do if [[ -z "${expl[$e]:-}" ]]; then echo "EXTRA LINK in ConfigMap: $e"; mismatch=1; fi; done
   163	
   164	exp_node_count=$(wc -w <<<"$exp_nodes" | tr -d ' ')
   165	cm_node_count=$(wc -w <<<"$cm_nodes" | tr -d ' ')
   166	exp_link_count=$(wc -w <<<"$exp_links" | tr -d ' ')
   167	cm_link_count=$(wc -w <<<"$cm_links" | tr -d ' ')
   168	
   169	echo "COUNTS: expected nodes=$exp_node_count, configmap nodes=$cm_node_count; expected links=$exp_link_count, configmap links=$cm_link_count"
   170	
   171	if [[ $mismatch -eq 0 ]]; then
   172	  echo "TOPOLOGY_PARITY_OK: nodes and links match between lab/topology.clab.yml and deploy/observability/topology-configmap.yaml"
   173	else
   174	  echo "TOPOLOGY_PARITY_FAIL: mismatch detected" >&2
   175	  exit 1
   176	fi
