    23	kind::recover_partial() {
    24	  # If cluster is listed but nodes are missing or unhealthy, delete and recreate
    25	  local nodes; nodes=$(kind::nodes)
    26	  if [[ -n "${nodes}" ]]; then
    27	    # Expect at least a control-plane node
    28	    if ! grep -q "${KIND_CLUSTER_NAME}-control-plane" <<<"${nodes}"; then
    29	      echo "[kind] partial cluster detected; deleting for recovery" >&2
    30	      kind delete cluster --name "${KIND_CLUSTER_NAME}" || true
    31	    fi
    32	  fi
    33	}
