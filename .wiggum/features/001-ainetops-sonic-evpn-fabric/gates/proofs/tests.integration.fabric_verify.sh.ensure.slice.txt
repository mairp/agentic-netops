    26	  local need_fetch=0

    28	  if [[ -z "${GNMI_USER:-}" || -z "${GNMI_PASS:-}" ]]; then need_fetch=1; fi

    31	  if ! command -v kubectl >/dev/null 2>&1; then
    32	    echo "[fabric-verify] SKIP: kubectl not available and local secrets missing — skipping fabric verification in CI/minimal environment" >&2
    33	    return 64 # special code meaning prereq-missing
    34	  fi
    35	
    36	  echo "[fabric-verify] fetching lab credentials/TLS from cluster secrets into ./secrets"
    37	  mkdir -p ./secrets
    38	  # Credentials
    39	  if kubectl --context "$KUBE_CTX" -n ainetops-system get secret gnmi-lab-creds >/dev/null 2>&1; then
    40	    GNMI_USER=$(kubectl --context "$KUBE_CTX" -n ainetops-system get secret gnmi-lab-creds -o jsonpath='{.data.username}' | base64 -d || true)
    41	    GNMI_PASS=$(kubectl --context "$KUBE_CTX" -n ainetops-system get secret gnmi-lab-creds -o jsonpath='{.data.password}' | base64 -d || true)
    42	    export GNMI_USER GNMI_PASS
    43	  else
    44	    echo "[fabric-verify] WARN: secret gnmi-lab-creds not found" >&2
    45	  fi
    46	  # TLS bundle
    47	  if kubectl --context "$KUBE_CTX" -n ainetops-system get secret gnmi-lab-tls >/dev/null 2>&1; then
    48	    kubectl --context "$KUBE_CTX" -n ainetops-system get secret gnmi-lab-tls -o jsonpath='{.data.ca\.crt}' | base64 -d > "$GNMI_CACERT" || true
    49	    kubectl --context "$KUBE_CTX" -n ainetops-system get secret gnmi-lab-tls -o jsonpath='{.data.tls\.crt}' | base64 -d > "$GNMI_CERT" || true
    50	    kubectl --context "$KUBE_CTX" -n ainetops-system get secret gnmi-lab-tls -o jsonpath='{.data.tls\.key}' | base64 -d > "$GNMI_KEY" || true
    51	  else
    52	    echo "[fabric-verify] WARN: secret gnmi-lab-tls not found" >&2
    53	  fi
    54	
    55	  # If still missing after first fetch attempt, proactively create via RBAC manifests and generator Job.
    56	  if { [[ ! -f "$GNMI_CACERT" || ! -f "$GNMI_CERT" || ! -f "$GNMI_KEY" ]] || [[ -z "${GNMI_USER:-}" || -z "${GNMI_PASS:-}" ]]; }; then
