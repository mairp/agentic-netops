    22	// TestProviderFinalization simulates deletion flow and ensures ordered finalization evidence.
    23	func TestProviderFinalization(t *testing.T) {
    24		scheme := runtime.NewScheme()
    25		_ = clientgoscheme.AddToScheme(scheme)
    26		_ = kubenet.AddToScheme(scheme)
    27		_ = sdc.AddToScheme(scheme)
    28		c := fake.NewClientBuilder().WithScheme(scheme).Build()
    29		rec := &sonicprovider.Reconciler{Client: c, Scheme: scheme}
    30	
    31		ctx := context.TODO()
    32		nd := &kubenet.NetworkDevice{ObjectMeta: metav1.ObjectMeta{Name: "leaf01", Namespace: "default"}}
    33		if err := c.Create(ctx, nd); err != nil { t.Fatalf("create nd: %v", err) }
    34		// Add finalizer and a Config owned by ND
    35		if err := c.Get(ctx, clientKey(nd), nd); err != nil { t.Fatalf("get nd: %v", err) }
    36		nd.Finalizers = append(nd.Finalizers, "ainetops.dev/finalizer")
    37		if err := c.Update(ctx, nd); err != nil { t.Fatalf("update nd: %v", err) }
    38		cfg := &sdc.Config{ObjectMeta: metav1.ObjectMeta{Name: "nd-leaf01", Namespace: "default"}}
    39		if err := c.Create(ctx, cfg); err != nil { t.Fatalf("create cfg: %v", err) }
    40		// Mark ND as deleting
    41		if err := c.Get(ctx, clientKey(nd), nd); err != nil { t.Fatalf("get nd: %v", err) }
    42		now := metav1.NewTime(time.Now())
    43		nd.DeletionTimestamp = &now
    44		if err := c.Update(ctx, nd); err != nil { t.Fatalf("update nd deleting: %v", err) }
    45	
    46		// First reconcile should issue delete and requeue (Config still exists)
    47		if _, err := rec.Reconcile(ctx, ctrl.Request{NamespacedName: types.NamespacedName{Namespace: "default", Name: "leaf01"}}); err != nil { t.Fatalf("reconcile: %v", err) }
    48		// Delete the Config to simulate confirmation
    49		if err := c.Delete(ctx, cfg); err != nil { t.Fatalf("delete cfg: %v", err) }
    50		// Second reconcile should record finalized-at and remove finalizer
    51		if _, err := rec.Reconcile(ctx, ctrl.Request{NamespacedName: types.NamespacedName{Namespace: "default", Name: "leaf01"}}); err != nil { t.Fatalf("reconcile2: %v", err) }
    52		if err := c.Get(ctx, clientKey(nd), nd); err != nil { t.Fatalf("get nd2: %v", err) }
    53		if nd.Annotations["ainetops.dev/finalized-at"] == "" { t.Fatalf("expected finalized-at annotation") }
    54		if contains(nd.Finalizers, "ainetops.dev/finalizer") { t.Fatalf("expected finalizer removed") }
    55	}
    56	
    57	func clientKey(nd *kubenet.NetworkDevice) clientKeyObj { return clientKeyObj{namespace: nd.Namespace, name: nd.Name} }
    58	
    59	type clientKeyObj struct{ namespace, name string }
    60	
    61	func contains(ss []string, s string) bool { for _, v := range ss { if v == s { return true } }; return false }
