     1	#!/usr/bin/env bash
     2	# Run the same deny-list policy as CI locally (T074a)
     3	set -euo pipefail
     4	ROOT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/../.." && pwd)
     5	PROOFS_DIR="$ROOT_DIR/.wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs"
     6	mkdir -p "$PROOFS_DIR"
     7	export PATH="$PATH:/usr/local/bin"
     8	
     9	# Prefer shared policy script (kept in-repo for stability)
    10	if [[ -x "$ROOT_DIR/scripts/ci/denylist_policy.sh" ]]; then
    11	  "$ROOT_DIR/scripts/ci/denylist_policy.sh"
    12	  exit $?
    13	fi
    14	
    15	# Fallback: extract the workflow run body
    16	awk '/^\s*run: \|/{f=1;next} f{print} ' "$ROOT_DIR/.github/workflows/denylist.yml" \
    17	  | bash -euo pipefail 2>&1 | tee "$PROOFS_DIR/denylist.run.log"
