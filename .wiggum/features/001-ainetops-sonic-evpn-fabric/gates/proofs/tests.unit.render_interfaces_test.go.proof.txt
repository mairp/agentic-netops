    10	func TestRenderInterfaces_OrderStable(t *testing.T) {
    11		ifs := []model.Interface{{Name: "Ethernet2"}, {Name: "Ethernet1", MTU: 9000}}
    12		res := render.RenderInterfaces(ifs, nil)
    13		arr, ok := res["/interfaces/interface"].([]map[string]any)
    14		if !ok { t.Fatalf("unexpected type") }
    15		if len(arr) != 2 { t.Fatalf("expected 2 entries") }
    16		if arr[0]["name"] != "Ethernet1" { t.Fatalf("expected Ethernet1 first, got %v", arr[0]["name"]) }
    17	}
