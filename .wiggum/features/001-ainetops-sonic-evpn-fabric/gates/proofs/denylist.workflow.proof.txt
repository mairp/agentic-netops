     1	name: Denylist Scan
     2	
     3	on:
     4	  push:
     5	    branches: [ main, master ]
     6	  pull_request:
     7	
     8	jobs:
     9	  denylist:
    10	    runs-on: ubuntu-latest
    11	    steps:
    12	      - name: Checkout
    13	        uses: actions/checkout@v4
    14	
    15	      - name: Install ripgrep and coreutils
    16	        run: |
    17	          sudo apt-get update
    18	          sudo apt-get install -y ripgrep coreutils
    19	
    20	      - name: Run repository-wide deny-list checks (SC-010, FR-020, FR-023, FR-032)
    21	        shell: bash
    22	        run: |
    23	          set -euo pipefail
    24	
    25	          fail=0
    26	
    27	          # Helper: filter allowed contexts from ripgrep matches
    28	          filter_allowed() {
    29	            local matches="$1"
    30	            # Allow citations in research.md and REVERSE.md
    31	            matches=$(printf "%s" "${matches}" | rg -v 'specs/.+/research\.md:' | rg -v '^REVERSE\.md:' || true)
    32	
    33	            # Allow mentions in spec.md "Scope and interpretation" section only
    34	            # Compute line range for the section in the pinned spec
    35	            local spec_path="specs/001-ainetops-sonic-evpn-fabric/spec.md"
    36	            if [[ -f "$spec_path" ]]; then
    37	              local start end
    38	              start=$(awk '/^##\s+Scope and interpretation/{print NR; exit}' "$spec_path" || true)
    39	              end=$(awk 'p && /^##\s+/{print NR; exit} /^##\s+Scope and interpretation/{p=1}' "$spec_path" || true)
    40	              if [[ -n "${start}" && -n "${end}" ]]; then
    41	                # Remove matches within allowed line range in spec.md
    42	                matches=$(printf "%s" "${matches}" | awk -v s="$start" -v e="$end" -F: '!( $1=="'"$spec_path"'" && $2>=s && $2<=e )')
    43	              fi
    44	            fi
    45	
    46	            # Allow mention of srl-telemetry-lab as a visualization reference only
    47	            matches=$(printf "%s" "${matches}" | rg -v 'srl-telemetry-lab' || true)
    48	
    49	            printf "%s" "${matches}"
    50	          }
    51	
    52	          echo "::group::Migration boundary terms (FR-020)"
    53	          MIG_PATTERN='\\b(cisco|crosswork|nso|cnc|proprietary\\s+ned|proprietary\\s+neds|ai-network-services-devnet-2606|devnet-2606)\\b'
    54	          MIG_MATCHES=$(rg -i -n --hidden --no-ignore -g '!*/.git/*' -e "$MIG_PATTERN" . || true)
    55	          MIG_VIOLATIONS=$(filter_allowed "$MIG_MATCHES")
    56	          if [[ -n "$MIG_VIOLATIONS" ]]; then
    57	            echo "::error title=Migration boundary term(s) found outside allowed contexts::${MIG_VIOLATIONS}"
    58	            fail=1
    59	          fi
    60	          echo "::endgroup::"
    61	
    62	          echo "::group::Visualization boundary terms (FR-032)"
    63	          VIS_PATTERN='\\b(sr[ -]?linux|nokia_srlinux)\\b|ghcr\\.io/nokia/srlinux'
    64	          VIS_MATCHES=$(rg -i -n --hidden --no-ignore -g '!*/.git/*' -e "$VIS_PATTERN" . || true)
    65	          VIS_VIOLATIONS=$(filter_allowed "$VIS_MATCHES")
    66	          if [[ -n "$VIS_VIOLATIONS" ]]; then
    67	            echo "::error title=SR Linux mention(s) found outside allowed contexts::${VIS_VIOLATIONS}"
    68	            fail=1
    69	          fi
    70	          echo "::endgroup::"
    71	
    72	          echo "::group::Placement boundary (FR-023): Compose/standalone indicators"
    73	          PL_PATTERN='(docker-compose|docker\\s+compose|compose\\.ya?ml|standalone\\s+container|standalone\\s+deployment)'
    74	          PL_MATCHES=$(rg -i -n --hidden --no-ignore -g '!*/.git/*' -e "$PL_PATTERN" . || true)
    75	          PL_VIOLATIONS=$(filter_allowed "$PL_MATCHES")
    76	          if [[ -n "$PL_VIOLATIONS" ]]; then
    77	            echo "::error title=Disallowed placement mention(s) found outside allowed contexts::${PL_VIOLATIONS}"
    78	            fail=1
    79	          fi
    80	          echo "::endgroup::"
    81	
    82	          if [[ "$fail" -ne 0 ]]; then
    83	            echo "Deny-list checks failed" >&2
    84	            exit 1
    85	          fi
    86	          echo "All deny-list checks passed"
