    21	// TestSRv6ServiceCRD_Envtest installs the CRD into a test API server and validates sample CRs
    22	// using server-side dry-run (positive and negative cases).
    23	func TestSRv6ServiceCRD_Envtest(t *testing.T) {
    24		t.Parallel()
    25		var (
    26			ctx = context.TODO()
    27		)
    28	
    29		// Determine envtest binaries; skip test when not available in CI.
    30		assets := os.Getenv("KUBEBUILDER_ASSETS")
    31		if assets == "" {
    32			// common default in many images
    33			if _, err := os.Stat("/usr/local/kubebuilder/bin/etcd"); err == nil {
    34				assets = "/usr/local/kubebuilder/bin"
    35			}
    36		}
    37		if assets == "" {
    38			t.Skip("envtest assets not found; set KUBEBUILDER_ASSETS to run this test")
    39		}
    40		// Start the envtest API server
    41		testEnv := &envtest.Environment{
    42			CRDDirectoryPaths: []string{
    43				filepath.Join("..", "..", "config", "crd", "bases"),
    44			},
    45			BinaryAssetsDirectory: assets,
    46		}
    47		cfg, err := testEnv.Start()
    48		if err != nil {
    49			t.Fatalf("failed to start envtest: %v", err)
    50		}
    51		t.Cleanup(func() {
    52			_ = testEnv.Stop()
    53		})
    54	
    55		// Ensure apiextensions scheme is registered for dry-run operations
    56		_ = apiextensionsv1.AddToScheme(runtime.NewScheme())
    57	
    58		dc, err := dynamic.NewForConfig(cfg)
    59		if err != nil {
    60			t.Fatalf("dynamic client: %v", err)
    61		}
    62	
    63		gvr := schemaFor("ainetops.io", "v1alpha1", "srv6services")
    64	
    65		// Positive sample
    66		posYAML, err := os.ReadFile(filepath.Join("..", "..", "config", "samples", "ainetops_v1alpha1_srv6service.yaml"))
    67		if err != nil {
    68			t.Fatalf("read sample: %v", err)
    69		}
    70		posObj := mustYAMLToUnstructured(t, posYAML)
    71		posObj.SetNamespace("default")
    72		// Dry-run create should succeed
    73		if _, err := dc.Resource(gvr).Namespace("default").Create(ctx, posObj, metav1.CreateOptions{DryRun: []string{"All"}}); err != nil {
    74			t.Fatalf("dry-run create positive sample failed: %v", err)
    75		}
    76	
    77		// Negative sample: duplicate attachments (violates CEL uniqueness rule)
    78		neg := posObj.DeepCopy()
    79		att, _, _ := unstructured.NestedSlice(neg.Object, "spec", "attachments")
    80		if len(att) != 2 {
    81			t.Fatalf("unexpected attachments len in sample: %d", len(att))
    82		}
    83		// make both nodes equal to violate uniqueness
    84		_ = unstructured.SetNestedField(att[1].(map[string]any), att[0].(map[string]any)["node"], "node")
    85		_ = unstructured.SetNestedSlice(neg.Object, att, "spec", "attachments")
    86		if _, err := dc.Resource(gvr).Namespace("default").Create(ctx, neg, metav1.CreateOptions{DryRun: []string{"All"}}); err == nil {
    87			t.Fatalf("expected dry-run create to fail for duplicate attachments, got nil error")
    88		}
    89	}

    91	func mustYAMLToUnstructured(t *testing.T, y []byte) *unstructured.Unstructured {
    92		t.Helper()
    93		dec := yaml.NewDecodingSerializer(unstructured.UnstructuredJSONScheme)
    94		obj := &unstructured.Unstructured{}
    95		_, _, err := dec.Decode(y, nil, obj)
    96		if err != nil {
    97			t.Fatalf("decode yaml: %v", err)
    98		}
    99		return obj
   100	}

   102	func schemaFor(group, version, resource string) schema.GroupVersionResource {
   103		return schema.GroupVersionResource{Group: group, Version: version, Resource: resource}
