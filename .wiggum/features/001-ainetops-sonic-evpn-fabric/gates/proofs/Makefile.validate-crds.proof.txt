     1	SHELL := /usr/bin/env bash
     2	.ONESHELL:
     3	.SHELLFLAGS := -eu -o pipefail -c
     4	
     5	.PHONY: help verify-pins validate-crds verify-compat
     6	
     7	help:
     8		@echo "Targets:"
     9		@echo "  verify-pins     Validate versions.lock.yaml has immutable pins and consistency"
    10		@echo "  validate-crds   Server-side dry-run validation of Kubenet/KUID/SDC CRDs and examples"
    11		@echo "  verify-compat   Run verify-pins and validate-crds together"
    12	
    13	verify-pins:
    14		@echo "[verify-pins] validating versions.lock.yaml"
    15		@"$(PWD)/scripts/lib/verify_pins.sh"
    16	
    17	validate-crds:
    18		@echo "[validate-crds] server-side validating CRDs and examples"
    19		@mkdir -p .wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs
    20		@"$(PWD)/scripts/lib/validate_crds.sh" 2>&1 | tee .wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs/validate-crds.run.log
    21	
    22	verify-compat: verify-pins validate-crds
    23		@echo "[verify-compat] pins and CRD validations passed"
