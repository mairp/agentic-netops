    22	// ServiceType enumerates supported normalized service kinds.
    23	type ServiceType string
    24	
    25	const (
    26		ServiceVPLS  ServiceType = "VPLS"         // multipoint L2VPN
    27		ServiceL3VPN ServiceType = "L3VPN"        // routed VPN
    28		ServiceVPWS ServiceType = "VPWS"          // E-Line
    29		ServiceIRB  ServiceType = "L2L3-IRB"      // integrated L2/L3 (symmetric-IRB)
    30	)

    32	// AddressFamilies captures AFI-specific properties for L3VPN.
    33	type AddressFamilies struct {
    34		IPv4Prefixes []string `json:"ipv4Prefixes,omitempty"`
    35		IPv6Prefixes []string `json:"ipv6Prefixes,omitempty"`
    36	}
    37	
    38	// RdRt are required route distinguisher and route-targets for VPNs.
    39	type RdRt struct {
    40		RD       string   `json:"rd"`
    41		ImportRT []string `json:"importRT"`
    42		ExportRT []string `json:"exportRT"`
    43	}
    44	
    45	// Endpoint describes one attachment point on a target node with VLAN or VRF context.
    46	type Endpoint struct {
    47		Node       string `json:"node"`                 // required leaf name
    48		Attachment string `json:"attachment"`          // human-readable peer/endpoint id
    49		VLAN       int    `json:"vlan,omitempty"`      // for L2 services
    50		VRF        string `json:"vrf,omitempty"`       // for L3/IRB attachments
    51	}
    52	
    53	// IRBGateway parameters for symmetric-IRB.
    54	type IRBGateway struct {
    55		VRF        string `json:"vrf"`
    56		GatewayV4  string `json:"gatewayIPv4"`
    57		GatewayV6  string `json:"gatewayIPv6"`
    58	}
    59	
    60	// Policies are explicit allow-listed options.
    61	type Policies struct {
    62		// VPWSLimitedEquivalence must be true to allow VPWS-to-L2VNI mapping.
    63		VPWSLimitedEquivalence bool `json:"vpwsLimitedEquivalence,omitempty"`
    64	}
    65	
    66	// UnsupportedClaims declare source-only features that must be rejected (FR-011).
    67	// Presence of any field here causes a terminal validation failure with details.
    68	type UnsupportedClaims struct {
    69		TEPolicy         any `json:"tePolicy,omitempty"`
    70		PseudowireOAM    any `json:"pseudowireOAM,omitempty"`
    71		ControlWord      any `json:"controlWord,omitempty"`
    72		MulticastVPN     any `json:"multicastVPN,omitempty"`
    73		ComplexQoS       any `json:"complexQoS,omitempty"`
    74		ServiceChain     any `json:"serviceChain,omitempty"`
    75		RawCLI           any `json:"rawCLI,omitempty"`
    76	}

    78	// ServiceInput is the strict normalized migration input schema.
    79	// Raw CLI is explicitly forbidden; unknown fields cause validation failure.
    80	// Deterministic translation requires RD/RT and VNI inputs to be explicit.
    81	type ServiceInput struct {
    82		// Identity
    83		ServiceID string      `json:"serviceId"`
    84		Type      ServiceType `json:"type"`      // VPLS | L3VPN | VPWS | L2L3-IRB
    85		Tenant    string      `json:"tenant"`
    86	
    87		// VPN properties
    88		RDRT   *RdRt            `json:"rdRt,omitempty"`
    89		L2VNI  int              `json:"l2vni,omitempty"` // for VPLS/VPWS/IRB bridge domains
    90		L3VNI  int              `json:"l3vni,omitempty"` // for L3VPN/IRB VRFs
    91		AF     *AddressFamilies `json:"addressFamilies,omitempty"`
    92	
    93		// IRB per-BD gateways (when Type == L2L3-IRB)
    94		IRBGateway *IRBGateway `json:"irbGateway,omitempty"`
    95	
    96		// Endpoints
    97		Endpoints []Endpoint `json:"endpoints"`
    98	
    99		// Explicit policy opt-ins
   100		Policies Policies `json:"policies,omitempty"`
   101	
   102		// Explicitly modeled unsupported features — any presence rejects the request.
   103		Unsupported UnsupportedClaims `json:"unsupported,omitempty"`
   104	}

   106	// CanonicalHash returns a deterministic SHA256 over the normalized input.
   107	func (in *ServiceInput) CanonicalHash() (string, error) {
   108		if in == nil {
   109			return "", errors.New("nil input")
   110		}
   111		// Marshal the typed struct (no maps) to stable JSON and hash it.
   112		b, err := json.Marshal(in)
   113		if err != nil {
   114			return "", err
   115		}
   116		sum := sha256.Sum256(b)
   117		return hex.EncodeToString(sum[:]), nil
   118	}

   120	// ValidateAllOrNothing enforces the FR-011 contract and returns an error with
   121	// structured causes; it does not modify input and does not perform allocation.
   122	type ValidationError struct {
   123		Causes []string `json:"causes"`
   124	}
   125	
   126	func (e *ValidationError) Error() string {
   127		return fmt.Sprintf("validation failed: %v", e.Causes)
   128	}
   129	
   130	// ValidateAllOrNothing performs strict validation, including:
   131	// - required fields by type;
   132	// - explicit opt-ins for limited equivalence;
   133	// - absent endpoints and count mismatches;
   134	// - presence of unsupported claims;
   135	// - collisions detected by the caller across a batch (detected via index arg).
   136	func (in *ServiceInput) ValidateAllOrNothing(batchIndex int, dupServiceID bool) error {
   137		var causes []string
   138		if in.ServiceID == "" {
   139			causes = append(causes, "serviceId: required")
   140		}
   141		if in.Tenant == "" {
   142			causes = append(causes, "tenant: required")
   143		}
   144		if in.Type == "" {
   145			causes = append(causes, "type: required")
   146		}
   147		if dupServiceID {
   148			causes = append(causes, fmt.Sprintf("collision: duplicate serviceId '%s' in batch", in.ServiceID))
   149		}
   150		// Unsupported claims
   151		u := in.Unsupported
   152		if u.TEPolicy != nil { causes = append(causes, "unsupported: tePolicy") }
   153		if u.PseudowireOAM != nil { causes = append(causes, "unsupported: pseudowireOAM") }
   154		if u.ControlWord != nil { causes = append(causes, "unsupported: controlWord") }
   155		if u.MulticastVPN != nil { causes = append(causes, "unsupported: multicastVPN") }
   156		if u.ComplexQoS != nil { causes = append(causes, "unsupported: complexQoS") }
   157		if u.ServiceChain != nil { causes = append(causes, "unsupported: serviceChain") }
   158		if u.RawCLI != nil { causes = append(causes, "unsupported: rawCLI") }
   159	
   160		// Endpoints basics
   161		if len(in.Endpoints) == 0 { causes = append(causes, "endpoints: at least one endpoint is required") }
   162	
   163		switch in.Type {
   164		case ServiceVPLS:
   165			if in.L2VNI == 0 { causes = append(causes, "l2vni: required for VPLS") }
   166			if in.RDRT == nil { causes = append(causes, "rdRt: required for VPLS") }
   167			if len(in.Endpoints) < 2 { causes = append(causes, "endpoints: VPLS requires >=2 endpoints") }
   168			// All endpoints must specify same VLAN
   169			var vlan int
   170			for i, ep := range in.Endpoints {
   171				if ep.VLAN == 0 { causes = append(causes, fmt.Sprintf("endpoints[%d].vlan: required for VPLS", i)) }
   172				if i == 0 { vlan = ep.VLAN } else if ep.VLAN != vlan { causes = append(causes, "endpoints.vlan: must be equal across all endpoints for VPLS") }
   173			}
   174		case ServiceL3VPN:
   175			if in.L3VNI == 0 { causes = append(causes, "l3vni: required for L3VPN") }
   176			if in.RDRT == nil { causes = append(causes, "rdRt: required for L3VPN") }
   177			if in.AF == nil || (len(in.AF.IPv4Prefixes) == 0 && len(in.AF.IPv6Prefixes) == 0) {
   178				causes = append(causes, "addressFamilies: at least one prefix is required for L3VPN")
   179			}
   180			for i, ep := range in.Endpoints {
   181				if ep.VRF == "" { causes = append(causes, fmt.Sprintf("endpoints[%d].vrf: required for L3VPN", i)) }
   182			}
   183		case ServiceVPWS:
   184			if in.L2VNI == 0 { causes = append(causes, "l2vni: required for VPWS") }
   185			if in.RDRT == nil { causes = append(causes, "rdRt: required for VPWS") }
   186			if len(in.Endpoints) != 2 { causes = append(causes, "endpoints: VPWS requires exactly 2 endpoints") }
   187			for i, ep := range in.Endpoints {
   188				if ep.VLAN == 0 { causes = append(causes, fmt.Sprintf("endpoints[%d].vlan: required for VPWS", i)) }
   189			}
   190			if !in.Policies.VPWSLimitedEquivalence {
   191				causes = append(causes, "policy: vpwsLimitedEquivalence must be true to allow limited equivalence mapping")
   192			}
   193		case ServiceIRB:
   194			if in.L2VNI == 0 { causes = append(causes, "l2vni: required for IRB bridge domain") }
   195			if in.L3VNI == 0 { causes = append(causes, "l3vni: required for IRB VRF") }
   196			if in.RDRT == nil { causes = append(causes, "rdRt: required for IRB VRF") }
   197			if in.IRBGateway == nil { causes = append(causes, "irbGateway: required for IRB") }
   198			if in.IRBGateway != nil {
   199				if in.IRBGateway.VRF == "" { causes = append(causes, "irbGateway.vrf: required") }
   200				if in.IRBGateway.GatewayV4 == "" { causes = append(causes, "irbGateway.gatewayIPv4: required") }
   201				if in.IRBGateway.GatewayV6 == "" { causes = append(causes, "irbGateway.gatewayIPv6: required") }
   202			}
   203			if len(in.Endpoints) < 1 { causes = append(causes, "endpoints: IRB requires at least one endpoint") }
   204			for i, ep := range in.Endpoints {
   205				if ep.VLAN == 0 { causes = append(causes, fmt.Sprintf("endpoints[%d].vlan: required for IRB", i)) }
   206			}
   207		default:
   208			causes = append(causes, fmt.Sprintf("type: unsupported '%s'", in.Type))
   209		}
   210	
   211		if len(causes) > 0 {
   212			return &ValidationError{Causes: causes}
   213		}
   214		return nil
   215	}
