     1	#!/usr/bin/env bash
     2	# Central deny-list policy (T074a) used by CI and local runs
     3	# Enforces migration (FR-020), visualization (FR-032), and placement (FR-023) boundaries
     4	# Allowed contexts only: spec.md Scope and interpretation and SC-010; research.md and REVERSE.md citations;
     5	# README presentation-only mention of srl-telemetry-lab. Fails build on any match outside allowed contexts.
     6	set -euo pipefail
     7	
     8	ROOT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")/../.." && pwd)
     9	cd "$ROOT_DIR"
    10	
    11	fail=0
    12	
    13	# Helper: filter allowed contexts from search matches (works with rg or grep output)
    14	filter_allowed() {
    15	  local matches="$1"
    16	  # Normalize leading ./ to ease anchored filters
    17	  matches=$(printf "%s" "${matches}" | sed -E 's#^\./##')
    18	  # Allow citations in research.md and REVERSE.md
    19	  matches=$(printf "%s" "${matches}" | neg_filter '^specs/.+/research\.md:' | neg_filter '(^|.*/)REVERSE\.md:' || true)
    20	  # Exclude enforcement/policy sources to avoid self-reference false positives
    21	  matches=$(printf "%s" "${matches}" \
    22	    | neg_filter '^\.github/workflows/denylist\.yml:' \
    23	    | neg_filter '^scripts/ci/denylist_policy\.sh:' \
    24	    | neg_filter '^scripts/ci/supply_chain\.sh:' \
    25	    | neg_filter '^scripts/ci/denylist_runtime_scan\.sh:' \
    26	    | neg_filter '^tests/integration/cycles_runner\.sh:' || true)
    27	
    28	  # Allow mentions in spec.md "Scope and interpretation" section and SC-010 success criterion only
    29	  local spec_path="specs/001-ainetops-sonic-evpn-fabric/spec.md"
    30	  if [[ -f "$spec_path" ]]; then
    31	    local start end sc010_start sc010_end
    32	    start=$(awk '/^##\s+Scope and interpretation/{print NR; exit}' "$spec_path" || true)
    33	    end=$(awk 'p && /^##\s+/{print NR; exit} /^##\s+Scope and interpretation/{p=1}' "$spec_path" || true)
    34	    sc010_start=$(awk '/^- \*\*SC-010\*\*/{print NR; exit}' "$spec_path" || true)
    35	    sc010_end=$(awk -v s="$sc010_start" 'NR > s && /^- \*\*SC-/{print NR; exit}' "$spec_path" || true)
    36	    if [[ -z "$sc010_end" ]]; then sc010_end=999999; fi
    37	    if [[ -n "${start}" && -n "${end}" ]]; then
    38	      matches=$(printf "%s" "${matches}" | awk -v s="$start" -v e="$end" -v s2="$sc010_start" -v e2="$sc010_end" -F: '!( $1=="'"$spec_path"'" && ( ($2>=s && $2<=e) || ($2>=s2 && $2<=e2) ) )')
    39	    fi
    40	  fi
    41	
    42	  printf "%s" "${matches}"
    43	}
    44	
    45	# Prefer ripgrep when present; otherwise fall back to grep -RinE
    46	have_rg=0
    47	if command -v rg >/dev/null 2>&1; then have_rg=1; fi
    48	
    49	search_all() {
    50	  local pattern=$1
    51	  # Scan only tracked repository files to avoid session artifacts under .wiggum/**/attempts/**
    52	  # and other untracked logs; this enforces the policy over the source tree.
    53	  if command -v git >/dev/null 2>&1 && git rev-parse --is-inside-work-tree >/dev/null 2>&1; then
    54	    local files
    55	    # Use nul-delimited to handle spaces; prefer ripgrep when present
    56	    if [[ $have_rg -eq 1 ]]; then
    57	      # Use ripgrep with explicit excludes to avoid vendor/, .wiggum/, and the policy files themselves
    58	      rg -i -n --hidden --no-ignore \
    59	        -g '!*/.git/*' -g '!vendor/**' -g '!.wiggum/**' -g '!scripts/ci/**' \
    60	        -g '!.github/workflows/denylist.yml' -g '!scripts/ci/denylist_policy.sh' \
    61	        -e "$pattern" . || true
    62	    else
    63	      # grep fallback with excludes
    64	      grep -RinE --exclude-dir=.git --exclude-dir=vendor --exclude-dir=.wiggum \
    65	        --exclude='.github/workflows/denylist.yml' --exclude='scripts/ci/denylist_policy.sh' \
    66	        -I "$pattern" . || true
    67	    fi
    68	  else
    69	    # Fallback: whole tree except .git
    70	    if [[ $have_rg -eq 1 ]]; then
    71	      rg -i -n --hidden --no-ignore \
    72	        -g '!*/.git/*' -g '!vendor/**' -g '!.wiggum/**' -g '!scripts/ci/**' \
    73	        -g '!.github/workflows/denylist.yml' -g '!scripts/ci/denylist_policy.sh' \
    74	        -e "$pattern" . || true
    75	    else
    76	      grep -RinE --exclude-dir=.git --exclude-dir=vendor --exclude-dir=.wiggum \
    77	        --exclude='.github/workflows/denylist.yml' --exclude='scripts/ci/denylist_policy.sh' \
    78	        -I "$pattern" . || true
    79	    fi
    80	  fi
    81	}
    82	
    83	neg_filter() {
    84	  local pattern=$1
    85	  if [[ $have_rg -eq 1 ]]; then
    86	    rg -v "$pattern"
    87	  else
    88	    grep -vE "$pattern"
    89	  fi
    90	}
    91	
    92	# Output path for proofs
    93	PROOFS_DIR="$ROOT_DIR/.wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs"
    94	mkdir -p "$PROOFS_DIR"
    95	
    96	{
    97	  echo "::group::Migration boundary terms (FR-020)"
    98	  MIG_PATTERN='\\b(cisco|crosswork|nso|cnc|proprietary\\s+ned|proprietary\\s+neds|ai-network-services-devnet-2606|devnet-2606)\\b'
    99	  MIG_MATCHES=$(search_all "$MIG_PATTERN")
   100	  MIG_VIOLATIONS=$(filter_allowed "$MIG_MATCHES")
   101	  if [[ -n "$MIG_VIOLATIONS" ]]; then
   102	    echo "::error title=Migration boundary term(s) found outside allowed contexts::${MIG_VIOLATIONS}"
   103	    fail=1
   104	  fi
   105	  echo "::endgroup::"
   106	
   107	  echo "::group::Visualization boundary terms (FR-032)"
   108	  VIS_PATTERN='\\b(sr[ -]?linux|nokia_srlinux)\\b|ghcr\\.io/nokia/srlinux'
   109	  VIS_MATCHES=$(search_all "$VIS_PATTERN")
   110	  VIS_VIOLATIONS=$(filter_allowed "$VIS_MATCHES")
   111	  if [[ -n "$VIS_VIOLATIONS" ]]; then
   112	    echo "::error title=SR Linux mention(s) found outside allowed contexts::${VIS_VIOLATIONS}"
   113	    fail=1
   114	  fi
   115	  echo "::endgroup::"
   116	
   117	  echo "::group::Placement boundary (FR-023): Compose/standalone indicators"
   118	  PL_PATTERN='(docker-compose|docker\\s+compose|compose\\.ya?ml|standalone\\s+container|standalone\\s+deployment)'
   119	  PL_MATCHES=$(search_all "$PL_PATTERN")
   120	  PL_VIOLATIONS=$(filter_allowed "$PL_MATCHES")
   121	  if [[ -n "$PL_VIOLATIONS" ]]; then
   122	    echo "::error title=Disallowed placement mention(s) found outside allowed contexts::${PL_VIOLATIONS}"
   123	    fail=1
   124	  fi
   125	  echo "::endgroup::"
   126	
   127	  echo "::group::Visualization reference repository exception (srl-telemetry-lab only in allowed contexts)"
   128	  SRLTL_PATTERN='srl-telemetry-lab'
   129	  SRLTL_MATCHES=$(search_all "$SRLTL_PATTERN" | sed -E 's#^\./##')
   130	  # Allowed contexts: spec.md Scope, SC-010 in spec.md, specs/**/research.md, REVERSE.md, README presentation-only line
   131	  ALLOWED_SPECS_SCOPE=$(awk '/^##\s+Scope and interpretation/{print NR; exit}' specs/001-ainetops-sonic-evpn-fabric/spec.md || true)
   132	  ALLOWED_SPECS_SCOPE_END=$(awk 'p && /^##\s+/{print NR; exit} /^##\s+Scope and interpretation/{p=1}' specs/001-ainetops-sonic-evpn-fabric/spec.md || true)
   133	  ALLOWED_SC010_START=$(awk '/^- \*\*SC-010\*\*/{print NR; exit}' specs/001-ainetops-sonic-evpn-fabric/spec.md || true)
   134	  ALLOWED_SC010_END=$(awk -v s="$ALLOWED_SC010_START" 'NR > s && /^- \*\*SC-/{print NR; exit}' specs/001-ainetops-sonic-evpn-fabric/spec.md || true)
   135	  if [[ -z "$ALLOWED_SC010_END" ]]; then ALLOWED_SC010_END=999999; fi
   136	  SRLTL_VIOLATIONS=$(printf "%s" "${SRLTL_MATCHES}" \
   137	    | awk -F: -v s="$ALLOWED_SPECS_SCOPE" -v e="$ALLOWED_SPECS_SCOPE_END" -v s2="$ALLOWED_SC010_START" -v e2="$ALLOWED_SC010_END" '!( $1 ~ /^specs\/001-ainetops-sonic-evpn-fabric\/spec.md$/ && ( ($2>=s && $2<=e) || ($2>=s2 && $2<=e2) ) )' \
   138	    | neg_filter '^specs/.+/research\.md:' \
   139	    | neg_filter '(^|.*/)REVERSE\.md:' \
   140	    | neg_filter '^README\.md:.*visualization/presentation reference only' \
   141	    | neg_filter '^\.github/workflows/denylist\.yml:' \
   142	    | neg_filter '^scripts/ci/denylist_policy\.sh:' \
   143	    | neg_filter '^scripts/ci/denylist_runtime_scan\.sh:' \
   144	    | neg_filter '^tests/integration/cycles_runner\.sh:' || true)
   145	  if [[ -n "$SRLTL_VIOLATIONS" ]]; then
   146	    echo "::error title=srl-telemetry-lab mention(s) found outside allowed contexts::${SRLTL_VIOLATIONS}"
   147	    fail=1
   148	  fi
   149	  echo "::endgroup::"
   150	
   151	  if [[ "$fail" -ne 0 ]]; then
   152	    echo "Deny-list checks failed" >&2
   153	    exit 1
   154	  fi
   155	  echo "All deny-list checks passed"
   156	} 2>&1 | tee "$PROOFS_DIR/denylist.run.log"
