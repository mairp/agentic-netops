     2	package render
     3	
     4	import (
     5		"sort"
     6	
     7		"github.com/mairp/ainetops/pkg/model"
     8	)
     9	
    10	// RenderBGP renders BGP global and neighbor config preferring OpenConfig paths.
    11	func RenderBGP(global model.BGPGlobal, neighbors []model.BGPNeighbor) map[string]any {
    12		res := map[string]any{}
    13		base := "/network-instances/network-instance"
    14		// We model default NI for BGP global in OpenConfig; fall back to SONiC-native for gaps
    15		res[base] = map[string]any{"name": "DEFAULT", "protocols": map[string]any{"bgp": map[string]any{
    16			"asn": global.ASN,
    17			"router-id": global.RouterID,
    18		}}}
    19		// Neighbors under OpenConfig
    20		names := make([]string, 0, len(neighbors))
    21		for _, n := range neighbors { names = append(names, n.NeighborAddress) }
    22		sort.Strings(names)
    23		neis := make([]map[string]any, 0, len(names))
    24		byKey := map[string]model.BGPNeighbor{}
    25		for _, n := range neighbors { byKey[n.NeighborAddress] = n }
    26		for _, k := range names {
    27			n := byKey[k]
    28			nei := map[string]any{
    29				"neighbor-address": n.NeighborAddress,
    30				"peer-as":        n.PeerASN,
    31			}
    32			if n.EVPN {
    33				nei["afi-safi"] = []string{"l2vpn-evpn"}
    34			}
    35			neis = append(neis, nei)
    36		}
    37		res[base+"/protocols/bgp/neighbors"] = neis
    38		return res
    39	}
