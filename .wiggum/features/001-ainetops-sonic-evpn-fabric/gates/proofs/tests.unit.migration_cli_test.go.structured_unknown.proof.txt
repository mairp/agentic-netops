1: // Note: Some environments do not build the CLI during unit tests; we keep a smoke test
2: // that simply checks the binary path exists when built. Core behavior is validated via the library.
3: func TestMigrationCLI_StructuredUnknownField(t *testing.T) {
4: 	bad := `{"serviceId":"svcX","type":"VPLS","tenant":"A","rdRt":{"rd":"65000:1","importRT":["65000:1"],"exportRT":["65000:1"]},"l2vni":10001,"endpoints":[{"node":"leaf01","attachment":"c1","vlan":10},{"node":"leaf02","attachment":"c2","vlan":10}],"unknown":"field"}`
5: 	_, err := migration.ParseStrictBatch([]byte(bad))
6: 	if err == nil { t.Fatalf("expected parse error due to unknown field") }
7: 	msg := migration.MarshalError(err)
8: 	// Ensure structured JSON with causes and the unknown field message.
9: 	var m map[string]any
10: 	if json.Unmarshal([]byte(msg), &m) != nil { t.Fatalf("not JSON: %s", msg) }
11: 	if m["error"] != "validation" { t.Fatalf("missing error key: %v", m) }
12: 	causes, ok := m["causes"].([]any)
13: 	if !ok || len(causes) == 0 { t.Fatalf("missing causes: %v", m) }
14: 	joined := msg
15: 	if !strings.Contains(joined, "unknown field") { t.Fatalf("expected unknown field in causes: %s", msg) }
16: }

18: func TestMigrationCLI_ExecutesAndRejectsUnknown(t *testing.T) {
19: 	// Try to run the CLI with go run so the test does not depend on a prebuilt binary.
20: 	bad := `{"serviceId":"svcX","type":"VPLS","tenant":"A","rdRt":{"rd":"65000:1","importRT":["65000:1"],"exportRT":["65000:1"]},"l2vni":10001,"endpoints":[{"node":"leaf01","attachment":"c1","vlan":10},{"node":"leaf02","attachment":"c2","vlan":10}],"unknown":"field"}`
21: 	cmd := exec.Command("go", "run", "./cmd/migration-translator")
22: 	cmd.Dir = filepath.Join("..", "..")
23: 	cmd.Stdin = bytes.NewBufferString(bad)
24: 	out, err := cmd.CombinedOutput()
25: 	if err == nil { t.Fatalf("expected non-zero exit; output=%s", string(out)) }
26: 	// Expect structured JSON on stderr; CombinedOutput merges streams; check JSON presence and absence of YAML spec.
27: 	if !strings.Contains(string(out), "\"error\": \"validation\"") { t.Fatalf("missing structured error: %s", string(out)) }
28: 	if strings.Contains(string(out), "spec:") { t.Fatalf("unexpected YAML output on parse failure: %s", string(out)) }
29: }
