     1	name: Denylist Scan
     2	
     3	on:
     4	  push:
     5	    branches: [ main, master ]
     6	  pull_request:
     7	
     8	jobs:
     9	  denylist:
    10	    runs-on: ubuntu-latest
    11	    steps:
    12	      - name: Checkout
    13	        uses: actions/checkout@v4
    14	
    15	      - name: Install ripgrep and coreutils
    16	        run: |
    17	          sudo apt-get update
    18	          sudo apt-get install -y ripgrep coreutils
    19	
    20	      - name: Run repository-wide deny-list checks (SC-010, FR-020, FR-023, FR-032)
    21	        shell: bash
    22	        run: |
    23	          set -euo pipefail
    24	
    25	          fail=0
    26	
    27	          # Helper: filter allowed contexts from ripgrep matches
    28	          filter_allowed() {
    29	            local matches="$1"
    30	            # Allow citations in research.md and REVERSE.md
    31	            matches=$(printf "%s" "${matches}" | rg -v 'specs/.+/research\.md:' | rg -v '^REVERSE\.md:' || true)
    32	
    33	            # Allow mentions in spec.md "Scope and interpretation" section and SC-010 success criterion only
    34	            # Compute line ranges for the sections in the pinned spec
    35	            local spec_path="specs/001-ainetops-sonic-evpn-fabric/spec.md"
    36	            if [[ -f "$spec_path" ]]; then
    37	              local start end sc010_start sc010_end
    38	              start=$(awk '/^##\s+Scope and interpretation/{print NR; exit}' "$spec_path" || true)
    39	              end=$(awk 'p && /^##\s+/{print NR; exit} /^##\s+Scope and interpretation/{p=1}' "$spec_path" || true)
    40	              sc010_start=$(awk '/^- \*\*SC-010\*\*/{print NR; exit}' "$spec_path" || true)
    41	              sc010_end=$(awk -v s="$sc010_start" 'NR > s && /^- \*\*SC-/{print NR; exit}' "$spec_path" || true)
    42	              if [[ -z "$sc010_end" ]]; then sc010_end=999999; fi
    43	              if [[ -n "${start}" && -n "${end}" ]]; then
    44	                # Remove matches within allowed line ranges in spec.md
    45	                matches=$(printf "%s" "${matches}" | awk -v s="$start" -v e="$end" -v s2="$sc010_start" -v e2="$sc010_end" -F: '!( $1=="'"$spec_path"'" && ( ($2>=s && $2<=e) || ($2>=s2 && $2<=e2) ) )')
    46	              fi
    47	            fi
    48	
    49	            # Do NOT globally allow srl-telemetry-lab — handle it in its own scoped check below
    50	
    51	            printf "%s" "${matches}"
    52	          }
    53	
    54	          echo "::group::Migration boundary terms (FR-020)"
    55	          MIG_PATTERN='\\b(cisco|crosswork|nso|cnc|proprietary\\s+ned|proprietary\\s+neds|ai-network-services-devnet-2606|devnet-2606)\\b'
    56	          MIG_MATCHES=$(rg -i -n --hidden --no-ignore -g '!*/.git/*' -e "$MIG_PATTERN" . || true)
    57	          MIG_VIOLATIONS=$(filter_allowed "$MIG_MATCHES")
    58	          if [[ -n "$MIG_VIOLATIONS" ]]; then
    59	            echo "::error title=Migration boundary term(s) found outside allowed contexts::${MIG_VIOLATIONS}"
    60	            fail=1
    61	          fi
    62	          echo "::endgroup::"
    63	
    64	          echo "::group::Visualization boundary terms (FR-032)"
    65	          VIS_PATTERN='\\b(sr[ -]?linux|nokia_srlinux)\\b|ghcr\\.io/nokia/srlinux'
    66	          VIS_MATCHES=$(rg -i -n --hidden --no-ignore -g '!*/.git/*' -e "$VIS_PATTERN" . || true)
    67	          VIS_VIOLATIONS=$(filter_allowed "$VIS_MATCHES")
    68	          if [[ -n "$VIS_VIOLATIONS" ]]; then
    69	            echo "::error title=SR Linux mention(s) found outside allowed contexts::${VIS_VIOLATIONS}"
    70	            fail=1
    71	          fi
    72	          echo "::endgroup::"
    73	
    74	          echo "::group::Placement boundary (FR-023): Compose/standalone indicators"
    75	          PL_PATTERN='(docker-compose|docker\\s+compose|compose\\.ya?ml|standalone\\s+container|standalone\\s+deployment)'
    76	          PL_MATCHES=$(rg -i -n --hidden --no-ignore -g '!*/.git/*' -e "$PL_PATTERN" . || true)
    77	          PL_VIOLATIONS=$(filter_allowed "$PL_MATCHES")
    78	          if [[ -n "$PL_VIOLATIONS" ]]; then
    79	            echo "::error title=Disallowed placement mention(s) found outside allowed contexts::${PL_VIOLATIONS}"
    80	            fail=1
    81	          fi
    82	          echo "::endgroup::"
    83	
    84	          echo "::group::Visualization reference repository exception (srl-telemetry-lab only in allowed contexts)"
    85	          SRLTL_PATTERN='srl-telemetry-lab'
    86	          SRLTL_MATCHES=$(rg -i -n --hidden --no-ignore -g '!*/.git/*' -e "$SRLTL_PATTERN" . || true)
    87	          # Allowed contexts: spec.md Scope and interpretation section, SC-010 in spec.md, specs/**/research.md, REVERSE.md, and README.md line stating presentation-only reference
    88	          ALLOWED_SPECS_SCOPE=$(awk '/^##\s+Scope and interpretation/{print NR; exit}' specs/001-ainetops-sonic-evpn-fabric/spec.md || true)
    89	          ALLOWED_SPECS_SCOPE_END=$(awk 'p && /^##\s+/{print NR; exit} /^##\s+Scope and interpretation/{p=1}' specs/001-ainetops-sonic-evpn-fabric/spec.md || true)
    90	          # Also allow SC-010 success criterion lines in spec.md
    91	          ALLOWED_SC010_START=$(awk '/^- \*\*SC-010\*\*/{print NR; exit}' specs/001-ainetops-sonic-evpn-fabric/spec.md || true)
    92	          ALLOWED_SC010_END=$(awk -v s="$ALLOWED_SC010_START" 'NR > s && /^- \*\*SC-/{print NR; exit}' specs/001-ainetops-sonic-evpn-fabric/spec.md || true)
    93	          if [[ -z "$ALLOWED_SC010_END" ]]; then ALLOWED_SC010_END=999999; fi
    94	          SRLTL_VIOLATIONS=$(printf "%s" "${SRLTL_MATCHES}" \
    95	            | awk -F: -v s="$ALLOWED_SPECS_SCOPE" -v e="$ALLOWED_SPECS_SCOPE_END" -v s2="$ALLOWED_SC010_START" -v e2="$ALLOWED_SC010_END" '!( $1 ~ /^specs\/001-ainetops-sonic-evpn-fabric\/spec.md$/ && ( ($2>=s && $2<=e) || ($2>=s2 && $2<=e2) ) )' \
    96	            | rg -v '^specs/.+/research\.md:' \
    97	            | rg -v '^REVERSE\.md:' \
    98	            | rg -v '^README\.md:.*visualization/presentation reference only' || true)
    99	          if [[ -n "$SRLTL_VIOLATIONS" ]]; then
   100	            echo "::error title=srl-telemetry-lab mention(s) found outside allowed contexts::${SRLTL_VIOLATIONS}"
   101	            fail=1
   102	          fi
   103	          echo "::endgroup::"
   104	
   105	          if [[ "$fail" -ne 0 ]]; then
   106	            echo "Deny-list checks failed" >&2
   107	            exit 1
   108	          fi
   109	          echo "All deny-list checks passed"
