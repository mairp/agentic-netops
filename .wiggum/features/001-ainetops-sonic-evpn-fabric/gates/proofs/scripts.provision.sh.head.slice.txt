     1	#!/usr/bin/env bash
     2	set -euo pipefail
     3	
     4	# AINETOPS SONiC EVPN/VXLAN Fabric — provision script (Phase 2 updates)
     5	# Sole implementation of environment creation/convergence per contracts/crd-api.md
     6	
     7	SCRIPT_DIR=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" &>/dev/null && pwd)
     8	REPO_ROOT=$(cd -- "${SCRIPT_DIR}/.." && pwd)
     9	LIB_DIR="${SCRIPT_DIR}/lib"
    10	
    11	# shellcheck source=./lib/preflight.sh
    12	if [[ -f "${LIB_DIR}/preflight.sh" ]]; then
    13	  # Preflight validates versions.lock.yaml, host resources, privileges, MTU, address overlaps, KVM
    14	  source "${LIB_DIR}/preflight.sh"
    15	  preflight::run "$@"
    16	fi
    17	
    18	# Ordered phases (Phase 3): verify pins, validate CRDs, create Kind, attach mgmt net, run capability gate
    19	if [[ -f "${REPO_ROOT}/Makefile" ]]; then
    20	  make -C "${REPO_ROOT}" verify-compat
    21	fi
    22	
    23	# Ensure external management network exists (idempotent)
    24	if command -v docker >/dev/null 2>&1; then
    25	  docker network inspect ainetops-mgmt >/dev/null 2>&1 || docker network create --label ainetops.owner=ainetops ainetops-mgmt
    26	fi
    27	
    28	# Ensure Kind cluster exists and is using pinned image; attach nodes to mgmt network
    29	if [[ -x "${LIB_DIR}/kind.sh" ]]; then
    30	  "${LIB_DIR}/kind.sh" ensure
    31	  "${LIB_DIR}/kind.sh" attach-mgmt
    32	  "${LIB_DIR}/kind.sh" verify-context
    33	else
    34	  echo "[provision] WARN: kind helper not found; skipping Kind cluster ensure" >&2
    35	fi
    36	
    37	# Deploy lab topology with containerlab if available
    38	if [[ -x "${LIB_DIR}/containerlab.sh" ]]; then
    39	  "${LIB_DIR}/containerlab.sh" deploy || { echo "[provision] containerlab deploy failed" >&2; exit 1; }
    40	  # Optional inspect step for diagnostics
    41	  "${LIB_DIR}/containerlab.sh" inspect || true
    42	else
    43	  echo "[provision] WARN: containerlab helper not found; skipping lab deploy" >&2
    44	fi
    45	
    46	# Install least-privilege RBAC base
    47	if [[ -x "${LIB_DIR}/rbac.sh" ]]; then
    48	  "${LIB_DIR}/rbac.sh"
    49	fi
    50	# Install pinned Kubenet/KUID and SDC into Kind and wait basic readiness
    51	if [[ -x "${REPO_ROOT}/deploy/kubenet/install.sh" ]]; then
    52	  "${REPO_ROOT}/deploy/kubenet/install.sh"
    53	fi
    54	if [[ -x "${REPO_ROOT}/deploy/sdc/install.sh" ]]; then
    55	  "${REPO_ROOT}/deploy/sdc/install.sh"
    56	fi
    57	
    58	# Build, load, and deploy provider and srv6-controller images into Kind (T041)
    59	if command -v docker >/dev/null 2>&1 && command -v kind >/dev/null 2>&1 && command -v kubectl >/dev/null 2>&1; then
    60	  echo "[provision] building controller images"
    61	  docker build -t ainetops-sonic-provider:dev -f "${REPO_ROOT}/cmd/sonic-provider/Dockerfile" "${REPO_ROOT}"
    62	  docker build -t ainetops-srv6-controller:dev -f "${REPO_ROOT}/cmd/srv6-controller/Dockerfile" "${REPO_ROOT}"
    63	  echo "[provision] loading images into Kind"
    64	  kind load docker-image ainetops-sonic-provider:dev --name "${AINETOPS_CLUSTER_NAME:-ainetops}" || true
    65	  kind load docker-image ainetops-srv6-controller:dev --name "${AINETOPS_CLUSTER_NAME:-ainetops}" || true
    66	  echo "[provision] deploying controllers"
    67	  # Use dev images by overriding the container image in manifests
    68	  kubectl --context "kind-${AINETOPS_CLUSTER_NAME:-ainetops}" -n ainetops-system apply -f "${REPO_ROOT}/deploy/ainetops/manifests/provider.yaml"
    69	  kubectl --context "kind-${AINETOPS_CLUSTER_NAME:-ainetops}" -n ainetops-system apply -f "${REPO_ROOT}/deploy/ainetops/manifests/srv6-controller.yaml"
    70	  kubectl --context "kind-${AINETOPS_CLUSTER_NAME:-ainetops}" -n ainetops-system set image deploy/ainetops-sonic-provider provider=ainetops-sonic-provider:dev || true
    71	  kubectl --context "kind-${AINETOPS_CLUSTER_NAME:-ainetops}" -n ainetops-system set image deploy/ainetops-srv6-controller srv6-controller=ainetops-srv6-controller:dev || true
    72	  echo "[provision] waiting for controller pods ready"
    73	  kubectl --context "kind-${AINETOPS_CLUSTER_NAME:-ainetops}" -n ainetops-system rollout status deploy/ainetops-sonic-provider --timeout=120s || true
    74	  kubectl --context "kind-${AINETOPS_CLUSTER_NAME:-ainetops}" -n ainetops-system rollout status deploy/ainetops-srv6-controller --timeout=120s || true
    75	  # Capture independent observation proof
    76	  mkdir -p "${REPO_ROOT}/.wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs"
    77	  kubectl --context "kind-${AINETOPS_CLUSTER_NAME:-ainetops}" -n ainetops-system get deploy,po,svc -o wide \
    78	    | nl -ba > "${REPO_ROOT}/.wiggum/features/001-ainetops-sonic-evpn-fabric/gates/proofs/kubectl-get-ainetops-system.txt"
    79	fi
    80	
    81	# Seed SDC schema/profile/discovery (address-based); tolerate apply idempotence
    82	if command -v kubectl >/dev/null 2>&1; then
    83	  CTX="kind-${AINETOPS_CLUSTER_NAME:-ainetops}"
    84	  kubectl --context "$CTX" apply -f "${REPO_ROOT}/deploy/sdc/seed/sonic-schema.yaml"
    85	  kubectl --context "$CTX" apply -f "${REPO_ROOT}/deploy/sdc/seed/discovery-rule.yaml"
    86	fi
    87	
    88	# Run lab capability qualification (fast profile can be selected via env)
    89	if [[ -x "${LIB_DIR}/qualify.sh" ]]; then
    90	  "${LIB_DIR}/qualify.sh" || { echo "[provision] capability gate failed" >&2; exit 1; }
    91	fi
    92	
    93	echo "[provision] Phase 3: pins verified, CRDs validated, Kind ensured/attached, apps installed, seed applied, capability gate executed."
