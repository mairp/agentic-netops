    13	func merge(m map[string]any, more ...map[string]any) map[string]any {
    14		for _, mm := range more { for k, v := range mm { m[k] = v } }
    15		return m
    16	}

    26		)

    31			render.RenderSIDList("spine-path", []string{"2001:db8:1::a", "2001:db8:1::b"}),

    33		)
